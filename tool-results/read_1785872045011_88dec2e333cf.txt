    88→function StatusBadge({ status, until }: { status: ModMember["status"]; until: number | null }) {
    89→  if (status === "banned") return <span className="text-[9px] font-bold px-1.5 py-0.5 rounded bg-red-500/15 text-red-400 border border-red-500/20">BANNED</span>;
    90→  if (status === "suspended") return <span className="text-[9px] font-bold px-1.5 py-0.5 rounded bg-amber-500/15 text-amber-400 border border-amber-500/20">SUSPENDED{until ? ` · ${new Date(until).toLocaleDateString()}` : ""}</span>;
    91→  return <span className="text-[9px] font-bold px-1.5 py-0.5 rounded bg-emerald-500/15 text-emerald-400 border border-emerald-500/20">ACTIVE</span>;
    92→}

   136→// Main ModPanel Component
   137→// ═══════════════════════════════════════════════════════════════
   138→
   139→export default function ModPanel() {
   140→  const user = useAppStore((s) => s.user);
   141→  const navigate = useAppStore((s) => s.navigate);
   142→
   143→  const [roleChecked, setRoleChecked] = useState(false);
   144→  const [isMod, setIsMod] = useState(false);
   145→  const [tab, setTab] = useState<Tab>("overview");
   146→  const [collapsedSections, setCollapsedSections] = useState<Record<string, boolean>>({});
   147→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   148→
   149→  const [members, setMembers] = useState<ModMember[] | null>(null);
   150→  const [reports, setReports] = useState<ReportRow[] | null>(null);
   151→  const [warns, setWarns] = useState<WarnRow[] | null>(null);
   152→  const [logs, setLogs] = useState<LogRow[] | null>(null);
   153→  const [overview, setOverview] = useState<Overview | null>(null);
   154→  const [bump, setBump] = useState(0);
   155→
   156→  useEffect(() => {
   157→    let cancel = false;
   158→    Promise.resolve().then(async () => {
   159→      if (!user) return "member" as const;
   160→      return getMyRole(user.username);
   161→    }).then((r) => { if (!cancel) { setIsMod(r === "mod"); setRoleChecked(true); } });
   162→    return () => { cancel = true; };
   163→  }, [user]);
   164→
   165→  const authQS = useMemo(() => user ? `u=${encodeURIComponent(user.username)}&token=${encodeURIComponent(getDeviceToken())}` : "", [user]);
   166→
   167→  useEffect(() => {
   168→    if (!isMod || !user) return;
   169→    let cancel = false;
   170→    const get = (path: string) => fetch(`${path}?${authQS}`, { cache: "no-store" }).then((r) => r.json()).catch(() => ({ ok: false }));
   171→    Promise.all([
   172→      get("/api/mod/moderation"), get("/api/mod/reports"), get("/api/mod/warns"), get("/api/mod/logs"), get("/api/mod/overview"),
   173→    ]).then(([m, r, w, l, o]) => {
   174→      if (cancel) return;
   175→      if (m.ok) setMembers(m.members);
   176→      if (r.ok) setReports(r.reports);
   177→      if (w.ok) setWarns(w.warns);
   178→      if (l.ok) setLogs(l.logs);
   179→      if (o.ok) setOverview({ totalComments: o.totalComments, totalCollections: o.totalCollections, totalViews: o.totalViews, onlineNow: o.onlineNow });
   180→    });
   181→    return () => { cancel = true; };
   182→  }, [isMod, user, authQS, bump]);
   183→
   184→  const refresh = () => setBump((b) => b + 1);
   185→
   186→  const act = useCallback(async (targetUsername: string, action: "ban" | "unban" | "suspend" | "unsuspend", opts?: { reason?: string; hours?: number }) => {
   187→    if (!user) return;
   188→    await fetch("/api/mod/moderation", {
   189→      method: "POST", headers: { "content-type": "application/json" },
   190→      body: JSON.stringify({ u: user.username, token: getDeviceToken(), targetUsername, action, ...opts }),
   191→    }).catch(() => {});
   192→    refresh();
   193→  }, [user]);
   194→
   195→  const warnUser = useCallback(async (targetUsername: string, reason: string) => {
   196→    if (!user) return { ok: false };
   197→    const res = await fetch("/api/mod/warns", {
   198→      method: "POST", headers: { "content-type": "application/json" },
   199→      body: JSON.stringify({ u: user.username, token: getDeviceToken(), targetUsername, reason }),
   200→    }).then((r) => r.json()).catch(() => ({ ok: false }));
   201→    refresh();
   202→    return res;
   203→  }, [user]);
   204→
   205→  const resolveReport = async (id: string, status: "open" | "resolved") => {
   206→    if (!user) return;
   207→    await fetch("/api/mod/reports", {
   208→      method: "PATCH", headers: { "content-type": "application/json" },
   209→      body: JSON.stringify({ u: user.username, token: getDeviceToken(), id, status }),
   210→    }).catch(() => {});
   211→    refresh();
   212→  };
   213→
   214→  // ─── Auth gates ───
   215→  if (!user) return <div className="max-w-2xl mx-auto py-24 text-center text-[#949ba4]" style={{ fontFamily: FONT }}><p>Sign in to access Moderator Tools.</p></div>;
   216→  if (!roleChecked) return <div className="flex items-center justify-center py-24"><div className="w-6 h-6 rounded-full border-2 border-[#949ba4]/30 border-t-[#f2f3f5] animate-spin" /></div>;
   217→  if (!isMod) return <div className="max-w-2xl mx-auto py-24 text-center text-[#949ba4]" style={{ fontFamily: FONT }}><p>You don&apos;t have moderator access.</p></div>;
   218→
   219→  // ─── Computed values ───
   220→  const bannedMembers = (members || []).filter((m) => m.status === "banned");
   221→  const suspendedMembers = (members || []).filter((m) => m.status === "suspended");
   222→  const openReports = (reports || []).filter((r) => r.status === "open");
   223→  const now = Date.now();
   224→  const newThisWeek = (members || []).filter((m) => now - new Date(m.createdAt).getTime() <= 7 * 86400000).length;
   225→  const newThisMonth = (members || []).filter((m) => now - new Date(m.createdAt).getTime() <= 30 * 86400000).length;
   226→  const adminUsers = (members || []).filter((m) => m.isMod || m.isOwner).length;
   227→  const pageLabel = NAV.flatMap((s) => s.items).find((i) => i.id === tab)?.label || "Overview";
   228→  const navFrame = frameSrcOf(user.avatarFrame);
   229→
   230→  return (
   231→    <div className="-mx-4 lg:-mx-8 min-h-screen bg-[#111214] text-[#f2f3f5] flex" style={{ fontFamily: FONT }}>
   232→      {/* ═══ Mobile overlay ═══ */}
   233→      {mobileMenuOpen && <div className="fixed inset-0 z-[60] bg-black/60 lg:hidden" onClick={() => setMobileMenuOpen(false)} />}
   234→
   235→      {/* ═══ SIDEBAR (Discord-style) ═══ */}
   236→      <aside className={`w-[260px] shrink-0 bg-[#18181b] border-r border-white/[0.04] p-4 hidden md:flex flex-col fixed md:sticky top-0 bottom-0 z-[70] md:z-auto transition-transform ${mobileMenuOpen ? "translate-x-0" : "-translate-x-full"} md:translate-x-0`}>
   237→        {/* ── User Profile Card ── */}
   238→        <div className="flex items-center gap-3 mb-6 p-2.5 rounded-xl bg-[#232428] border border-white/[0.06]">
   239→          <div className="relative w-11 h-11 shrink-0">
   240→            <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-[78%] h-[78%] z-[1]">
   241→              <div className="w-full h-full rounded-full flex items-center justify-center text-sm font-black overflow-hidden" style={{ backgroundColor: (user.avatarColor || "#7c3aed") + "44", color: user.avatarColor || "#7c3aed" }}>
   242→                {user.avatarImage ? <img src={user.avatarImage} alt="" className="w-full h-full object-cover" loading="lazy" /> : (user.avatarEmoji || user.name.charAt(0)).toUpperCase()}
   243→              </div>
   244→            </div>
   245→            {navFrame && <img src={navFrame} alt="" loading="lazy" className="pointer-events-none absolute inset-0 w-full h-full max-w-none select-none z-[2]" />}
   246→            {!navFrame && <div className="absolute inset-0 rounded-full border border-white/10 z-[1] pointer-events-none" />}
   247→          </div>
   248→          <div className="min-w-0">
   249→            <p className="font-bold text-sm text-[#f2f3f5] leading-none truncate">{user.name || user.username}</p>
   250→            <span className="inline-flex items-center gap-1.5 mt-1.5 text-[10px] font-bold px-2 py-0.5 rounded bg-[#23a559]/15 text-[#23a559] border border-[#23a559]/20">
   251→              <svg className="w-2.5 h-2.5" viewBox="0 0 10 10" fill="currentColor"><circle cx="5" cy="5" r="5" /></svg>
   252→              MOD
   253→            </span>
   254→          </div>
   255→        </div>
   256→
   257→        {/* ── Navigation sections ── */}
   258→        {NAV.map((group) => (
   259→          <div key={group.section} className="mb-4">
   260→            <button
   261→              onClick={() => setCollapsedSections((s) => ({ ...s, [group.section]: !s[group.section] }))}
   262→              className="w-full flex items-center justify-between text-[11px] font-bold text-[#949ba4] uppercase tracking-[0.05em] px-2 py-1.5 hover:text-[#f2f3f5] transition-colors"
   263→            >
   264→              {group.section}
   265→              <svg className={`w-[12px] h-[12px] transition-transform ${collapsedSections[group.section] ? "-rotate-90" : ""}`} width={12} height={12} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2} strokeLinecap="round">
   266→                <path d={SVG_PATHS.chevron} />
   267→              </svg>
   268→            </button>
   269→            {!collapsedSections[group.section] && (
   270→              <nav className="space-y-0.5">
   271→                {group.items.map((item) => (
   272→                  <button
   273→                    key={item.id}
   274→                    onClick={() => { setTab(item.id); setMobileMenuOpen(false); }}
   275→                    className={`w-full flex items-center gap-2.5 px-2.5 py-2 rounded-lg text-[13px] font-semibold transition-colors ${
   276→                      tab === item.id
   277→                        ? "bg-[#404249] text-[#f2f3f5]"
   278→                        : "text-[#949ba4] hover:text-[#f2f3f5] hover:bg-[#2b2d31]"
   279→                    }`}
   280→                  >
   281→                    <span className={tab === item.id ? "text-[#5865f2]" : ""}><NavIcon name={item.icon} size={17} /></span>
   282→                    {item.label}
   283→                  </button>
   284→                ))}
   285→              </nav>
   286→            )}
   287→          </div>
   288→        ))}
   289→
   290→        {/* ── Back to site ── */}
   291→        <div className="mt-auto pt-4 border-t border-white/[0.06]">
   292→          <button
   293→            onClick={() => navigate({ page: "home" })}
   294→            className="w-full flex items-center gap-2.5 px-2.5 py-2 rounded-lg text-[13px] font-semibold text-[#949ba4] hover:text-[#f2f3f5] hover:bg-[#2b2d31] transition-colors"
   295→          >
   296→            <NavIcon name="history" size={17} />
   297→            Back to Site
   298→          </button>
   299→        </div>
   300→      </aside>
   301→
   302→      {/* ═══ CONTENT AREA ═══ */}
   303→      <div className="flex-1 min-w-0 p-4 sm:p-6 lg:p-8">
   304→        {/* ── Header with title + refresh + mobile toggle ── */}
   305→        <div className="flex items-center justify-between mb-6">
   306→          <div className="flex items-center gap-3">
   307→            {/* Mobile menu toggle */}
   308→            <button onClick={() => setMobileMenuOpen((v) => !v)} className="md:hidden p-2 rounded-lg bg-[#232428] border border-white/[0.06]">
   309→              <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2} strokeLinecap="round">
   310→                <path d="M4 6h16M4 12h16M4 18h16" />
   311→              </svg>
   312→            </button>
   313→            <h1 className="text-2xl font-bold text-[#f2f3f5]">{pageLabel}</h1>
   314→          </div>
   315→          <button
   316→            onClick={refresh}
   317→            className="flex items-center gap-1.5 px-3.5 py-2 rounded-lg border border-white/[0.08] bg-[#2b2d31] text-xs font-bold text-[#949ba4] hover:text-[#f2f3f5] hover:bg-[#404249] transition-colors"
   318→          >
   319→            <NavIcon name="refresh" size={14} />
   320→            Refresh
   321→          </button>
   322→        </div>
   323→
   324→        {/* ═══ OVERVIEW PAGE ═══ */}
   325→        {tab === "overview" && (
   326→          <div className="space-y-6">
   327→            {/* Row 1: Primary User Metrics */}
   328→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   329→              <StatCard label="Total Users" value={members?.length ?? "—"} icon="users" color="#5865f2" />
   330→              <StatCard label="Active Sessions" value={overview?.onlineNow ?? "—"} icon="pulse" color="#23a559" />
   331→              <StatCard label="New This Week" value={newThisWeek} icon="trending" color="#f59e0b" />
   332→            </div>
   333→
   334→            {/* Row 2: Secondary User Metrics */}
   335→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   336→              <StatCard label="New This Month" value={newThisMonth} icon="trending" color="#06b6d4" />
   337→              <StatCard label="Admin Users" value={adminUsers} icon="shield" color="#9b59b6" />
   338→              <StatCard label="Banned Users" value={bannedMembers.length} icon="ban" color="#ed4245" />
   339→            </div>

   341→            {/* Content section */}

   506→function WarnsView({ warns, onWarn }: { warns: WarnRow[] | null; onWarn: (target: string, reason: string) => Promise<{ ok: boolean; error?: string }> }) {
   507→  const [target, setTarget] = useState("");
   508→  const [reason, setReason] = useState("");
   509→  const [err, setErr] = useState("");
   510→  const [busy, setBusy] = useState(false);
   511→
   512→  const submit = async () => {
   513→    if (!target.trim() || !reason.trim()) return;
   514→    setBusy(true); setErr("");
   515→    const res = await onWarn(target.trim(), reason.trim());
   516→    setBusy(false);
   517→    if (res.ok) { setTarget(""); setReason(""); } else setErr(res.error || "Failed to warn user");
   518→  };
   519→
   520→  return (
   521→    <div className="space-y-6">
   522→      <div className="rounded-xl border border-white/[0.06] bg-[#1e1f22] p-5">
   523→        <SectionLabel>WARN A USER</SectionLabel>
   524→        <div className="flex flex-col sm:flex-row gap-2">
   525→          <input value={target} onChange={(e) => setTarget(e.target.value)} placeholder="Username" className="flex-1 px-3 py-2.5 rounded-lg bg-[#111214] border border-white/[0.06] text-sm text-[#f2f3f5] placeholder-[#949ba4]/60 outline-none focus:border-[#f59e0b]/50 transition-colors" />
   526→          <input value={reason} onChange={(e) => setReason(e.target.value)} placeholder="Reason" className="flex-[2] px-3 py-2.5 rounded-lg bg-[#111214] border border-white/[0.06] text-sm text-[#f2f3f5] placeholder-[#949ba4]/60 outline-none focus:border-[#f59e0b]/50 transition-colors" />
   527→          <button onClick={submit} disabled={busy} className="px-4 py-2.5 rounded-lg text-sm font-bold bg-[#f59e0b] text-[#111214] disabled:opacity-50 hover:bg-[#fbbf24] transition-colors">{busy ? "..." : "Warn"}</button>
   528→        </div>
   529→        {err && <p className="text-xs text-[#ed4245] mt-2">{err}</p>}
   530→      </div>
   531→
   532→      <div>
   533→        <SectionLabel>WARNINGS ({warns?.length ?? 0})</SectionLabel>
   534→        {warns === null && <div className="flex items-center justify-center py-16"><div className="w-6 h-6 rounded-full border-2 border-[#949ba4]/30 border-t-[#f2f3f5] animate-spin" /></div>}
   535→        {warns !== null && warns.length === 0 && <p className="text-sm text-[#949ba4]/60">No warnings issued yet.</p>}
   536→        <div className="space-y-2">
   537→          {(warns || []).map((w) => (
   538→            <div key={w.id} className="rounded-xl border border-white/[0.06] bg-[#1e1f22] p-4 hover:border-white/[0.10] transition-colors">
   539→              <p className="text-sm text-[#f2f3f5]/90">@{w.username} — {w.reason}</p>
   540→              <p className="text-[11px] text-[#949ba4]/40 mt-1">by @{w.byUsername} · {new Date(w.createdAt).toLocaleString()}</p>
   541→            </div>
   542→          ))}
   543→        </div>
   544→      </div>
   545→    </div>
   546→  );
