    77→/**
    78→ * Hard ceiling on a provider promise.
    79→ *
    80→ * Several sources in the fan-out below accept no timeout option at all
    81→ * (fetchRawEpisodes, the animex slug->servers chain, the AniVault fetches,
    82→ * fetchAllAnimePaheSources), so a single slow upstream set the runtime for the
    83→ * whole route — measured at 93-180s against this function's 60s maxDuration,
    84→ * which meant Vercel killed it and NONE of the 17 providers reached the player.
    85→ * Promise.allSettled waits for the slowest member; this makes "slowest" bounded.
    86→ *
    87→ * Resolves to `fallback` on timeout rather than rejecting, so one dead provider
    88→ * is simply an empty slot instead of a lost response.
    89→ */
    90→function withTimeout<T>(p: Promise<T> | T, ms: number, fallback: T): Promise<T> {
    91→  return Promise.race([
    92→    Promise.resolve(p).catch(() => fallback),
    93→    new Promise<T>(resolve => setTimeout(() => resolve(fallback), ms)),
    94→  ]);
    95→}

   143→export async function GET(
   144→  _req: NextRequest,
   145→  { params }: { params: Promise<{ anilistId: string; episode: string }> }
   146→) {
   147→  const { anilistId, episode } = await params;
   148→  const id = parseInt(anilistId, 10);
   149→  const epNum = parseInt(episode, 10);
   150→  if (isNaN(id) || id <= 0) {
   151→    return NextResponse.json({ error: "Invalid anilistId" }, { status: 400 });
   152→  }
   153→
   154→  /**
   155→   * Provider group. Everything in one request measured 98-173s against a 60s
   156→   * maxDuration, so Vercel killed it and the player got NOTHING — capping the
   157→   * providers hard enough to fit meant silently dropping miruro/animex, the two
   158→   * highest-priority sources.
   159→   *
   160→   * Splitting instead: the client fires ?group=fast and ?group=slow in
   161→   * parallel, so each half gets its own 60s budget and merges as it lands.
   162→   * Nothing is dropped; the slow half just shows up later.
   163→   *
   164→   * ?group=all (the default) preserves the original single-request behaviour
   165→   * for any caller that isn't group-aware.
   166→   */
   167→  const group = (_req.nextUrl.searchParams.get("group") || "all").toLowerCase();
   168→  const wantFast = group === "all" || group === "fast";
   169→  const wantSlow = group === "all" || group === "slow";
   170→  // The slow half is alone in its request, so it can afford real patience.
   171→  const SLOW_CAP = group === "slow" ? 45000 : 12000;
   172→
   173→  try {
   174→  // ─── Resolve anime title from AniList (needed for AnixTV search) ────────────
   175→  // AnixTV's watch URL requires the anime title as a query param — without it,
   176→  // AnixTV can't find the anime and returns no iframe → no hindi servers.
   177→  let animeTitle = "Anime";
   178→  let animeTitles: { english?: string; romaji?: string; native?: string } = {};
   179→  let animeSeason = 1;  // hoisted out of try-block so AnixTV block can use it
   180→  try {
   181→    const titleRes = await Promise.race([
   182→      fetch("https://graphql.anilist.co", {
   183→        method: "POST",
   184→        headers: { "Content-Type": "application/json" },
   185→        body: JSON.stringify({
   186→          query: `query($id:Int){Media(id:$id,type:ANIME){id title{english romaji native}}}`,
   187→          variables: { id },
   188→        }),
   189→      }),
   190→      new Promise<Response | null>(r => setTimeout(() => r(null), 5000)),
   191→    ]);
   192→    if (titleRes && titleRes.ok) {
   193→      const titleData = await titleRes.json();
   194→      const t = titleData?.data?.Media?.title;
   195→      animeTitles = { english: t?.english, romaji: t?.romaji, native: t?.native };
   196→      animeTitle = t?.english || t?.romaji || t?.native || "Anime";
   197→    }
   198→
   199→    // Also fetch the anime's season + relations to determine the correct
   200→    // season number for AnixTV (e.g. Slime S4 → season=4, not 1)
   201→    //
   202→    // Two strategies:
   203→    //   1. Title heuristic: parse "Season N" / "S2" / "Part 2" from the title
   204→    //      (most reliable for sequels that include the season number in title)
   205→    //   2. AniList relation walk: count PREQUEL chain (any format — TV, ONA,
   206→    //      OVA, MOVIE, SPECIAL — AniList sometimes lists OVAs as the prequel
   207→    //      for sequel seasons, e.g. Slime S2 → "Visions of Coleus" OVA → S1)
   208→    try {
   209→      // Strategy 1: title heuristic
   210→      const titleForSeason = animeTitles.english || animeTitles.romaji || animeTitle;
   211→      const seasonMatch = titleForSeason.match(/\bSeason\s+(\d+)\b/i)
   212→                       || titleForSeason.match(/\bS(\d{1,2})\b\s*$/i);
   213→      if (seasonMatch) {
   214→        const parsed = parseInt(seasonMatch[1], 10);
   215→        if (parsed >= 1 && parsed <= 30) {
   216→          animeSeason = parsed;
   217→          console.log(`[Servers] AnixTV season from title: ${animeTitle} → season ${animeSeason}`);
   218→        }
   219→      }
   220→
   221→      // Strategy 2: AniList relation walk (if title heuristic didn't fire)
   222→      if (animeSeason === 1) {
   223→        const seasonRes = await Promise.race([
   224→          fetch("https://graphql.anilist.co", {
   225→            method: "POST",
   226→            headers: { "Content-Type": "application/json" },
   227→            body: JSON.stringify({
   228→              query: `query($id:Int){Media(id:$id,type:ANIME){id idMal season seasonYear format relations{edges{relationType node{id title{english romaji} format}}} } }`,
   229→              variables: { id },
   230→            }),
   231→          }),
   232→          new Promise<Response | null>(r => setTimeout(() => r(null), 5000)),
   233→        ]);
   234→        if (seasonRes && seasonRes.ok) {
   235→          const seasonData = await seasonRes.json();
   236→          const media = seasonData?.data?.Media;
   237→          if (media?.relations?.edges) {
   238→            // Include ALL formats in the prequel walk — AniList sometimes
   239→            // lists OVAs/specials/movies as the prequel for a sequel season.
   240→            const prequels = media.relations.edges.filter(
   241→              (e: any) => e.relationType === "PREQUEL"
   242→            );
   243→            if (prequels.length > 0) {
   244→              let count = 1;
   245→              let currentPrequel = prequels[0];
   246→              const visited = new Set<number>([id]);
   247→              for (let i = 0; i < 15; i++) {
   248→                if (!currentPrequel?.node?.id) break;
   249→                if (visited.has(currentPrequel.node.id)) break;
   250→                visited.add(currentPrequel.node.id);
   251→                try {
   252→                  const prequelRes = await fetch("https://graphql.anilist.co", {
   253→                    method: "POST",
   254→                    headers: { "Content-Type": "application/json" },
   255→                    body: JSON.stringify({
   256→                      query: `query($id:Int){Media(id:$id,type:ANIME){relations{edges{relationType node{id format title{english}}}} } }`,
   257→                      variables: { id: currentPrequel.node.id },
   258→                    }),
   259→                  });
   260→                  if (prequelRes.ok) {
   261→                    const prequelData = await prequelRes.json();
   262→                    const prequelEdges = prequelData?.data?.Media?.relations?.edges || [];
   263→                    const nextPrequel = prequelEdges.find(
   264→                      (e: any) => e.relationType === "PREQUEL"
   265→                    );
   266→                    if (nextPrequel) {
   267→                      count++;
   268→                      currentPrequel = nextPrequel;
   269→                    } else {
   270→                      count++;
   271→                      break;
   272→                    }
   273→                  } else break;
   274→                } catch { break; }
   275→              }
   276→              animeSeason = count;
   277→            }
   278→          }
   279→          console.log(`[Servers] AnixTV season from relations: ${animeTitle} → season ${animeSeason}`);
   280→        }
   281→      }
   282→    } catch { /* fallback to season 1 */ }
