    18→declare global {

    21→
    22→function loadDashJs(): Promise<void> {
    23→  return new Promise((resolve, reject) => {
    24→    if (window.dashjs) return resolve();
    25→    const existing = document.querySelector('script[src*="dash.mediaplayer"]') as HTMLScriptElement | null;
    26→    if (existing) {
    27→      existing.addEventListener("load", () => resolve());
    28→      existing.addEventListener("error", () => reject(new Error("dash.js load error")));
    29→      return;
    30→    }
    31→    const script = document.createElement("script");
    32→    script.src = "https://cdn.jsdelivr.net/npm/dashjs@4.7.4/dist/dash.mediaplayer.min.js";
    33→    script.async = true;
    34→    script.onload = () => resolve();
    35→    script.onerror = () => reject(new Error("Failed to load dash.js"));
    36→    document.head.appendChild(script);
    37→  });
    38→}

    40→function DashPlayer({
    41→  url,
    42→  subtitleTracks,
    43→  onEnded,
    44→  autoplay = true,
    45→}: {
    46→  url: string;
    47→  subtitleTracks?: Array<{ url: string; label: string; lang?: string }>;
    48→  onEnded?: () => void;
    49→  autoplay?: boolean;
    50→}) {
    51→  const videoRef = useRef<HTMLVideoElement>(null);
    52→  const playerRef = useRef<any>(null);
    53→  const [error, setError] = useState<string | null>(null);
    54→  const [loading, setLoading] = useState(true);
    55→
    56→  useEffect(() => {
    57→    let cancelled = false;
    58→
    59→    async function init() {
    60→      try {
    61→        await loadDashJs();
    62→        if (cancelled || !videoRef.current) return;
    63→
    64→        const video = videoRef.current;
    65→        const player = window.dashjs.MediaPlayer().create();
    66→        player.initialize(video, url, autoplay);
    67→        player.updateSettings({
    68→          streaming: {
    69→            buffer: {
    70→              fastSwitchEnabled: true,
    71→              bufferTimeAtTopQuality: 30,
    72→              bufferTimeAtTopQualityLongForm: 60,
    73→            },
    74→          },
    75→        });
    76→        playerRef.current = player;
    77→        setLoading(false);
    78→      } catch (e: any) {
    79→        if (!cancelled) {
    80→          setError(e?.message || "Failed to load DASH player");
    81→          setLoading(false);
    82→        }
    83→      }
    84→    }
    85→
    86→    init();
    87→
    88→    return () => {
    89→      cancelled = true;
    90→      if (playerRef.current) {
    91→        try { playerRef.current.reset(); } catch {}
    92→        playerRef.current = null;
    93→      }
    94→    };
    95→  }, [url, autoplay]);
    96→
    97→  return (
    98→    <div className="absolute inset-0 w-full h-full bg-black">
    99→      {loading && (
   100→        <div className="absolute inset-0 flex items-center justify-center">
   101→          <div className="ltv-spinner ltv-spinner-lg" />
   102→        </div>
   103→      )}
   104→      {error && (
   105→        <div className="absolute inset-0 flex items-center justify-center">
   106→          <div className="text-center space-y-2">
   107→            <p className="text-rose-400 text-sm">{error}</p>
   108→            <button onClick={() => window.location.reload()} className="text-xs text-white/60 hover:text-white">
   109→              Reload
   110→            </button>
   111→          </div>
   112→        </div>
   113→      )}
   114→      <video
   115→        ref={videoRef}
   116→        className="w-full h-full"
   117→        controls
   118→        onEnded={onEnded}
   119→        crossOrigin="anonymous"
   120→      />
   121→      {subtitleTracks && subtitleTracks.length > 0 && (
   122→        <div className="absolute bottom-14 right-4 z-10 flex gap-1">
   123→          {subtitleTracks.slice(0, 5).map((s) => (
   124→            <button
   125→              key={s.lang || s.label}
   126→              onClick={() => {
   127→                const video = videoRef.current;
   128→                if (!video) return;
   129→                // Remove existing text tracks
   130→                while (video.textTracks.length > 0) {
   131→                  video.removeChild(video.textTracks[0] as any);
   132→                }
   133→                const track = document.createElement("track");
   134→                track.kind = "subtitles";
   135→                track.label = s.label;
   136→                track.srclang = s.lang || "en";
   137→                track.src = s.url;
   138→                track.default = true;
   139→                video.appendChild(track);
   140→              }}
   141→              className="px-2 py-1 text-[10px] font-bold bg-black/60 text-white/70 hover:bg-white/20 hover:text-white rounded transition-colors"
   142→            >
   143→              {s.lang || s.label}
   144→            </button>
   145→          ))}
   146→        </div>
   147→      )}
   148→    </div>
   149→  );
   150→}

   226→// ── Embed Player with auto-fallback on 410/dead links ──
   227→// If the embed source returns a dead page (410 Gone, etc.), shows a "switch server" overlay
   228→function EmbedPlayerWithFallback({
   229→  src,
   230→  animeTitle,
   231→  episodeNum,
   232→  provider,
   233→  providersForCurrentEp,
   234→  failedProviders,
   235→  onProviderFailed,
   236→  onProviderSelect,
   237→  getProviderDisplayName,
   238→}: {
   239→  src: string;
   240→  animeTitle: string;
   241→  episodeNum: number;
   242→  provider: string;
   243→  providersForCurrentEp: string[];
   244→  failedProviders: Set<string>;
   245→  onProviderFailed: (p: string) => void;
   246→  onProviderSelect: (p: string) => void;
   247→  getProviderDisplayName: (p: string) => string;
   248→}) {
   249→  const iframeRef = useRef<HTMLIFrameElement>(null);
   250→  const [embedFailed, setEmbedFailed] = useState(false);
   251→  const [showServerOverlay, setShowServerOverlay] = useState(false);
   252→
   253→  // Detect iframe load failure — embed sources often return 410 Gone
   254→  useEffect(() => {
   255→    const timer = setTimeout(() => {
   256→      // After 8 seconds, if iframe loaded but content might be dead,
   257→      // show the switch server button as a floating hint
   258→      setShowServerOverlay(true);
   259→    }, 8000);
   260→    return () => clearTimeout(timer);
   261→  }, [src]);
   262→
   263→  const handleIframeLoad = useCallback(() => {
   264→    // We can't read iframe content due to cross-origin, but we can
   265→    // check if the URL might be a dead embed by doing a HEAD request
   266→    try {
   267→      fetch(src, { method: "HEAD", mode: "no-cors" }).catch(() => {});
   268→    } catch {}
   269→  }, [src]);
   270→
   271→  const handleEmbedError = useCallback(() => {
   272→    setEmbedFailed(true);
   273→    onProviderFailed(provider);
   274→  }, [provider, onProviderFailed]);
   275→
   276→  const otherProviders = providersForCurrentEp.filter(
   277→    p => p !== provider && !failedProviders.has(p)
   278→  );
   279→
   280→  if (embedFailed) {
   281→    return (
   282→      <div className="absolute inset-0 flex items-center justify-center bg-black z-20">
   283→        <div className="text-center space-y-4 max-w-sm px-6">
   284→          <div className="w-12 h-12 rounded-xl bg-[#ffffff]/10 border border-[#ffffff]/20 flex items-center justify-center mx-auto">
   285→            <svg className="w-6 h-6 text-[#ffffff]/70" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
   286→              <path d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z" />
   287→            </svg>
   288→          </div>
   289→          <p className="text-white/75 text-sm">This embed source is unavailable (410 Gone)</p>
   290→          {otherProviders.length > 0 ? (
   291→            <div className="flex flex-wrap gap-2 justify-center">
   292→              {otherProviders.map(p => (
   293→                <button
   294→                  key={p}
   295→                  onClick={() => onProviderSelect(p)}
   296→                  className="px-3 py-1.5 rounded-lg bg-[#ffffff] text-black text-xs font-bold hover:bg-white/90 transition-colors"
   297→                >
   298→                  Try {getProviderDisplayName(p)}
   299→                </button>
   300→              ))}
   301→            </div>
   302→          ) : (
   303→            <button
   304→              onClick={() => window.location.reload()}
   305→              className="px-5 py-2 rounded-lg bg-[#ffffff] text-black text-sm font-bold hover:bg-white/90 transition-colors"
   306→            >
   307→              Refresh Page
   308→            </button>
   309→          )}
   310→        </div>
   311→      </div>
   312→    );
   313→  }
   314→
   315→  return (
   316→    <div className="absolute inset-0">
   317→      <iframe
   318→        ref={iframeRef}
   319→        src={src}
   320→        className="w-full h-full border-0"
   321→        allowFullScreen
   322→        allow="autoplay; fullscreen; picture-in-picture; encrypted-media"
   323→        referrerPolicy="no-referrer-when-downgrade"
   324→        title={`${animeTitle} - Episode ${episodeNum}`}
   325→        onLoad={handleIframeLoad}
   326→        onError={handleEmbedError}
   327→      />
   328→      {/* Floating "embed dead?" overlay — shows after 8s so user can switch servers */}
   329→      {showServerOverlay && otherProviders.length > 0 && (
   330→        <div className="absolute top-3 right-3 z-30">
   331→          <div className="flex items-center gap-2 bg-black/80 backdrop-blur-md rounded-lg border border-white/[0.08] px-3 py-2 shadow-xl">
   332→            <span className="text-[10px] text-white/55">Embed not working?</span>
   333→            {otherProviders.slice(0, 3).map(p => (
   334→              <button
   335→                key={p}
   336→                onClick={() => onProviderSelect(p)}
   337→                className="px-2.5 py-1 rounded-md bg-[#ffffff] text-black text-[10px] font-bold hover:bg-white/90 transition-colors"
   338→              >
   339→                {getProviderDisplayName(p)}
   340→              </button>
   341→            ))}
   342→            <button
   343→              onClick={() => setShowServerOverlay(false)}
   344→              className="p-1 text-white/40 hover:text-white transition-colors"
   345→            >
   346→              <svg className="w-3 h-3" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2.5}>
   347→                <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
   348→              </svg>
   349→            </button>
   350→          </div>
   351→        </div>
   352→      )}
   353→    </div>
   354→  );
   355→}
