    41→    41→export default function HLSPlayerNew({

    51→    51→  // the outro button never appears at the start of the video (which

    57→    57→  // effects DON'T list them as deps. Parents (e.g. watch-page-shell) re-render

    59→    59→  // callbacks — a new identity each render. If those were in the effect deps,
    60→    60→  // the stream effect would destroy + recreate the hls.js instance on every
    61→    61→  // frame, refetching the m3u8 + all segments in an infinite loop. Reading
    62→    62→  // through a ref keeps the effect stable (only url/sourceType/autoplay matter).
    63→    63→  const onProviderFailedRef = useRef(onProviderFailed);
    64→    64→  const onEndedRef = useRef(onEnded);

    69→    69→  onEndedRef.current = onEnded;
    70→    70→  onCanPlayRef.current = onCanPlay;
    71→    71→  autoSkipRef.current = autoSkip;
    72→    72→  const [playing, setPlaying] = useState(false);
    73→    73→  const [currentTime, setCurrentTime] = useState(0);
    74→    74→  const [duration, setDuration] = useState(0);
    75→    75→  const [buffered, setBuffered] = useState(0);
    76→    76→  const [volume, setVolume] = useState(1);
    77→    77→  const [muted, setMuted] = useState(false);
    78→    78→  const [fullscreen, setFullscreen] = useState(false);
    79→    79→  const [loading, setLoading] = useState(true);
    80→    80→  const [error, setError] = useState<string | null>(null);
    81→    81→  const [showControls, setShowControls] = useState(true);
    82→    82→  const [qualities, setQualities] = useState<any[]>([]);
    83→    83→  const [currentQuality, setCurrentQuality] = useState(-1);
    84→    84→  const [activeMenu, setActiveMenu] = useState<'quality' | 'subtitles' | 'speed' | null>(null);
    85→    85→  const [showSubtitleSettings, setShowSubtitleSettings] = useState(false);
    86→    86→  const [showDownloadModal, setShowDownloadModal] = useState(false);
    87→    87→  const { settings: subtitleSettings, update: updateSubtitleSettings, reset: resetSubtitleSettings } = useSubtitleSettings();
    88→    88→  const [currentSubtitle, setCurrentSubtitle] = useState(-1);
    89→    89→  const [hlsSubtitles, setHlsSubtitles] = useState<any[]>([]);
    90→    90→  const [playbackRate, setPlaybackRate] = useState(1);
    91→    91→  const [showSkipIntro, setShowSkipIntro] = useState(false);
    92→    92→  const [showSkipOutro, setShowSkipOutro] = useState(false);
    93→    93→  // Track when user clicked skip — prevents the button from re-appearing
    94→    94→  // immediately after the click (the timeupdate loop would otherwise keep
    95→    95→  // showing it because video.currentTime is still in the skip range).
    96→    96→  const skipIntroClickedRef = useRef(0);
    97→    97→  const skipOutroClickedRef = useRef(0);
    98→    98→  const [hoverTime, setHoverTime] = useState<number | null>(null);
    99→    99→  const [hoverX, setHoverX] = useState(0);
   100→   100→  const [seekRipple, setSeekRipple] = useState<{ x: number; dir: 'left' | 'right' } | null>(null);
   101→   101→  const [volumeHover, setVolumeHover] = useState(false);
   102→   102→  const [screenshotFlash, setScreenshotFlash] = useState(false);
   103→   103→  const [screenshotToast, setScreenshotToast] = useState<string | null>(null);
   104→   104→  const [downloadLoading, setDownloadLoading] = useState(false);
   105→   105→  const [downloadToast, setDownloadToast] = useState<string | null>(null);
   106→   106→  const [streamHealth, setStreamHealth] = useState<'good' | 'fair' | 'poor' | 'unknown'>('unknown');
   107→   107→  const [showStreamInfo, setShowStreamInfo] = useState(false);
   108→   108→  const controlsTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   109→   109→  const retryCountRef = useRef(0);
   110→   110→  const lastTapRef = useRef<{ time: number; x: number } | null>(null);
   111→   111→  const canvasRef = useRef<HTMLCanvasElement>(null);
   112→   112→
   113→   113→  // ─── Load stream ──────────────────────────────────────────────────
   114→   114→  useEffect(() => {
   115→   115→    const video = videoRef.current;
   116→   116→    if (!video || !url) return;
   117→   117→
   118→   118→    setLoading(true);
   119→   119→    setError(null);
   120→   120→    retryCountRef.current = 0;
   121→   121→    setCurrentSubtitle(-1);
   122→   122→    setHlsSubtitles([]);
   123→   123→    onCanPlayFiredRef.current = false; // reset for new stream
   124→   124→
   125→   125→    if (hlsRef.current) {
   126→   126→      hlsRef.current.destroy();
   127→   127→      hlsRef.current = null;
   128→   128→    }
   129→   129→
   130→   130→    if (sourceType === 'mp4' || url.endsWith('.mp4') || url.includes('video.mp4')) {
   131→   131→      video.src = url;
   132→   132→      video.load();
   133→   133→      if (autoplay) video.play().catch(() => {});
   134→   134→      return;
   135→   135→    }
   136→   136→
   137→   137→    if (Hls.isSupported()) {
   138→   138→      const hls = new Hls({
   139→   139→        enableWorker: true,
   140→   140→        lowLatencyMode: false,
   141→   141→        backBufferLength: 90,
   142→   142→        maxBufferLength: 90,
   143→   143→        maxMaxBufferLength: 180,
   144→   144→        maxBufferSize: 90 * 1000 * 1000,
   145→   145→        maxBufferHole: 0.5,
   146→   146→        // Start at the HIGHEST quality level (1080p), not lowest (360p).
   147→   147→        // -1 = auto-select based on bandwidth, but we set a high default estimate
   148→   148→        // so it picks the best quality immediately on fast connections.
   149→   149→        startLevel: -1,
   150→   150→        // Assume 10Mbps bandwidth by default — this makes hls.js start at 1080p
   151→   151→        // instead of 360p. The ABR controller will adjust down if needed.
   152→   152→        abrEwmaDefaultEstimate: 10000000,
   153→   153→        abrBandWidthFactor: 0.95,
   154→   154→        abrBandWidthUpFactor: 0.7,
   155→   155→        maxStarvationDelay: 4,
   156→   156→        abrEwmaDefaultEstimateMax: 20000000,
   157→   157→        manifestLoadingTimeOut: 20000,
   158→   158→        manifestLoadingMaxRetry: 4,
   159→   159→        levelLoadingTimeOut: 20000,
   160→   160→        levelLoadingMaxRetry: 4,
   161→   161→        fragLoadingTimeOut: 45000,
   162→   162→        fragLoadingMaxRetry: 8,
   163→   163→        fragLoadingRetryDelay: 1000,
   164→   164→        enableWebVTT: true,
   165→   165→        xhrSetup: (xhr) => { xhr.withCredentials = false; },
   166→   166→      });
   167→   167→      hlsRef.current = hls;
   168→   168→      hls.loadSource(url);
   169→   169→      hls.attachMedia(video);
   170→   170→
   171→   171→      hls.on(Hls.Events.MANIFEST_PARSED, (_e, data) => {
   172→   172→        setQualities(data.levels);
   173→   173→        setLoading(false);
   174→   174→        if (autoplay) {
   175→   175→          video.play().catch(() => {
   176→   176→            video.muted = true;
   177→   177→            video.play().catch(() => {});
   178→   178→          });
   179→   179→        }
   180→   180→      });
   181→   181→
   182→   182→      hls.on(Hls.Events.SUBTITLE_TRACKS_UPDATED, (_e, data) => {
   183→   183→        setHlsSubtitles(data.subtitleTracks || []);
   184→   184→        if (data.subtitleTracks.length > 0 && hls.subtitleTrack === -1) {
   185→   185→          hls.subtitleTrack = 0;
   186→   186→          setCurrentSubtitle(0);
   187→   187→          // After HLS.js sets the subtitle track, it creates a textTrack with
   188→   188→          // mode='showing' (native rendering). We need to switch it to 'hidden'
   189→   189→          // so only our CustomSubtitleOverlay renders (prevents double subtitles).
   190→   190→          setTimeout(() => {
   191→   191→            const video = videoRef.current;
   192→   192→            if (video) {
   193→   193→              for (let i = 0; i < video.textTracks.length; i++) {
   194→   194→                if (i === 0) video.textTracks[i].mode = 'hidden';
   195→   195→              }
   196→   196→            }
   197→   197→          }, 200);
   198→   198→        }
   199→   199→      });
   200→   200→
   201→   201→      hls.on(Hls.Events.LEVEL_SWITCHED, (_e, data) => {
   202→   202→        setCurrentQuality(data.level);
   203→   203→        // After a level switch, HLS.js may re-enable native subtitle rendering.
   204→   204→        // Force the active track back to 'hidden' mode so only our overlay shows.
   205→   205→        if (currentSubtitle >= 0) {
   206→   206→          setTimeout(() => {
   207→   207→            const video = videoRef.current;
   208→   208→            if (video && video.textTracks.length > currentSubtitle) {
   209→   209→              video.textTracks[currentSubtitle].mode = 'hidden';
   210→   210→            }
   211→   211→          }, 100);
   212→   212→        }
   213→   213→      });
   214→   214→
   215→   215→      hls.on(Hls.Events.ERROR, (_e, data) => {
   216→   216→        // Non-fatal errors are expected during streaming (a segment/playlist
   217→   217→        // fetch hiccup) and hls.js already recovers from them on its own —
   218→   218→        // console.warn so they don't trip Next's dev error overlay for
   219→   219→        // something that isn't actually broken.
   220→   220→        const log = data.fatal ? console.error : console.warn;
   221→   221→        log(`[HLS] Error: type=${data.type} details=${data.details} fatal=${data.fatal}`);
   222→   222→        if (data.fatal) {
   223→   223→          if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
   224→   224→            if (retryCountRef.current < 3) {
   225→   225→              retryCountRef.current++;
   226→   226→              setTimeout(() => hls.startLoad(), 2000);
   227→   227→            } else {
   228→   228→              setError('Stream failed. Try another server.');
   229→   229→              setLoading(false);
   230→   230→              onProviderFailedRef.current?.();
   231→   231→            }
   232→   232→          } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
   233→   233→            hls.recoverMediaError();
   234→   234→          } else {
   235→   235→            setError('Playback error. Try another server.');
   236→   236→            setLoading(false);
   237→   237→            onProviderFailedRef.current?.();
   238→   238→          }
   239→   239→        }
   240→   240→      });
   241→   241→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   242→   242→      video.src = url;
   243→   243→      if (autoplay) video.play().catch(() => {});
   244→   244→    } else {
   245→   245→      setError('HLS not supported');
   246→   246→      setLoading(false);
   247→   247→    }
   248→   248→
   249→   249→    return () => {
   250→   250→      if (hlsRef.current) {
   251→   251→        hlsRef.current.destroy();
   252→   252→        hlsRef.current = null;
   253→   253→      }
   254→   254→    };
   255→   255→  }, [url, sourceType, autoplay]);
   256→   256→
   257→   257→  // ─── Video events ─────────────────────────────────────────────────
   258→   258→  useEffect(() => {
   259→   259→    const video = videoRef.current;
   260→   260→    if (!video) return;
   261→   261→
   262→   262→    const onPlay = () => setPlaying(true);
   263→   263→    const onPause = () => setPlaying(false);
   264→   264→    const onTime = () => {
   265→   265→      setCurrentTime(video.currentTime);
   266→   266→      if (video.buffered.length > 0) {
   267→   267→        setBuffered(video.buffered.end(video.buffered.length - 1));
   268→   268→      }
   269→   269→      // Auto Skip — when enabled, jump straight past the opening/ending.
   270→   270→      if (autoSkipRef.current) {
   271→   271→        if (intro && video.currentTime >= intro.start && video.currentTime < intro.end - 0.3) {
   272→   272→          video.currentTime = intro.end;
   273→   273→          return;
   274→   274→        }
   275→   275→        if (outro && video.currentTime >= outro.start && video.currentTime < outro.end - 0.3) {
   276→   276→          video.currentTime = Math.min(outro.end, video.duration || outro.end);
   277→   277→          return;
   278→   278→        }
   279→   279→      }
   280→   280→      // Show skip intro button from 3s BEFORE intro starts until 5s AFTER it ends.
   281→   281→      // Also show immediately if video just loaded and we're already in the intro range
   282→   282→      // (happens when mimi loads super fast and starts playing at 0s, which is before intro.start)
   283→   283→      //
   284→   284→      // BUT: if the user just clicked Skip Intro (within last 8s), DON'T re-show the
   285→   285→      // button — otherwise it instantly pops back up because video.currentTime is still
   286→   286→      // in the intro range right after the seek.
   287→   287→      const sinceIntroClick = Date.now() - skipIntroClickedRef.current;
   288→   288→      const sinceOutroClick = Date.now() - skipOutroClickedRef.current;
   289→   289→
   290→   290→      if (intro && video.currentTime >= (intro.start - 3) && video.currentTime < (intro.end + 5) && sinceIntroClick > 8000) {
   291→   291→        setShowSkipIntro(true);
   292→   292→      } else {
   293→   293→        setShowSkipIntro(false);
   294→   294→      }
   295→   295→      if (outro && video.currentTime >= (outro.start - 3) && video.currentTime < (outro.end + 10) && sinceOutroClick > 8000) {
   296→   296→        setShowSkipOutro(true);
   297→   297→      } else {
   298→   298→        setShowSkipOutro(false);
   299→   299→      }
   300→   300→    };
   301→   301→
   302→   302→    // Also check on MANIFEST_PARSED / loadedmetadata — if intro data is available
   303→   303→    // and we're at position 0, show the skip button immediately (before timeupdate fires)
   304→   304→    const onLoadedMetadata = () => {
   305→   305→      setDuration(video.duration || 0);
   306→   306→      // If intro starts soon (within first 60s), show the button immediately
   307→   307→      if (intro && intro.start < 60 && video.currentTime < intro.end) {
   308→   308→        setShowSkipIntro(true);
   309→   309→      }
   310→   310→      if (outro && video.duration && outro.start < video.duration) {
   311→   311→        // Don't show outro yet, just confirm we have the data
   312→   312→      }
   313→   313→    };
   314→   314→    const onDur = () => setDuration(video.duration || 0);
   315→   315→
   316→   316→    let waitingTimer: ReturnType<typeof setTimeout> | null = null;
   317→   317→    const onWaiting = () => {
   318→   318→      if (waitingTimer) clearTimeout(waitingTimer);
   319→   319→      // Show loading after 1.5s of waiting (was 5s — too long, user gets confused)
   320→   320→      waitingTimer = setTimeout(() => {
   321→   321→        if (video.readyState < 3) setLoading(true);
   322→   322→      }, 1500);
   323→   323→    };
   324→   324→    const onPlaying = () => {
   325→   325→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   326→   326→      setLoading(false);
   327→   327→    };
   328→   328→    const onCanPlay = () => {
   329→   329→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   330→   330→      setLoading(false);
   331→   331→      // Notify parent that video is ready to play (for loading screen dismissal)
   332→   332→      if (!onCanPlayFiredRef.current) {
   333→   333→        onCanPlayFiredRef.current = true;
   334→   334→        onCanPlayRef.current?.();
   335→   335→      }
   336→   336→    };
   337→   337→    const onEnd = () => onEndedRef.current?.();
   338→   338→    const onErr = () => { setError('Playback error.'); setLoading(false); };
   339→   339→
   340→   340→    video.addEventListener('play', onPlay);
   341→   341→    video.addEventListener('pause', onPause);
   342→   342→    video.addEventListener('timeupdate', onTime);
   343→   343→    video.addEventListener('durationchange', onDur);
   344→   344→    video.addEventListener('loadedmetadata', onLoadedMetadata);
   345→   345→    video.addEventListener('waiting', onWaiting);
   346→   346→    video.addEventListener('playing', onPlaying);
   347→   347→    video.addEventListener('canplay', onCanPlay);
   348→   348→    video.addEventListener('ended', onEnd);
   349→   349→    video.addEventListener('error', onErr);
   350→   350→
   351→   351→    return () => {
   352→   352→      video.removeEventListener('play', onPlay);
   353→   353→      video.removeEventListener('pause', onPause);
   354→   354→      video.removeEventListener('timeupdate', onTime);
   355→   355→      video.removeEventListener('durationchange', onDur);
   356→   356→      video.removeEventListener('loadedmetadata', onLoadedMetadata);
   357→   357→      video.removeEventListener('waiting', onWaiting);
   358→   358→      video.removeEventListener('playing', onPlaying);
   359→   359→      video.removeEventListener('canplay', onCanPlay);
   360→   360→      video.removeEventListener('ended', onEnd);
   361→   361→      video.removeEventListener('error', onErr);
   362→   362→    };
   363→   363→  }, [intro, outro]);
   364→   364→
   365→   365→  // ─── Controls auto-hide ───────────────────────────────────────────
   366→   366→  const showControlsTemp = useCallback(() => {
   367→   367→    setShowControls(true);
   368→   368→    if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   369→   369→    controlsTimeoutRef.current = setTimeout(() => {
   370→   370→      if (playing) { setShowControls(false); setActiveMenu(null); }
   371→   371→    }, 3500);
   372→   372→  }, [playing]);
   373→   373→
   374→   374→  // ─── Fullscreen ───────────────────────────────────────────────────
   375→   375→  useEffect(() => {
   376→   376→    const onFs = () => setFullscreen(!!document.fullscreenElement);
   377→   377→    document.addEventListener('fullscreenchange', onFs);
   378→   378→    return () => document.removeEventListener('fullscreenchange', onFs);
   379→   379→  }, []);
   380→   380→
   381→   381→  const toggleFullscreen = async () => {
   382→   382→    try {
   383→   383→      if (document.fullscreenElement) {
   384→   384→        await document.exitFullscreen();
   385→   385→        // Release orientation lock when exiting fullscreen
   386→   386→        try { await (screen.orientation as any)?.unlock(); } catch {}
   387→   387→      } else {
   388→   388→        await containerRef.current?.requestFullscreen();
   389→   389→        // On mobile, lock to landscape so the video fills the screen
   390→   390→        try {
   391→   391→          if (screen.orientation && typeof (screen.orientation as any).lock === 'function') {
   392→   392→            await (screen.orientation as any).lock('landscape');
   393→   393→          }
   394→   394→        } catch {
   395→   395→          // Orientation lock not supported or denied — fullscreen still works
   396→   396→        }
   397→   397→      }
   398→   398→    } catch {
   399→   399→      // Fullscreen request failed silently
   400→   400→    }
   401→   401→  };
   402→   402→
   403→   403→  const seek = (time: number) => { if (videoRef.current) videoRef.current.currentTime = time; };
   404→   404→  const togglePlay = () => {
   405→   405→    const v = videoRef.current; if (!v) return;
   406→   406→    if (v.paused) v.play(); else v.pause();
   407→   407→  };
   408→   408→  const toggleMute = () => {
   409→   409→    const v = videoRef.current; if (!v) return;
   410→   410→    v.muted = !v.muted; setMuted(v.muted);
   411→   411→  };
   412→   412→  const changeVolume = (vol: number) => {
   413→   413→    const v = videoRef.current; if (!v) return;
   414→   414→    v.volume = vol; v.muted = vol === 0;
   415→   415→    setVolume(vol); setMuted(vol === 0);
   416→   416→  };
   417→   417→  const changeQuality = (level: number) => {
   418→   418→    if (hlsRef.current) { hlsRef.current.currentLevel = level; setCurrentQuality(level); }
   419→   419→    setActiveMenu(null);
   420→   420→  };
   421→   421→
   422→   422→  // ── Auto-enable the first external subtitle track ──
