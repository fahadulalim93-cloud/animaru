    77→/**
    78→ * Hard ceiling on a provider promise.
    79→ *
    80→ * Several sources in the fan-out below accept no timeout option at all
    81→ * (fetchRawEpisodes, the animex slug->servers chain, the AniVault fetches,
    82→ * fetchAllAnimePaheSources), so a single slow upstream set the runtime for the
    83→ * whole route — measured at 93-180s against this function's 60s maxDuration,
    84→ * which meant Vercel killed it and NONE of the 17 providers reached the player.
    85→ * Promise.allSettled waits for the slowest member; this makes "slowest" bounded.
    86→ *
    87→ * Resolves to `fallback` on timeout rather than rejecting, so one dead provider
    88→ * is simply an empty slot instead of a lost response.
    89→ */
    90→function withTimeout<T>(p: Promise<T> | T, ms: number, fallback: T): Promise<T> {
    91→  return Promise.race([
    92→    Promise.resolve(p).catch(() => fallback),
    93→    new Promise<T>(resolve => setTimeout(() => resolve(fallback), ms)),
    94→  ]);
    95→}

   143→export async function GET(
   144→  _req: NextRequest,
   145→  { params }: { params: Promise<{ anilistId: string; episode: string }> }
   146→) {
   147→  const { anilistId, episode } = await params;
   148→  const id = parseInt(anilistId, 10);
   149→  const epNum = parseInt(episode, 10);
   150→  if (isNaN(id) || id <= 0) {
   151→    return NextResponse.json({ error: "Invalid anilistId" }, { status: 400 });
   152→  }
   153→
   154→  /**
   155→   * Provider group. Everything in one request measured 98-173s against a 60s
   156→   * maxDuration, so Vercel killed it and the player got NOTHING — capping the
   157→   * providers hard enough to fit meant silently dropping miruro/animex, the two
   158→   * highest-priority sources.
   159→   *
   160→   * Splitting instead: the client fires ?group=fast and ?group=slow in
   161→   * parallel, so each half gets its own 60s budget and merges as it lands.
   162→   * Nothing is dropped; the slow half just shows up later.
   163→   *
   164→   * ?group=all (the default) preserves the original single-request behaviour
   165→   * for any caller that isn't group-aware.
   166→   */
   167→  const group = (_req.nextUrl.searchParams.get("group") || "all").toLowerCase();
   168→  const wantFast = group === "all" || group === "fast";
   169→  const wantSlow = group === "all" || group === "slow";
   170→  // The slow half is alone in its request, so it can afford real patience.
   171→  // Reduced from 45s to 25s — dead servers shouldn't hang for 45s.
   172→  // Most working providers finish in 5-10s; 25s is generous.
   173→  const SLOW_CAP = group === "slow" ? 25000 : 12000;
   174→
   175→  try {
   176→  // ─── Resolve anime title from AniList (needed for AnixTV search) ────────────
   177→  // AnixTV's watch URL requires the anime title as a query param — without it,
   178→  // AnixTV can't find the anime and returns no iframe → no hindi servers.
   179→  let animeTitle = "Anime";
   180→  let animeTitles: { english?: string; romaji?: string; native?: string } = {};
   181→  let animeSeason = 1;  // hoisted out of try-block so AnixTV block can use it
   182→  try {
   183→    const titleRes = await Promise.race([
   184→      fetch("https://graphql.anilist.co", {
   185→        method: "POST",
   186→        headers: { "Content-Type": "application/json" },
   187→        body: JSON.stringify({
   188→          query: `query($id:Int){Media(id:$id,type:ANIME){id title{english romaji native}}}`,
   189→          variables: { id },
   190→        }),
   191→      }),
   192→      new Promise<Response | null>(r => setTimeout(() => r(null), 5000)),
   193→    ]);
   194→    if (titleRes && titleRes.ok) {
   195→      const titleData = await titleRes.json();
   196→      const t = titleData?.data?.Media?.title;
   197→      animeTitles = { english: t?.english, romaji: t?.romaji, native: t?.native };
   198→      animeTitle = t?.english || t?.romaji || t?.native || "Anime";
   199→    }
   200→
   201→    // Also fetch the anime's season + relations to determine the correct
   202→    // season number for AnixTV (e.g. Slime S4 → season=4, not 1)
   203→    //
   204→    // Two strategies:
   205→    //   1. Title heuristic: parse "Season N" / "S2" / "Part 2" from the title
   206→    //      (most reliable for sequels that include the season number in title)
   207→    //   2. AniList relation walk: count PREQUEL chain (any format — TV, ONA,
   208→    //      OVA, MOVIE, SPECIAL — AniList sometimes lists OVAs as the prequel
   209→    //      for sequel seasons, e.g. Slime S2 → "Visions of Coleus" OVA → S1)
   210→    try {
   211→      // Strategy 1: title heuristic
   212→      const titleForSeason = animeTitles.english || animeTitles.romaji || animeTitle;
   213→      const seasonMatch = titleForSeason.match(/\bSeason\s+(\d+)\b/i)
   214→                       || titleForSeason.match(/\bS(\d{1,2})\b\s*$/i);
   215→      if (seasonMatch) {
   216→        const parsed = parseInt(seasonMatch[1], 10);
   217→        if (parsed >= 1 && parsed <= 30) {
   218→          animeSeason = parsed;
   219→          console.log(`[Servers] AnixTV season from title: ${animeTitle} → season ${animeSeason}`);
   220→        }
   221→      }
   222→
   223→      // Strategy 2: AniList relation walk (if title heuristic didn't fire)
   224→      if (animeSeason === 1) {
   225→        const seasonRes = await Promise.race([
   226→          fetch("https://graphql.anilist.co", {
   227→            method: "POST",
   228→            headers: { "Content-Type": "application/json" },
   229→            body: JSON.stringify({
   230→              query: `query($id:Int){Media(id:$id,type:ANIME){id idMal season seasonYear format relations{edges{relationType node{id title{english romaji} format}}} } }`,
   231→              variables: { id },
   232→            }),
   233→          }),
   234→          new Promise<Response | null>(r => setTimeout(() => r(null), 5000)),
   235→        ]);
   236→        if (seasonRes && seasonRes.ok) {
   237→          const seasonData = await seasonRes.json();
   238→          const media = seasonData?.data?.Media;
   239→          if (media?.relations?.edges) {
   240→            // Include ALL formats in the prequel walk — AniList sometimes
   241→            // lists OVAs/specials/movies as the prequel for a sequel season.
   242→            const prequels = media.relations.edges.filter(
   243→              (e: any) => e.relationType === "PREQUEL"
   244→            );
   245→            if (prequels.length > 0) {
   246→              let count = 1;
   247→              let currentPrequel = prequels[0];
   248→              const visited = new Set<number>([id]);
   249→              for (let i = 0; i < 15; i++) {
   250→                if (!currentPrequel?.node?.id) break;
   251→                if (visited.has(currentPrequel.node.id)) break;
   252→                visited.add(currentPrequel.node.id);
   253→                try {
   254→                  const prequelRes = await fetch("https://graphql.anilist.co", {
   255→                    method: "POST",
   256→                    headers: { "Content-Type": "application/json" },
   257→                    body: JSON.stringify({
   258→                      query: `query($id:Int){Media(id:$id,type:ANIME){relations{edges{relationType node{id format title{english}}}} } }`,
   259→                      variables: { id: currentPrequel.node.id },
   260→                    }),
   261→                  });
   262→                  if (prequelRes.ok) {
   263→                    const prequelData = await prequelRes.json();
   264→                    const prequelEdges = prequelData?.data?.Media?.relations?.edges || [];
   265→                    const nextPrequel = prequelEdges.find(
   266→                      (e: any) => e.relationType === "PREQUEL"
   267→                    );
   268→                    if (nextPrequel) {
   269→                      count++;
   270→                      currentPrequel = nextPrequel;
   271→                    } else {
   272→                      count++;
   273→                      break;
   274→                    }
   275→                  } else break;
   276→                } catch { break; }
   277→              }
   278→              animeSeason = count;
   279→            }
   280→          }
   281→          console.log(`[Servers] AnixTV season from relations: ${animeTitle} → season ${animeSeason}`);
   282→        }
   283→      }
   284→    } catch { /* fallback to season 1 */ }
   285→  } catch { /* fallback to "Anime" */ }
