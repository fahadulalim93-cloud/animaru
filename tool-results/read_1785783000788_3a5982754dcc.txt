    78→    78→ * Hard ceiling on a provider promise.
    79→    79→ *
    80→    80→ * Several sources in the fan-out below accept no timeout option at all
    81→    81→ * (fetchRawEpisodes, the animex slug->servers chain, the AniVault fetches,
    82→    82→ * fetchAllAnimePaheSources), so a single slow upstream set the runtime for the
    83→    83→ * whole route — measured at 93-180s against this function's 60s maxDuration,
    84→    84→ * which meant Vercel killed it and NONE of the 17 providers reached the player.
    85→    85→ * Promise.allSettled waits for the slowest member; this makes "slowest" bounded.
    86→    86→ *
    87→    87→ * Resolves to `fallback` on timeout rather than rejecting, so one dead provider
    88→    88→ * is simply an empty slot instead of a lost response.
    89→    89→ */
    90→    90→function withTimeout<T>(p: Promise<T> | T, ms: number, fallback: T): Promise<T> {
    91→    91→  return Promise.race([
    92→    92→    Promise.resolve(p).catch(() => fallback),
    93→    93→    new Promise<T>(resolve => setTimeout(() => resolve(fallback), ms)),
    94→    94→  ]);
    95→    95→}

   153→   153→
   154→   154→  /**
   155→   155→   * Provider group. Everything in one request measured 98-173s against a 60s
   156→   156→   * maxDuration, so Vercel killed it and the player got NOTHING — capping the
   157→   157→   * providers hard enough to fit meant silently dropping miruro/animex, the two
   158→   158→   * highest-priority sources.
   159→   159→   *

   161→   161→   * parallel, so each half gets its own 60s budget and merges as it lands.
   162→   162→   * Nothing is dropped; the slow half just shows up later.
   163→   163→   *
   164→   164→   * ?group=all (the default) preserves the original single-request behaviour
   165→   165→   * for any caller that isn't group-aware.
   166→   166→   */
   167→   167→  const group = (_req.nextUrl.searchParams.get("group") || "all").toLowerCase();

   169→   169→  const wantSlow = group === "all" || group === "slow";
   170→   170→  // The slow half is alone in its request, so it can afford real patience.
   171→   171→  const SLOW_CAP = group === "slow" ? 45000 : 12000;

   176→   176→  // AnixTV can't find the anime and returns no iframe → no hindi servers.

   188→   188→        }),
   189→   189→      }),

   191→   191→    ]);

   195→   195→      animeTitles = { english: t?.english, romaji: t?.romaji, native: t?.native };
   196→   196→      animeTitle = t?.english || t?.romaji || t?.native || "Anime";
   197→   197→    }
   198→   198→
   199→   199→    // Also fetch the anime's season + relations to determine the correct
   200→   200→    // season number for AnixTV (e.g. Slime S4 → season=4, not 1)
   201→   201→    //
   202→   202→    // Two strategies:
   203→   203→    //   1. Title heuristic: parse "Season N" / "S2" / "Part 2" from the title
   204→   204→    //      (most reliable for sequels that include the season number in title)

   206→   206→    //      OVA, MOVIE, SPECIAL — AniList sometimes lists OVAs as the prequel

   218→   218→        }

   223→   223→        const seasonRes = await Promise.race([
   224→   224→          fetch("https://graphql.anilist.co", {
   225→   225→            method: "POST",
   226→   226→            headers: { "Content-Type": "application/json" },

   237→   237→          if (media?.relations?.edges) {

   241→   241→              (e: any) => e.relationType === "PREQUEL"
   242→   242→            );
   243→   243→            if (prequels.length > 0) {
   244→   244→              let count = 1;
   245→   245→              let currentPrequel = prequels[0];
   246→   246→              const visited = new Set<number>([id]);
   247→   247→              for (let i = 0; i < 15; i++) {

   249→   249→                if (visited.has(currentPrequel.node.id)) break;

   257→   257→                      variables: { id: currentPrequel.node.id },
   258→   258→                    }),
   259→   259→                  });
