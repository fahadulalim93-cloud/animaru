    76→// ── Countdown Timer — pure B/W ──
    77→function CountdownTimer({ airingAt, episode }: { airingAt: number; episode: number }) {
    78→  const [timeLeft, setTimeLeft] = useState("");
    79→  const [expired, setExpired] = useState(false);
    80→
    81→  useEffect(() => {
    82→    function update() {
    83→      const now = Math.floor(Date.now() / 1000);
    84→      const diff = airingAt - now;
    85→      if (diff <= 0) {
    86→        setExpired(true);
    87→        setTimeLeft("");
    88→        return;
    89→      }
    90→      const d = Math.floor(diff / 86400);
    91→      const h = Math.floor((diff % 86400) / 3600);
    92→      const m = Math.floor((diff % 3600) / 60);
    93→      const s = diff % 60;
    94→      setTimeLeft(`${d > 0 ? `${d}d ` : ""}${h}h ${m}m ${s}s`);
    95→    }
    96→    update();
    97→    const interval = setInterval(update, 1000);
    98→    return () => clearInterval(interval);
    99→  }, [airingAt]);
   100→
   101→  if (expired) return null;
   102→
   103→  return (
   104→    <div className="inline-flex items-center gap-2 px-4 h-10 rounded-full bg-white/[0.06] hover:bg-white/10 transition-colors">
   105→      <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2.5} strokeLinecap="round" strokeLinejoin="round"><polyline points="13 17 18 12 13 7" /><polyline points="6 17 11 12 6 7" /></svg>
   106→      <span className="text-sm font-bold text-white tabular-nums">Ep {episode} in {timeLeft}</span>
   107→    </div>
   108→  );
   109→}
