    23→// ─── Tiny dropdown (click to open, backdrop to close) ───────────
    24→function MenuSelect({ label, value, options, onChange, disabledIds = [] }: {
    25→  label?: string;
    26→  value: string;
    27→  options: Array<{ id: string; label: string }>;
    28→  onChange: (id: string) => void;
    29→  disabledIds?: string[];
    30→}) {
    31→  const [open, setOpen] = useState(false);
    32→  const btnRef = useRef<HTMLButtonElement>(null);
    33→  const [menuPos, setMenuPos] = useState<{ top: number; left: number; width: number }>({ top: 0, left: 0, width: 0 });
    34→  const current = options.find(o => o.id === value);
    35→
    36→  // Calculate position for fixed dropdown on open
    37→  const handleOpen = () => {
    38→    if (!open && btnRef.current) {
    39→      const rect = btnRef.current.getBoundingClientRect();
    40→      const isMobile = window.innerWidth < 640;
    41→      if (isMobile) {
    42→        // On mobile: position dropdown below the button, full-width with padding
    43→        setMenuPos({
    44→          top: rect.bottom + 4,
    45→          left: 8,
    46→          width: window.innerWidth - 16,
    47→        });
    48→      } else {
    49→        // On desktop: position aligned to right edge of button
    50→        setMenuPos({
    51→          top: rect.bottom + 4,
    52→          left: rect.right - Math.max(140, rect.width),
    53→          width: Math.max(140, rect.width),
    54→        });
    55→      }
    56→    }
    57→    setOpen(!open);
    58→  };
    59→
    60→  return (
    61→    <div className="relative">
    62→      <button
    63→        ref={btnRef}
    64→        onClick={handleOpen}
    65→        className="flex items-center gap-1.5 h-8 px-3 rounded-lg bg-white/[0.06] hover:bg-white/[0.1] border border-white/[0.08] text-xs font-bold text-white/85 transition-all w-full sm:w-auto"
    66→      >
    67→        {label && <span className="text-[9px] font-bold text-white/35 uppercase tracking-wider">{label}</span>}
    68→        <span className="max-w-[110px] truncate flex-1 sm:flex-none text-left">{current?.label || value || "—"}</span>
    69→        <svg className={`w-3 h-3 text-white/40 transition-transform shrink-0 ${open ? "rotate-180" : ""}`} viewBox="0 0 24 24" fill="currentColor"><path d="M7 10l5 5 5-5z" /></svg>
    70→      </button>
    71→      {open && (
    72→        <>
    73→          <button className="fixed inset-0 z-30 cursor-default" onClick={() => setOpen(false)} aria-label="Close menu" />
    74→          <div
    75→            className="fixed bg-[#0a0a0a] border border-white/15 rounded-lg overflow-hidden py-1 shadow-2xl z-40 max-h-[280px] overflow-y-auto"
    76→            style={{ top: menuPos.top, left: menuPos.left, width: menuPos.width }}
    77→          >
    78→            {options.map(o => {
    79→              const disabled = disabledIds.includes(o.id);
    80→              return (
    81→                <button
    82→                  key={o.id}
    83→                  disabled={disabled}
    84→                  onClick={() => { if (!disabled) { onChange(o.id); setOpen(false); } }}
    85→                  className={`block w-full text-left px-3 py-2.5 sm:py-2 text-xs sm:text-xs transition-colors ${o.id === value ? "font-bold" : "text-white/60 hover:bg-white/10 hover:text-white"} ${disabled ? "opacity-30 cursor-not-allowed" : ""}`}
    86→                  style={o.id === value ? { color: ACCENT } : undefined}
    87→                >
    88→                  {o.label}
    89→                </button>
    90→              );
    91→            })}
    92→          </div>
    93→        </>
    94→      )}
    95→    </div>
    96→  );
    97→}

    99→// ─── Checkbox toggle (purple squares like Miruro) ────────────────
   100→function ToggleCheck({ label, state, onToggle }: { label: string; state: boolean; onToggle: () => void }) {
   101→  return (
   102→    <button onClick={onToggle} className="flex items-center gap-1.5 group shrink-0">
   103→      <div
   104→        className={`w-4 h-4 rounded border-2 flex items-center justify-center transition-all ${state ? "" : "bg-transparent border-white/20 group-hover:border-white/40"}`}
   105→        style={state ? { background: ACCENT, borderColor: ACCENT } : undefined}
   106→      >
   107→        {state && <svg className="w-2.5 h-2.5 text-black" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}><path strokeLinecap="round" strokeLinejoin="round" d="M5 13l4 4L19 7" /></svg>}
   108→      </div>
   109→      <span className={`text-xs font-medium transition-colors ${state ? "" : "text-white/40 group-hover:text-white/70"}`} style={state ? { color: ACCENT } : undefined}>{label}</span>
   110→    </button>
   111→  );
   112→}

   115→function SideMediaCard({ item, subtitle, navigate }: { item: any; subtitle?: string; navigate: (r: any) => void }) {
   116→  const title = item.title?.english || item.title?.romaji || item.title?.native || "Unknown";
   117→  const img = item.coverImage?.extraLarge || item.coverImage?.large || item.coverImage?.medium || "";
   118→  const format = (item.format || item.type || "").replace(/_/g, " ");
   119→  return (
   120→    <button
   121→      onClick={() => navigate({ page: "anime", id: String(item.id) })}
   122→      className="relative flex w-full items-stretch gap-3 rounded-xl overflow-hidden border border-white/[0.07] bg-white/[0.03] hover:bg-white/[0.06] hover:border-white/20 transition-all text-left group"
   123→    >
   124→      {/* Backdrop art (faint, right side) */}
   125→      {img && (
   126→        <div
   127→          className="absolute inset-0 opacity-[0.1] group-hover:opacity-[0.16] transition-opacity pointer-events-none"
   128→          style={{ backgroundImage: `url(${img})`, backgroundSize: "cover", backgroundPosition: "center 20%" }}
   129→        />
   130→      )}
   131→      <div className="relative shrink-0 w-[64px] self-stretch min-h-[92px] overflow-hidden bg-white/5">
   132→        {img
   133→          ? <img src={img} alt={title} className="absolute inset-0 w-full h-full object-cover" loading="lazy" />
   134→          : <div className="absolute inset-0 flex items-center justify-center text-white/15 font-bold text-lg">{title.charAt(0)}</div>}
   135→      </div>
   136→      <div className="relative flex flex-col justify-center gap-1.5 py-2.5 pr-3 min-w-0">
   137→        <div className="flex items-start gap-1.5">
   138→          <span className="mt-[5px] w-1.5 h-1.5 rounded-full shrink-0" style={{ background: "#60A5FA" }} />
   139→          <p className="text-[13px] font-semibold text-white leading-snug line-clamp-2">{title}</p>
   140→        </div>
   141→        <div className="flex items-center gap-2 text-[10px] font-bold text-white/45 pl-3">
   142→          {format && <span className="uppercase">{format}</span>}
   143→          {item.episodes ? (
   144→            <span className="flex items-center gap-0.5 bg-white/[0.07] rounded px-1 py-0.5">
   145→              <svg className="w-2.5 h-2.5" viewBox="0 0 24 24" fill="currentColor"><path d="M19 4H5a2 2 0 0 0-2 2v12a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2V6a2 2 0 0 0-2-2zm-8 9H6v-2h5v2zm7-4H6V7h12v2z"/></svg>
   146→              {item.episodes}
   147→            </span>
   148→          ) : null}
   149→          {item.averageScore ? (
   150→            <span className="flex items-center gap-0.5">
   151→              <svg className="w-2.5 h-2.5 text-white/45" fill="currentColor" viewBox="0 0 24 24"><path d="M12 17.27L18.18 21l-1.64-7.03L22 9.24l-7.19-.61L12 2 9.19 8.63 2 9.24l5.46 4.73L5.82 21z"/></svg>
   152→              {item.averageScore}
   153→            </span>
   154→          ) : null}
   155→          {subtitle && <span className="uppercase text-white/30">{subtitle}</span>}
   156→        </div>
   157→      </div>
   158→    </button>
   159→  );
   160→}

   216→    if (translation === "hardsub") {
   217→      // Hardsub tab: true hardsub servers + 4animo (user wants 4animo here too)
   218→      return s.type === "sub" && (s.hardsub === true || s.source === "animo4");
   219→    }

   705→// ============================================================
   706→// MiruroEpisodeSidebar — episode list panel
   707→//   - Bordered cards, spoiler blur w/ eye toggle
   708→//   - 1-100 range dropdown for long anime
   709→//   - Filter search
   710→//   - Next-episode countdown footer bar
   711→// ============================================================
   712→
   713→type EpisodeViewMode = "card" | "block" | "list";
   714→
   715→function MiruroEpisodeSidebar({
   716→  episodeList, filteredEps, epSearch, setEpSearch, switchEpisode,
   717→  episodeNum, nextEp, animeImage, visibleEpCount, setVisibleEpCount,
   718→  softsubAvailable, hardsubAvailable, dubAvailable,
   719→  animeNextAiring, countdown,
   720→}: any) {
   721→  const EPS_PER_PAGE = 100;
   722→  const totalEps = episodeList?.length || 0;
   723→  const totalPages = Math.ceil(totalEps / EPS_PER_PAGE);
   724→
   725→  const [spoilerOn, setSpoilerOn] = useState(true); // spoiler blur on by default
   726→  // View mode: "card" (picture + text), "block" (number grid), "list" (compact rows)
   727→  const [viewMode, setViewMode] = useState<EpisodeViewMode>("card");
   728→  // Search toggle — hidden by default, user clicks the magnifier to reveal
   729→  const [showSearch, setShowSearch] = useState(false);
   730→  // Range window starts at the page containing the current episode
   731→  // (e.g. open One Piece EP 540 → show 501-600)
   732→  const [page, setPage] = useState(() => Math.max(0, Math.floor((episodeNum - 1) / EPS_PER_PAGE)));
   733→  const [showPageMenu, setShowPageMenu] = useState(false);
   734→  const pageMenuBtnRef = useRef<HTMLButtonElement>(null);
   735→  const [pageMenuPos, setPageMenuPos] = useState<{ top: number; left: number; width: number }>({ top: 0, left: 0, width: 0 });
   736→  const listRef = useRef<HTMLDivElement | null>(null);
   737→
   738→  // When search is opened, focus the input
   739→  const searchInputRef = useRef<HTMLInputElement | null>(null);
   740→  useEffect(() => {
   741→    if (showSearch && searchInputRef.current) {
   742→      searchInputRef.current.focus();
   743→    }
   744→  }, [showSearch]);
   745→
   746→  // Derived-state adjustment during render (React's documented pattern):
   747→  // when the episode changes, jump the range window to its page.
   748→  // Search doesn't need a reset — pagination is bypassed while searching.
   749→  const [prevEpNum, setPrevEpNum] = useState(episodeNum);
   750→  if (prevEpNum !== episodeNum) {
   751→    setPrevEpNum(episodeNum);
   752→    const target = Math.floor((episodeNum - 1) / EPS_PER_PAGE);
   753→    if (target >= 0) setPage(target);
   754→  }
   755→
   756→  // Apply pagination: if not searching and >100 eps, show only current page
   757→  const pagedEps = (() => {
   758→    if (epSearch) return filteredEps; // search bypasses pagination
   759→    if (totalPages <= 1) return filteredEps;
   760→    const start = page * EPS_PER_PAGE;
   761→    return filteredEps?.slice(start, start + EPS_PER_PAGE) || [];
   762→  })();
   763→
   764→  const pageLabel = totalPages > 1
   765→    ? `${page * EPS_PER_PAGE + 1} - ${Math.min((page + 1) * EPS_PER_PAGE, totalEps)}`
   766→    : `1 - ${totalEps}`;
   767→
   768→  // Countdown footer: "Episode 1170 in 6d 17h · Sun, Jul 12, 06:22"
   769→  const countdownShort = (countdown || "").split(" ").slice(0, 2).join(" ");
   770→  const airDateLabel = animeNextAiring?.airingAt
   771→    ? new Date(animeNextAiring.airingAt * 1000).toLocaleString("en-US", {
   772→        weekday: "short", month: "short", day: "numeric",
   773→        hour: "2-digit", minute: "2-digit", hour12: false,
   774→      })
   775→    : "";
   776→
   777→  return (
   778→    <div className="flex flex-col w-full h-full bg-white/[0.02] rounded-xl border border-white/[0.06] overflow-hidden">
   779→      {/* ─── Header: Up Next + view toggle + search + spoiler toggle ─── */}
   780→      <div className="flex items-center justify-between px-3 py-2.5 border-b border-white/[0.06] gap-2">
   781→        <div className="flex flex-col gap-0.5 min-w-0 flex-1">
   782→          <div className="text-sm font-bold line-clamp-1">
