     1→'use client';
     2→
     3→import { useEffect, useRef, useState, useCallback } from 'react';
     4→import Hls from 'hls.js';
     5→import { proxify, proxifyM3u8 } from '@/lib/proxy';
     6→import { validateSkipTime } from '@/lib/episode-metadata';
     7→import {
     8→  useSubtitleSettings,
     9→  CustomSubtitleOverlay,
    10→  SubtitleSettingsPanel,
    11→} from './subtitle-settings';
    12→
    13→interface HLSPlayerProps {
    14→  url: string;
    15→  animeId?: string;
    16→  episodeNum?: number;
    17→  sourceType?: 'hls' | 'mp4';
    18→  intro?: { start: number; end: number } | null;
    19→  outro?: { start: number; end: number } | null;
    20→  allStreams?: Array<{ url: string; quality: string; label: string }>;
    21→  subtitleTracks?: Array<{ url: string; lang: string; label: string }>;
    22→  onEnded?: () => void;
    23→  onProviderFailed?: () => void;
    24→  onCanPlay?: () => void;
    25→  autoplay?: boolean;
    26→  autoSkip?: boolean;
    27→}
    28→
    29→// ============================================================
    30→// Glass Player — Animetsu-inspired glassmorphism design
    31→// Features:
    32→//   - Floating glass control bar (backdrop-blur, translucent)
    33→//   - Animated progress bar with glow + gradient
    34→//   - Smooth scale/translate animations on all interactions
    35→//   - Glass menus for quality/subtitles/speed
    36→//   - Animated center play button with pulse ring
    37→//   - Double-tap to seek with ripple animation
    38→//   - Keyboard shortcuts
    39→// ============================================================
    40→
    41→export default function HLSPlayerNew({
    42→  url, animeId, episodeNum, sourceType = 'hls',
    43→  intro: introProp, outro: outroProp, allStreams, subtitleTracks, onEnded, onProviderFailed, onCanPlay, autoplay = true, autoSkip = false,
    44→}: HLSPlayerProps) {
    45→  const videoRef = useRef<HTMLVideoElement>(null);
    46→  const hlsRef = useRef<Hls | null>(null);
    47→  const containerRef = useRef<HTMLDivElement>(null);
    48→
    49→  // Defense-in-depth: validate skip times at the player level too.
    50→  // Even if upstream code fails to filter bad data, this guarantees
    51→  // the outro button never appears at the start of the video (which
    52→  // happens when outro.start = 0 from a provider's "no data" sentinel).
    53→  const intro = validateSkipTime(introProp ?? null, "intro");
    54→  const outro = validateSkipTime(outroProp ?? null, "outro");
    55→
    56→  // Keep the latest callbacks in refs so the stream-loading and video-event
    57→  // effects DON'T list them as deps. Parents (e.g. watch-page-shell) re-render
    58→  // on every `timeupdate` (several times/sec) and often pass inline arrow
    59→  // callbacks — a new identity each render. If those were in the effect deps,
    60→  // the stream effect would destroy + recreate the hls.js instance on every
    61→  // frame, refetching the m3u8 + all segments in an infinite loop. Reading
    62→  // through a ref keeps the effect stable (only url/sourceType/autoplay matter).
    63→  const onProviderFailedRef = useRef(onProviderFailed);
    64→  const onEndedRef = useRef(onEnded);
    65→  const onCanPlayRef = useRef(onCanPlay);
    66→  const onCanPlayFiredRef = useRef(false);
    67→  const autoSkipRef = useRef(autoSkip);
    68→  onProviderFailedRef.current = onProviderFailed;
    69→  onEndedRef.current = onEnded;
    70→  onCanPlayRef.current = onCanPlay;
    71→  autoSkipRef.current = autoSkip;
    72→  const [playing, setPlaying] = useState(false);
    73→  const [currentTime, setCurrentTime] = useState(0);
    74→  const [duration, setDuration] = useState(0);
    75→  const [buffered, setBuffered] = useState(0);
    76→  const [volume, setVolume] = useState(1);
    77→  const [muted, setMuted] = useState(false);
    78→  const [fullscreen, setFullscreen] = useState(false);
    79→  const [loading, setLoading] = useState(true);
    80→  const [error, setError] = useState<string | null>(null);
    81→  const [showControls, setShowControls] = useState(true);
    82→  const isMobileRef = useRef(false);
    83→  if (typeof window !== 'undefined') isMobileRef.current = 'ontouchstart' in window || window.innerWidth < 768;
    84→  const [qualities, setQualities] = useState<any[]>([]);
    85→  const [currentQuality, setCurrentQuality] = useState(-1);
    86→  const [activeMenu, setActiveMenu] = useState<'quality' | 'subtitles' | 'speed' | null>(null);
    87→  const [showSubtitleSettings, setShowSubtitleSettings] = useState(false);
    88→  const [showDownloadModal, setShowDownloadModal] = useState(false);
    89→  const { settings: subtitleSettings, update: updateSubtitleSettings, reset: resetSubtitleSettings } = useSubtitleSettings();
    90→  const [currentSubtitle, setCurrentSubtitle] = useState(-1);
    91→  const currentSubtitleRef = useRef(currentSubtitle);
    92→  currentSubtitleRef.current = currentSubtitle;
    93→  const [hlsSubtitles, setHlsSubtitles] = useState<any[]>([]);
    94→  const [playbackRate, setPlaybackRate] = useState(1);
    95→  const [showSkipIntro, setShowSkipIntro] = useState(false);
    96→  const [showSkipOutro, setShowSkipOutro] = useState(false);
    97→  // Track when user clicked skip — prevents the button from re-appearing
    98→  // immediately after the click (the timeupdate loop would otherwise keep
    99→  // showing it because video.currentTime is still in the skip range).
   100→  const skipIntroClickedRef = useRef(0);
   101→  const skipOutroClickedRef = useRef(0);
   102→  const [hoverTime, setHoverTime] = useState<number | null>(null);
   103→  const [hoverX, setHoverX] = useState(0);
   104→  const [seekRipple, setSeekRipple] = useState<{ x: number; dir: 'left' | 'right' } | null>(null);
   105→  const [volumeHover, setVolumeHover] = useState(false);
   106→  const [screenshotFlash, setScreenshotFlash] = useState(false);
   107→  const [screenshotToast, setScreenshotToast] = useState<string | null>(null);
   108→  const [downloadLoading, setDownloadLoading] = useState(false);
   109→  const [downloadToast, setDownloadToast] = useState<string | null>(null);
   110→  const [streamHealth, setStreamHealth] = useState<'good' | 'fair' | 'poor' | 'unknown'>('unknown');
   111→  const [showStreamInfo, setShowStreamInfo] = useState(false);
   112→  const controlsTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   113→  const retryCountRef = useRef(0);
   114→  const lastTapRef = useRef<{ time: number; x: number } | null>(null);
   115→  const canvasRef = useRef<HTMLCanvasElement>(null);
   116→
   117→  // ─── Load stream ──────────────────────────────────────────────────
   118→  useEffect(() => {
   119→    const video = videoRef.current;
   120→    if (!video || !url) return;
   121→
   122→    setLoading(true);
   123→    setError(null);
   124→    retryCountRef.current = 0;
   125→    setCurrentSubtitle(-1);
   126→    setHlsSubtitles([]);
   127→    onCanPlayFiredRef.current = false; // reset for new stream
   128→
   129→    if (hlsRef.current) {
   130→      hlsRef.current.destroy();
   131→      hlsRef.current = null;
   132→    }
   133→
   134→    if (sourceType === 'mp4' || url.endsWith('.mp4') || url.includes('video.mp4')) {
   135→      // Proxy the MP4 URL through the worker so the correct Referer is sent
   136→      video.src = proxify(url);
   137→      video.load();
   138→      if (autoplay) video.play().catch(() => {});
   139→      return;
   140→    }
   141→
   142→    if (Hls.isSupported()) {
   143→      const hls = new Hls({
   144→        enableWorker: true,
   145→        // ─── PERFORMANCE TUNING — optimized for FAST initial load ───
   146→        // Previous values were bloated (90s back buffer, 45s frag timeout)
   147→        // which caused slow initial playback. These values prioritize:
   148→        // 1. FASTEST time-to-first-frame (TTFF)
   149→        // 2. Smooth playback once buffering starts
   150→        // 3. Quick failure + retry on bad segments
   151→        lowLatencyMode: false,
   152→        backBufferLength: 30,           // was 90 — 30s back buffer is plenty, saves memory
   153→        maxBufferLength: 30,            // was 90 — 30s ahead buffer for fast start, grows later
   154→        maxMaxBufferLength: 60,         // was 180 — 60s max buffer (1 min ahead is enough)
   155→        maxBufferSize: 60 * 1000 * 1000, // was 90MB — 60MB max buffer size
   156→        maxBufferHole: 0.5,
   157→        // Start at the HIGHEST quality level (1080p), not lowest (360p).
   158→        // -1 = auto-select based on bandwidth, but we set a high default estimate
   159→        // so it picks the best quality immediately on fast connections.
   160→        startLevel: -1,
   161→        // Assume 15Mbps bandwidth by default — start at 1080p immediately.
   162→        // Higher estimate = less quality-shifting at start = faster perceived load.
   163→        abrEwmaDefaultEstimate: 15000000,  // was 10Mbps → 15Mbps
   164→        abrBandWidthFactor: 0.95,
   165→        abrBandWidthUpFactor: 0.7,
   166→        maxStarvationDelay: 2,          // was 4 — switch quality faster if buffer runs dry
   167→        abrEwmaDefaultEstimateMax: 30000000,  // was 20Mbps → 30Mbps
   168→        // ─── TIMEOUTS — fail FAST, retry FAST ───
   169→        // 20s manifest timeout is way too long — user sees loading spinner forever.
   170→        // 8s is enough for any CDN + proxy round-trip.
   171→        manifestLoadingTimeOut: 8000,   // was 20000 — fail faster on dead servers
   172→        manifestLoadingMaxRetry: 3,     // was 4
   173→        levelLoadingTimeOut: 8000,      // was 20000
   174→        levelLoadingMaxRetry: 3,        // was 4
   175→        fragLoadingTimeOut: 15000,      // was 45000 — 15s per segment is generous
   176→        fragLoadingMaxRetry: 4,         // was 8 — don't waste time on dead segments
   177→        fragLoadingRetryDelay: 500,     // was 1000 — retry faster
   178→        // ─── PREFETCH + PROGRESSIVE for instant start ───
   179→        startFragPrefetch: true,        // Prefetch first fragment while parsing manifest
   180→        progressive: true,              // Stream data progressively (faster TTFF)
   181→        // ─── SUBTITLES ───
   182→        enableWebVTT: true,
   183→        xhrSetup: (xhr) => { xhr.withCredentials = false; },
   184→      });
   185→      hlsRef.current = hls;
   186→      hls.loadSource(proxifyM3u8(url));
   187→      hls.attachMedia(video);
   188→
   189→      hls.on(Hls.Events.MANIFEST_PARSED, (_e, data) => {
   190→        setQualities(data.levels);
   191→        setLoading(false);
   192→        // Reset retry counter on successful manifest parse
   193→        retryCountRef.current = 0;
   194→        if (autoplay) {
   195→          video.play().catch(() => {
   196→            video.muted = true;
   197→            video.play().catch(() => {});
   198→          });
   199→        }
   200→      });
   201→
   202→      hls.on(Hls.Events.SUBTITLE_TRACKS_UPDATED, (_e, data) => {
   203→        setHlsSubtitles(data.subtitleTracks || []);
   204→        if (data.subtitleTracks.length > 0 && hls.subtitleTrack === -1) {
   205→          hls.subtitleTrack = 0;
   206→          setCurrentSubtitle(0);
   207→          // After HLS.js sets the subtitle track, it creates a textTrack with
   208→          // mode='showing' (native rendering). We need to switch it to 'hidden'
   209→          // so only our CustomSubtitleOverlay renders (prevents double subtitles).
   210→          setTimeout(() => {
   211→            const video = videoRef.current;
   212→            if (video) {
   213→              for (let i = 0; i < video.textTracks.length; i++) {
   214→                if (i === 0) video.textTracks[i].mode = 'hidden';
   215→              }
   216→            }
   217→          }, 200);
   218→        }
   219→      });
   220→
   221→      hls.on(Hls.Events.LEVEL_SWITCHED, (_e, data) => {
   222→        setCurrentQuality(data.level);
   223→        // After a level switch, HLS.js may re-enable native subtitle rendering.
   224→        // Force the active track back to 'hidden' mode so only our overlay shows.
   225→        if (currentSubtitleRef.current >= 0) {
   226→          setTimeout(() => {
   227→            const video = videoRef.current;
   228→            if (video && video.textTracks.length > currentSubtitleRef.current) {
   229→              video.textTracks[currentSubtitleRef.current].mode = 'hidden';
   230→            }
   231→          }, 100);
   232→        }
   233→      });
   234→
   235→      hls.on(Hls.Events.ERROR, (_e, data) => {
   236→        // Non-fatal errors are expected during streaming (a segment/playlist
   237→        // fetch hiccup) and hls.js already recovers from them on its own —
   238→        // console.warn so they don't trip Next's dev error overlay for
   239→        // something that isn't actually broken.
   240→        const log = data.fatal ? console.error : console.warn;
   241→        log(`[HLS] Error: type=${data.type} details=${data.details} fatal=${data.fatal}`);
   242→        if (data.fatal) {
   243→          if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
   244→            if (retryCountRef.current < 3) {
   245→              retryCountRef.current++;
   246→              setTimeout(() => hls.startLoad(), 2000);
   247→            } else {
   248→              setError('Stream failed. Try another server.');
   249→              setLoading(false);
   250→              onProviderFailedRef.current?.();
   251→            }
   252→          } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
   253→            hls.recoverMediaError();
   254→          } else {
   255→            setError('Playback error. Try another server.');
   256→            setLoading(false);
   257→            onProviderFailedRef.current?.();
   258→          }
   259→        }
   260→      });
   261→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   262→      // Safari/iOS — native HLS
   263→      video.src = proxifyM3u8(url);
   264→      const onNativeError = () => { setError('Stream failed to load.'); setLoading(false); };
   265→      video.addEventListener('error', onNativeError);
   266→      if (autoplay) video.play().catch(() => {});
   267→      return () => {
   268→        video.removeEventListener('error', onNativeError);
   269→      };
   270→    } else {
   271→      setError('HLS not supported');
   272→      setLoading(false);
   273→    }
   274→
   275→    return () => {
   276→      if (hlsRef.current) {
   277→        hlsRef.current.destroy();
   278→        hlsRef.current = null;
   279→      }
   280→    };
   281→  }, [url, sourceType, autoplay]);
   282→
   283→  // ─── Video events ─────────────────────────────────────────────────
   284→  useEffect(() => {
   285→    const video = videoRef.current;
   286→    if (!video) return;
   287→
   288→    const onPlay = () => setPlaying(true);
   289→    const onPause = () => setPlaying(false);
   290→    const onTime = () => {
   291→      setCurrentTime(video.currentTime);
   292→      if (video.buffered.length > 0) {
   293→        setBuffered(video.buffered.end(video.buffered.length - 1));
   294→      }
   295→      // Auto Skip — when enabled, jump straight past the opening/ending.
   296→      if (autoSkipRef.current) {
   297→        if (intro && video.currentTime >= intro.start && video.currentTime < intro.end - 0.3) {
   298→          video.currentTime = intro.end;
   299→          return;
   300→        }
   301→        if (outro && video.currentTime >= outro.start && video.currentTime < outro.end - 0.3) {
   302→          video.currentTime = Math.min(outro.end, video.duration || outro.end);
   303→          return;
   304→        }
   305→      }
   306→      // Show skip intro button from 3s BEFORE intro starts until 5s AFTER it ends.
   307→      // Also show immediately if video just loaded and we're already in the intro range
   308→      // (happens when mimi loads super fast and starts playing at 0s, which is before intro.start)
   309→      //
   310→      // BUT: if the user just clicked Skip Intro (within last 8s), DON'T re-show the
   311→      // button — otherwise it instantly pops back up because video.currentTime is still
   312→      // in the intro range right after the seek.
   313→      const sinceIntroClick = Date.now() - skipIntroClickedRef.current;
   314→      const sinceOutroClick = Date.now() - skipOutroClickedRef.current;
   315→
   316→      if (intro && video.currentTime >= (intro.start - 3) && video.currentTime < (intro.end + 5) && sinceIntroClick > 8000) {
   317→        setShowSkipIntro(true);
   318→      } else {
   319→        setShowSkipIntro(false);
   320→      }
   321→      if (outro && video.currentTime >= (outro.start - 3) && video.currentTime < (outro.end + 10) && sinceOutroClick > 8000) {
   322→        setShowSkipOutro(true);
   323→      } else {
   324→        setShowSkipOutro(false);
   325→      }
   326→    };
   327→
   328→    // Also check on MANIFEST_PARSED / loadedmetadata — if intro data is available
   329→    // and we're at position 0, show the skip button immediately (before timeupdate fires)
   330→    const onLoadedMetadata = () => {
   331→      setDuration(video.duration || 0);
   332→      // If intro starts soon (within first 60s), show the button immediately
   333→      if (intro && intro.start < 60 && video.currentTime < intro.end) {
   334→        setShowSkipIntro(true);
   335→      }
   336→      if (outro && video.duration && outro.start < video.duration) {
   337→        // Don't show outro yet, just confirm we have the data
   338→      }
   339→    };
   340→    const onDur = () => setDuration(video.duration || 0);
   341→
   342→    let waitingTimer: ReturnType<typeof setTimeout> | null = null;
   343→    const onWaiting = () => {
   344→      if (waitingTimer) clearTimeout(waitingTimer);
   345→      // Show loading after 1.5s of waiting (was 5s — too long, user gets confused)
   346→      waitingTimer = setTimeout(() => {
   347→        if (video.readyState < 3) setLoading(true);
   348→      }, 1500);
   349→    };
   350→    const onPlaying = () => {
   351→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   352→      setLoading(false);
   353→    };
   354→    const onCanPlay = () => {
   355→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   356→      setLoading(false);
   357→      // Notify parent that video is ready to play (for loading screen dismissal)
   358→      if (!onCanPlayFiredRef.current) {
   359→        onCanPlayFiredRef.current = true;
   360→        onCanPlayRef.current?.();
   361→      }
   362→    };
   363→    const onEnd = () => onEndedRef.current?.();
   364→    const onErr = () => { setError('Playback error.'); setLoading(false); };
   365→
   366→    video.addEventListener('play', onPlay);
   367→    video.addEventListener('pause', onPause);
   368→    video.addEventListener('timeupdate', onTime);
   369→    video.addEventListener('durationchange', onDur);
   370→    video.addEventListener('loadedmetadata', onLoadedMetadata);
   371→    video.addEventListener('waiting', onWaiting);
   372→    video.addEventListener('playing', onPlaying);
   373→    video.addEventListener('canplay', onCanPlay);
   374→    video.addEventListener('ended', onEnd);
   375→    video.addEventListener('error', onErr);
   376→
   377→    return () => {
   378→      video.removeEventListener('play', onPlay);
   379→      video.removeEventListener('pause', onPause);
   380→      video.removeEventListener('timeupdate', onTime);
   381→      video.removeEventListener('durationchange', onDur);
   382→      video.removeEventListener('loadedmetadata', onLoadedMetadata);
   383→      video.removeEventListener('waiting', onWaiting);
   384→      video.removeEventListener('playing', onPlaying);
   385→      video.removeEventListener('canplay', onCanPlay);
   386→      video.removeEventListener('ended', onEnd);
   387→      video.removeEventListener('error', onErr);
   388→    };
   389→  }, [intro, outro]);
   390→
   391→  // ─── Controls auto-hide ───────────────────────────────────────────
   392→  const CONTROLS_TIMEOUT = isMobileRef.current ? 2500 : 3500;
   393→  const showControlsTemp = useCallback(() => {
   394→    setShowControls(true);
   395→    if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   396→    controlsTimeoutRef.current = setTimeout(() => {
   397→      if (playing) { setShowControls(false); setActiveMenu(null); }
   398→    }, CONTROLS_TIMEOUT);
   399→  }, [playing, CONTROLS_TIMEOUT]);
   400→
   401→  // ─── Auto-hide controls when video starts playing ─────────────────
   402→  // On mobile, there's no mousemove to trigger showControlsTemp(),
   403→  // so the controls stay visible forever. This effect starts the
   404→  // auto-hide timer whenever playback begins.
   405→  useEffect(() => {
   406→    if (playing) {
   407→      // Start the auto-hide timer — controls will disappear after CONTROLS_TIMEOUT
   408→      if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   409→      controlsTimeoutRef.current = setTimeout(() => {
   410→        setShowControls(false);
   411→        setActiveMenu(null);
   412→      }, CONTROLS_TIMEOUT);
   413→    } else {
   414→      // Paused — show controls and cancel any pending hide
   415→      if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   416→      setShowControls(true);
   417→    }
   418→  }, [playing, CONTROLS_TIMEOUT]);
   419→
   420→  // ─── Fullscreen ───────────────────────────────────────────────────
   421→  useEffect(() => {
   422→    const onFs = () => setFullscreen(!!document.fullscreenElement);
   423→    document.addEventListener('fullscreenchange', onFs);
   424→    return () => document.removeEventListener('fullscreenchange', onFs);
   425→  }, []);
   426→
   427→  const toggleFullscreen = async () => {
   428→    try {
   429→      if (document.fullscreenElement) {
   430→        await document.exitFullscreen();
   431→        // Release orientation lock when exiting fullscreen
   432→        try { await (screen.orientation as any)?.unlock(); } catch {}
   433→      } else {
   434→        await containerRef.current?.requestFullscreen();
   435→        // On mobile, lock to landscape so the video fills the screen
   436→        try {
   437→          if (screen.orientation && typeof (screen.orientation as any).lock === 'function') {
   438→            await (screen.orientation as any).lock('landscape');
   439→          }
   440→        } catch {
   441→          // Orientation lock not supported or denied — fullscreen still works
   442→        }
   443→      }
   444→    } catch {
   445→      // Fullscreen request failed silently
   446→    }
   447→  };
   448→
   449→  const seek = (time: number) => { if (videoRef.current) videoRef.current.currentTime = time; };
   450→  const togglePlay = () => {
   451→    const v = videoRef.current; if (!v) return;
   452→    if (v.paused) v.play(); else v.pause();
   453→  };
   454→  const toggleMute = () => {
   455→    const v = videoRef.current; if (!v) return;
   456→    v.muted = !v.muted; setMuted(v.muted);
   457→  };
   458→  const changeVolume = (vol: number) => {
   459→    const v = videoRef.current; if (!v) return;
   460→    v.volume = vol; v.muted = vol === 0;
   461→    setVolume(vol); setMuted(vol === 0);
   462→  };
   463→  const changeQuality = (level: number) => {
   464→    if (hlsRef.current) { hlsRef.current.currentLevel = level; setCurrentQuality(level); }
   465→    setActiveMenu(null);
   466→  };
   467→
   468→  // ── Auto-enable the first external subtitle track ──
   469→  // The `default` attribute on <track> doesn't always work reliably across
   470→  // browsers (especially when the track loads after the video starts). This
   471→  // effect explicitly enables the first external subtitle track once the
   472→  // video metadata has loaded AND there are no HLS-embedded subtitles.
   473→  // This ensures subtitles show automatically when the user starts watching.
   474→  useEffect(() => {
   475→    if (hlsSubtitles.length > 0) return; // HLS-embedded subs handle their own default
   476→    if (!subtitleTracks || subtitleTracks.length === 0) return;
   477→    const video = videoRef.current;
   478→    if (!video) return;
   479→
   480→    const enableFirstSub = () => {
   481→      // video.textTracks includes both HLS-embedded and external <track> elements.
   482→      // External tracks come AFTER HLS tracks in the list. Since we have no HLS
   483→      // subs (checked above), the first external track is at index 0.

  1276→// ─── Download Modal ─────────────────────────────────────────────
  1277→function DownloadModal({ animeId, episodeNum, onClose }: { animeId: string; episodeNum: number; onClose: () => void }) {
  1278→  const [loading, setLoading] = useState(true);
  1279→  const [links, setLinks] = useState<Array<{ text: string; decodedUrl: string }>>([]);
  1280→  const [error, setError] = useState('');
  1281→
  1282→  useEffect(() => {
  1283→    const fetchDownloads = async () => {
  1284→      try {
  1285→        setLoading(true);
  1286→        setError('');
  1287→        let title = '';
  1288→        try {
  1289→          const titleRes = await fetch('https://graphql.anilist.co', {
  1290→            method: 'POST',
  1291→            headers: { 'Content-Type': 'application/json' },
  1292→            body: JSON.stringify({
  1293→              query: `query($id:Int){Media(id:$id,type:ANIME){title{english romaji}}}`,
  1294→              variables: { id: parseInt(animeId) },
  1295→            }),
  1296→          });
  1297→          if (titleRes.ok) {
  1298→            const titleData = await titleRes.json();
  1299→            title = titleData?.data?.Media?.title?.english || titleData?.data?.Media?.title?.romaji || '';
  1300→          }
  1301→        } catch { /* ignore */ }
  1302→
  1303→        if (!title) {
  1304→          setError('Could not determine anime title');
  1305→          setLoading(false);
  1306→          return;
  1307→        }
  1308→
  1309→        const res = await fetch(`/api/anime/download?title=${encodeURIComponent(title)}&auto=1`);
  1310→        if (!res.ok) {
  1311→          setError('Failed to fetch download links');
  1312→          setLoading(false);
  1313→          return;
  1314→        }
  1315→        const data = await res.json();
  1316→        if (data.links && data.links.length > 0) {
  1317→          setLinks(data.links);
  1318→        } else {
  1319→          setError('No download links found for this anime');
  1320→        }
  1321→      } catch (e: any) {
  1322→        setError(e?.message || 'Failed to load downloads');
  1323→      } finally {
  1324→        setLoading(false);
  1325→      }
  1326→    };
  1327→    fetchDownloads();
  1328→  }, [animeId]);
  1329→
  1330→  // Show only top 2 links (quick access) + "explore more" link
  1331→  const quickLinks = links.slice(0, 2);
  1332→
  1333→  return (
  1334→    <div
  1335→      className="absolute inset-0 z-50 flex items-center justify-center"
  1336→      style={{ background: 'rgba(0,0,0,0.85)', backdropFilter: 'blur(4px)' }}
  1337→      onClick={onClose}
  1338→    >
  1339→      <div
  1340→        className="bg-black border border-white/10 rounded-2xl shadow-2xl w-full max-w-sm mx-4 overflow-hidden"
  1341→        onClick={(e) => e.stopPropagation()}
  1342→      >
  1343→        {/* Header */}
  1344→        <div className="flex items-center justify-between px-5 py-3.5 border-b border-white/10">
  1345→          <div className="flex items-center gap-2">
  1346→            <svg className="w-4 h-4 text-white" viewBox="0 0 24 24" fill="currentColor"><path d="M19 9h-4V5a3 3 0 0 0-6 0v4H5a2 2 0 0 0-2 2v8a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-8a2 2 0 0 0-2-2zm-6 6v3h-2v-3H8l4-4 4 4h-3z" /></svg>
  1347→            <h3 className="text-sm font-bold text-white">Download</h3>
  1348→          </div>
  1349→          <button onClick={onClose} className="text-white/40 hover:text-white transition-colors">
  1350→            <svg className="w-4 h-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2}><path d="M18 6L6 18M6 6l12 12" strokeLinecap="round" strokeLinejoin="round" /></svg>
  1351→          </button>
  1352→        </div>
  1353→
  1354→        {/* Body */}
  1355→        <div className="px-4 py-4">
  1356→          {loading && (
  1357→            <div className="flex items-center justify-center py-8">
  1358→              <div className="w-5 h-5 border-2 border-white/20 border-t-white rounded-full animate-spin" />
  1359→              <span className="ml-2.5 text-xs text-white/50">Finding links...</span>
  1360→            </div>
  1361→          )}
  1362→
  1363→          {error && !loading && (
  1364→            <div className="text-center py-6">
  1365→              <p className="text-xs text-white/40 mb-3">{error}</p>
  1366→              {/* Even on error, show the explore link */}
  1367→              <a
  1368→                href="https://animex.one/community/download"
  1369→                target="_blank"
  1370→                rel="noopener noreferrer"
  1371→                className="inline-flex items-center gap-1.5 text-[11px] font-semibold text-[#1E88FF] hover:underline"
  1372→              >
  1373→                Want to explore downloads?
  1374→                <svg className="w-3 h-3" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2}><path d="M7 17L17 7M7 7h10v10" strokeLinecap="round" strokeLinejoin="round" /></svg>
  1375→              </a>
  1376→            </div>
  1377→          )}
  1378→
  1379→          {!loading && !error && quickLinks.length > 0 && (

  1392→                      <div className="w-7 h-7 rounded-lg bg-white/5 flex items-center justify-center shrink-0 group-hover:bg-white/10 transition-colors">

  1406→
  1407→              {/* "Want to explore downloads?" link at bottom */}
  1408→              <div className="mt-3 pt-3 border-t border-white/8 text-center">
