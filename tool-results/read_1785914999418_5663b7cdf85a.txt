    24→    24→function MenuSelect({ label, value, options, onChange, disabledIds = [] }: {
    25→    25→  label?: string;
    26→    26→  value: string;
    27→    27→  options: Array<{ id: string; label: string }>;
    28→    28→  onChange: (id: string) => void;
    29→    29→  disabledIds?: string[];
    30→    30→}) {
    31→    31→  const [open, setOpen] = useState(false);
    32→    32→  const current = options.find(o => o.id === value);
    33→    33→  return (
    34→    34→    <div className="relative">
    35→    35→      <button
    36→    36→        onClick={() => setOpen(!open)}
    37→    37→        className="flex items-center gap-1.5 h-8 px-3 rounded-lg bg-white/[0.06] hover:bg-white/[0.1] border border-white/[0.08] text-xs font-bold text-white/85 transition-all"
    38→    38→      >
    39→    39→        {label && <span className="text-[9px] font-bold text-white/35 uppercase tracking-wider">{label}</span>}
    40→    40→        <span className="max-w-[110px] truncate">{current?.label || value || "—"}</span>
    41→    41→        <svg className={`w-3 h-3 text-white/40 transition-transform ${open ? "rotate-180" : ""}`} viewBox="0 0 24 24" fill="currentColor"><path d="M7 10l5 5 5-5z" /></svg>
    42→    42→      </button>
    43→    43→      {open && (
    44→    44→        <>
    45→    45→          <button className="fixed inset-0 z-30 cursor-default" onClick={() => setOpen(false)} aria-label="Close menu" />
    46→    46→          <div className="absolute top-full right-0 mt-1 bg-[#0a0a0a] border border-white/15 rounded-lg overflow-hidden min-w-[140px] py-1 shadow-2xl z-40 max-h-[280px] overflow-y-auto">
    47→    47→            {options.map(o => {
    48→    48→              const disabled = disabledIds.includes(o.id);
    49→    49→              return (
    50→    50→                <button
    51→    51→                  key={o.id}
    52→    52→                  disabled={disabled}
    53→    53→                  onClick={() => { if (!disabled) { onChange(o.id); setOpen(false); } }}
    54→    54→                  className={`block w-full text-left px-3 py-2 text-xs transition-colors ${o.id === value ? "font-bold" : "text-white/60 hover:bg-white/10 hover:text-white"} ${disabled ? "opacity-30 cursor-not-allowed" : ""}`}
    55→    55→                  style={o.id === value ? { color: ACCENT } : undefined}
    56→    56→                >
    57→    57→                  {o.label}
    58→    58→                </button>
    59→    59→              );
    60→    60→            })}
    61→    61→          </div>
    62→    62→        </>
    63→    63→      )}
    64→    64→    </div>
    65→    65→  );
    66→    66→}
