   341→export const useAppStore = create<AppState>()(
   342→  persist(
   343→    (set, get) => ({
   344→  route: { page: "home" },
   345→  sectionSubPage: "home",
   346→  setSectionSubPage: (subPage) => {
   347→    set({ sectionSubPage: subPage });
   348→    // Sync the URL bar when sub-page changes (e.g. Browse, Schedule, Genres)
   349→    if (typeof window !== "undefined") {
   350→      const currentRoute = get().route;
   351→      if (currentRoute.page === "home") {
   352→        const subPagePath: Record<string, string> = {
   353→          "home": "/",
   354→          "browse": "/browse",
   355→          "schedule": "/schedule",
   356→          "genres": "/genres",
   357→          "sub": "/sub",
   358→          "dub": "/dub",
   359→          "trending": "/trending",
   360→          "top-rated": "/top-rated",
   361→        };
   362→        const path = subPagePath[subPage] || `/${subPage}`;
   363→        history.replaceState(null, "", path);
   364→      }
   365→    }
   366→  },
   367→  navigate: (route) => {
   368→    // Reset section sub-page when navigating to a new section
   369→    const subPage = "home";
   370→    set({ route, sectionSubPage: subPage });
   371→    if (typeof window !== "undefined") {
   372→      let path = "/";
   373→      if (route.page === "landing") path = "/";
   374→      else if (route.page === "hub") path = "/hub";
   375→      else if (route.page === "home") path = "/";
   376→      else if (route.page === "discover") path = "/discover";
   377→      else if (route.page === "search" && route.query)
   378→        path = `/search/${encodeURIComponent(route.query)}`;
   379→      else if (route.page === "search") path = "/search";
   380→      else if (route.page === "anime")
   381→        path = `/anime/${route.id}`;
   382→      else if (route.page === "watch")
   383→        path = `/watch/${route.id}/${route.episode}`;
   384→      else if (route.page === "genre")
   385→        path = `/genre/${encodeURIComponent(route.genre)}`;
   386→      else if (route.page === "bookmarks") path = "/bookmarks";
   387→      else if (route.page === "watchlist") path = "/watchlist";
   388→      else if (route.page === "history") path = "/history";
   389→      else if (route.page === "movies") path = "/movies";
   390→      else if (route.page === "tv") path = "/tv";
   391→      else if (route.page === "manga") path = "/manga";
   392→      else if (route.page === "manga-detail")
   393→        path = `/manga/${route.id}`;
   394→      else if (route.page === "manga-read")
   395→        path = `/read-manga/${route.id}/${route.chapterId}`;
   396→      else if (route.page === "movie-detail")
   397→        path = `/movie/${route.id}`;
   398→      else if (route.page === "tv-detail")
   399→        path = `/tvshow/${route.id}`;
   400→      else if (route.page === "movie-watch")
   401→        path = `/watch-movie/${route.id}`;
   402→      else if (route.page === "tv-watch")
   403→        path = `/watch-tv/${route.id}/${route.season}/${route.episode}`;
   404→      else if (route.page === "watchnow") path = "/watchnow";
   405→      else if (route.page === "contact") path = "/contact";
   406→      else if (route.page === "donate") path = "/donate";
   407→      else if (route.page === "updates") path = "/updates";
   408→      else if (route.page === "donate-crypto") path = "/donate/crypto";
   409→      else if (route.page === "features") path = "/features";
   410→      else if (route.page === "novel") path = "/novel";
   411→      else if (route.page === "novel-detail") path = `/novel/${encodeURIComponent(route.novelId)}`;
   412→      else if (route.page === "novel-read") path = `/read-novel/${encodeURIComponent(route.novelId)}/${route.chapterNum}`;
   413→      else if (route.page === "signin") path = "/signin";
   414→      else if (route.page === "signup") path = "/signup";
   415→      else if (route.page === "profile") path = "/profile";
   416→      else if (route.page === "settings") path = "/settings";
   417→      else if (route.page === "music") path = "/music";
   418→      else if (route.page === "torrent") path = "/torrent";
   419→      else if (route.page === "download") path = "/download";
   420→      else if (route.page === "scraper") path = "/scraper";
   421→      else if (route.page === "scraper-anime") path = `/scraper/anime/${route.id}`;
   422→      else if (route.page === "scraper-watch") path = `/scraper/watch/${route.site}/${route.id}/${encodeURIComponent(route.episode)}`;
   423→      history.pushState(null, "", path);
   424→      window.scrollTo(0, 0);
   425→    }
   426→  },
   427→  bookmarks: [],
   428→  setBookmarks: (items) => set({ bookmarks: items }),
   429→  history: [],
   430→  setHistory: (items) => set({ history: items }),
   431→  addToHistory: (item) => set((state) => {
   432→    const id = `${item.animeId}-${item.episodeNum}`;
   433→    const now = new Date().toISOString();
   434→    // Remove any existing entry for this anime+episode, then prepend
   435→    const filtered = state.history.filter(
   436→      (h) => !(h.animeId === item.animeId && h.episodeNum === item.episodeNum)
   437→    );
   438→    const newEntry: HistoryItem = {
   439→      id,
   440→      animeId: item.animeId,
   441→      animeName: item.animeName,
   442→      thumbnail: item.thumbnail,
   443→      episodeNum: item.episodeNum,
   444→      progress: item.progress || 0,
   445→      duration: item.duration || 0,
   446→      updatedAt: now,
   447→    };
   448→    // Keep at most 50 entries
   449→    // ── Secret XP: award 10 XP for watching an episode (fire-and-forget) ──
   450→    trackXPForUser(get().user, 10, "episode_watched");
   451→    return { history: [newEntry, ...filtered].slice(0, 50) };
   452→  }),
   453→  updateHistoryProgress: (animeId, episodeNum, progress, duration) => set((state) => ({
   454→    history: state.history.map((h) =>
   455→      h.animeId === animeId && h.episodeNum === episodeNum
   456→        ? { ...h, progress, duration, updatedAt: new Date().toISOString() }
   457→        : h
   458→    ),
   459→  })),
   460→  isBookmarked: (animeId) => get().bookmarks.some((b) => b.animeId === animeId),
   461→  // ── Generic library / progress / activity ──
   462→  library: [],
   463→  mediaProgress: [],
   464→  activity: [],
   465→  addToLibrary: (entry) => set((state) => {
   466→    const key = `${entry.kind}:${entry.mediaId}`;
   467→    if (state.library.some((e) => e.key === key)) return {};
   468→    const newEntry: LibraryEntry = { ...entry, key, addedAt: entry.addedAt ?? Date.now() };
   469→    return { library: [newEntry, ...state.library].slice(0, 500) };
   470→  }),
   471→  removeFromLibrary: (kind, mediaId) => set((state) => ({
   472→    library: state.library.filter((e) => e.key !== `${kind}:${mediaId}`),
   473→  })),
   474→  isInLibrary: (kind, mediaId) => get().library.some((e) => e.key === `${kind}:${mediaId}`),
   475→  recordMediaProgress: (entry, xp = 0) => set((state) => {
   476→    const key = `${entry.kind}:${entry.mediaId}`;
   477→    const now = entry.updatedAt ?? Date.now();
   478→    const existing = state.mediaProgress.find((p) => p.key === key);
   479→    // Award XP at most once per title per calendar day (keeps XP/heatmap honest
   480→    // even though progress ticks fire often).
   481→    const sameDay = (a: number, b: number) => {
   482→      const da = new Date(a), db = new Date(b);
   483→      return da.getFullYear() === db.getFullYear() && da.getMonth() === db.getMonth() && da.getDate() === db.getDate();
   484→    };
   485→    const grant = xp > 0 && !(existing && sameDay(existing.updatedAt, now));
   486→    const filtered = state.mediaProgress.filter((p) => p.key !== key);
   487→    const newEntry: MediaProgressEntry = { ...entry, key, updatedAt: now };
   488→    const activity = grant
   489→      ? [{ ts: now, kind: entry.kind, xp }, ...state.activity].slice(0, 3000)
   490→      : state.activity;
   491→    return {
   492→      mediaProgress: [newEntry, ...filtered].slice(0, 200),
   493→      activity,
   494→    };
   495→  }),
   496→  // ── Auth state ──
   497→  user: null,
   498→  setUser: (user) => {
   499→    set({ user });
   500→    // Close the auth modal automatically on successful sign-in / sign-up
   501→    if (user) set({ authModal: null });
   502→  },
   503→  logout: () => set({ user: null }),
   504→  // ── Auth modal actions ──
   505→  authModal: null,
   506→  openAuthModal: (mode, message) => set({ authModal: { mode, message } }),
   507→  closeAuthModal: () => set({ authModal: null }),
   508→  // ── AniList account link ──
   509→  anilistToken: null,

   582→export function parsePath(pathname: string): { route: Route; subPage: SectionSubPage } {
   583→  // Remove leading slash
   584→  const p = pathname.startsWith("/") ? pathname.slice(1) : pathname;
   585→  // Root path → anime home
   586→  if (!p) return { route: { page: "home" }, subPage: "home" };
   587→  const parts = p.split("/");
   588→
   589→  // ── First-class sub-page paths (SEO-friendly URLs) ──
   590→  if (parts[0] === "browse") return { route: { page: "home" }, subPage: "browse" };
   591→  if (parts[0] === "schedule") return { route: { page: "home" }, subPage: "schedule" };
   592→  if (parts[0] === "genres") return { route: { page: "home" }, subPage: "genres" };
   593→  if (parts[0] === "sub") return { route: { page: "home" }, subPage: "sub" };
   594→  if (parts[0] === "dub") return { route: { page: "home" }, subPage: "dub" };
   595→  if (parts[0] === "trending") return { route: { page: "home" }, subPage: "trending" };
   596→  if (parts[0] === "top-rated") return { route: { page: "home" }, subPage: "top-rated" };
   597→
   598→  if (parts[0] === "hub") return { route: { page: "hub" }, subPage: "home" };
   599→  if (parts[0] === "home") return { route: { page: "home" }, subPage: "home" };
   600→  if (parts[0] === "discover") return { route: { page: "discover" }, subPage: "home" };
   601→  if (parts[0] === "search") return { route: { page: "search", query: decodeURIComponent(parts[1] || "") }, subPage: "home" };
   602→  if (parts[0] === "anime" && parts[1]) return { route: { page: "anime", id: parts[1] }, subPage: "home" };
   603→  if (parts[0] === "watch" && parts[1] && parts[2])
   604→    return { route: { page: "watch", id: parts[1], episode: parseInt(parts[2], 10) || 1 }, subPage: "home" };
   605→  if (parts[0] === "bookmarks") return { route: { page: "bookmarks" }, subPage: "home" };
   606→  if (parts[0] === "watchlist") return { route: { page: "watchlist" }, subPage: "home" };
   607→  if (parts[0] === "history") return { route: { page: "history" }, subPage: "home" };
   608→  // Retired sections — redirect to anime home
   609→  if (parts[0] === "movies") return { route: { page: "home" }, subPage: "home" };
   610→  if (parts[0] === "tv") return { route: { page: "home" }, subPage: "home" };
   611→  if (parts[0] === "live") return { route: { page: "home" }, subPage: "home" };
   612→  if (parts[0] === "watchnow") return { route: { page: "home" }, subPage: "home" };
   613→  if (parts[0] === "genre") return { route: { page: "home" }, subPage: "genres" };
   614→  if (parts[0] === "movie" && parts[1]) return { route: { page: "home" }, subPage: "home" };
   615→  if (parts[0] === "tvshow" && parts[1]) return { route: { page: "home" }, subPage: "home" };
   616→  if (parts[0] === "watch-movie" && parts[1]) return { route: { page: "home" }, subPage: "home" };
   617→  if (parts[0] === "watch-tv") return { route: { page: "home" }, subPage: "home" };
   618→  if (parts[0] === "live-watch") return { route: { page: "home" }, subPage: "home" };
   619→  if (parts[0] === "live-tv-watch") return { route: { page: "home" }, subPage: "home" };
   620→  if (parts[0] === "manga" && parts[1]) return { route: { page: "manga-detail", id: parts[1] }, subPage: "home" };
   621→  if (parts[0] === "manga") return { route: { page: "manga" }, subPage: "home" };
   622→  if (parts[0] === "read-manga" && parts[1] && parts[2])
   623→    return { route: { page: "manga-read", id: parts[1], chapterId: parts[2] }, subPage: "home" };
   624→  if (parts[0] === "contact") return { route: { page: "contact" }, subPage: "home" };
   625→  if (parts[0] === "donate" && parts[1] === "crypto") return { route: { page: "donate-crypto" }, subPage: "home" };
   626→  if (parts[0] === "donate") return { route: { page: "donate" }, subPage: "home" };
   627→  if (parts[0] === "updates") return { route: { page: "updates" }, subPage: "home" };
   628→  if (parts[0] === "guide") return { route: { page: "guide" }, subPage: "home" };
   629→  if (parts[0] === "features") return { route: { page: "features" }, subPage: "home" };
   630→  if (parts[0] === "novel" && parts[1]) return { route: { page: "novel-detail", novelId: decodeURIComponent(parts[1]), novelTitle: "", novelCover: "", novelAuthor: "", novelSource: "readlightnovel" }, subPage: "home" };
   631→  if (parts[0] === "novel") return { route: { page: "novel" }, subPage: "home" };
   632→  if (parts[0] === "read-novel" && parts[1] && parts[2]) return { route: { page: "novel-read", novelId: decodeURIComponent(parts[1]), novelTitle: "", chapterId: `chapter-${parts[2]}`, chapterNum: parseInt(parts[2]), chapterTitle: "", totalChapters: 0, novelSource: "readlightnovel" }, subPage: "home" };
   633→  if (parts[0] === "signin") return { route: { page: "signin" }, subPage: "home" };
   634→  if (parts[0] === "signup") return { route: { page: "signup" }, subPage: "home" };
   635→  if (parts[0] === "profile") return { route: { page: "profile" }, subPage: "home" };
   636→  if (parts[0] === "leaderboard") return { route: { page: "leaderboard" }, subPage: "home" };
   637→  if (parts[0] === "mod") return { route: { page: "mod" }, subPage: "home" };
   638→  if (parts[0] === "settings") return { route: { page: "settings" }, subPage: "home" };
   639→  if (parts[0] === "scraper" && parts[1] === "anime" && parts[2])
   640→    return { route: { page: "scraper-anime", id: parts[2] }, subPage: "home" };
   641→  if (parts[0] === "scraper" && parts[1] === "watch" && parts[2] && parts[3] && parts[4])
   642→    return { route: { page: "scraper-watch", site: parts[2], id: parts[3], episode: decodeURIComponent(parts[4]) }, subPage: "home" };
   643→  if (parts[0] === "scraper") return { route: { page: "scraper" }, subPage: "home" };
   644→  if (parts[0] === "music") return { route: { page: "music" }, subPage: "home" };
   645→  if (parts[0] === "torrent") return { route: { page: "torrent" }, subPage: "home" };
   646→  if (parts[0] === "download") return { route: { page: "download" }, subPage: "home" };
   647→  // Fallback
   648→  return { route: { page: "home" }, subPage: "home" };
   649→}
