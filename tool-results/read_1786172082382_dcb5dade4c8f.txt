    23→// ─── Tiny dropdown (click to open, backdrop to close) ───────────
    24→function MenuSelect({ label, value, options, onChange, disabledIds = [] }: {
    25→  label?: string;
    26→  value: string;
    27→  options: Array<{ id: string; label: string }>;
    28→  onChange: (id: string) => void;
    29→  disabledIds?: string[];
    30→}) {
    31→  const [open, setOpen] = useState(false);
    32→  const current = options.find(o => o.id === value);
    33→
    34→  return (
    35→    <div className="relative">
    36→      <button
    37→        onClick={() => setOpen(!open)}
    38→        className="flex items-center gap-1.5 h-8 px-3 rounded-lg bg-white/[0.06] hover:bg-white/[0.1] border border-white/[0.08] text-xs font-bold text-white/85 transition-all w-full sm:w-auto"
    39→      >
    40→        {label && <span className="text-[9px] font-bold text-white/35 uppercase tracking-wider">{label}</span>}
    41→        <span className="max-w-[110px] truncate flex-1 sm:flex-none text-left">{current?.label || value || "—"}</span>
    42→        <svg className={`w-3 h-3 text-white/40 transition-transform shrink-0 ${open ? "rotate-180" : ""}`} viewBox="0 0 24 24" fill="currentColor"><path d="M7 10l5 5 5-5z" /></svg>
    43→      </button>
    44→      {open && (
    45→        <>
    46→          {/* Backdrop — fixed to cover entire viewport */}
    47→          <div className="fixed inset-0 z-30" onClick={() => setOpen(false)} />
    48→          {/* Dropdown — absolute positioned below the button.
    49→              Uses absolute (not fixed) to avoid CSS transform bugs where
    50→              fixed-positioned elements detach from the viewport when a
    51→              parent has transform/filter/will-change. */}
    52→          <div
    53→            className="absolute top-full left-0 sm:left-auto sm:right-0 mt-1 bg-[#0a0a0a] border border-white/15 rounded-lg overflow-hidden py-1 shadow-2xl z-40 max-h-[280px] overflow-y-auto min-w-[140px] w-max sm:w-auto"
    54→            style={{ maxWidth: "min(90vw, 320px)" }}
    55→          >
    56→            {options.map(o => {
    57→              const disabled = disabledIds.includes(o.id);
    58→              return (
    59→                <button
    60→                  key={o.id}
    61→                  disabled={disabled}
    62→                  onClick={() => { if (!disabled) { onChange(o.id); setOpen(false); } }}
    63→                  className={`block w-full text-left px-3 py-2.5 sm:py-2 text-xs sm:text-xs transition-colors whitespace-nowrap ${o.id === value ? "font-bold" : "text-white/60 hover:bg-white/10 hover:text-white"} ${disabled ? "opacity-30 cursor-not-allowed" : ""}`}
    64→                  style={o.id === value ? { color: ACCENT } : undefined}
    65→                >
    66→                  {o.label}
    67→                </button>
    68→              );
    69→            })}
    70→          </div>
    71→        </>
    72→      )}
    73→    </div>
    74→  );
    75→}

    77→// ─── Checkbox toggle (purple squares like Miruro) ────────────────
    78→function ToggleCheck({ label, state, onToggle }: { label: string; state: boolean; onToggle: () => void }) {
    79→  return (
    80→    <button onClick={onToggle} className="flex items-center gap-1.5 group shrink-0">
    81→      <div
    82→        className={`w-4 h-4 rounded border-2 flex items-center justify-center transition-all ${state ? "" : "bg-transparent border-white/20 group-hover:border-white/40"}`}
    83→        style={state ? { background: ACCENT, borderColor: ACCENT } : undefined}
    84→      >
    85→        {state && <svg className="w-2.5 h-2.5 text-black" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}><path strokeLinecap="round" strokeLinejoin="round" d="M5 13l4 4L19 7" /></svg>}
    86→      </div>
    87→      <span className={`text-xs font-medium transition-colors ${state ? "" : "text-white/40 group-hover:text-white/70"}`} style={state ? { color: ACCENT } : undefined}>{label}</span>
    88→    </button>
    89→  );
    90→}

    97→  return (
    98→    <button
    99→      onClick={() => navigate({ page: "anime", id: String(item.id) })}
   100→      className="relative flex w-full items-stretch gap-3 rounded-xl overflow-hidden border border-white/[0.07] bg-white/[0.03] hover:bg-white/[0.06] hover:border-white/20 transition-all text-left group"
   101→    >
   102→      {/* Backdrop art (faint, right side) */}
   103→      {img && (
   104→        <div
   105→          className="absolute inset-0 opacity-[0.1] group-hover:opacity-[0.16] transition-opacity pointer-events-none"
   106→          style={{ backgroundImage: `url(${img})`, backgroundSize: "cover", backgroundPosition: "center 20%" }}
   107→        />
   108→      )}
   109→      <div className="relative shrink-0 w-[64px] self-stretch min-h-[92px] overflow-hidden bg-white/5">
   110→        {img
   111→          ? <img src={img} alt={title} className="absolute inset-0 w-full h-full object-cover" loading="lazy" />
   112→          : <div className="absolute inset-0 flex items-center justify-center text-white/15 font-bold text-lg">{title.charAt(0)}</div>}
   113→      </div>
   114→      <div className="relative flex flex-col justify-center gap-1.5 py-2.5 pr-3 min-w-0">
   115→        <div className="flex items-start gap-1.5">
   116→          <span className="mt-[5px] w-1.5 h-1.5 rounded-full shrink-0" style={{ background: "#60A5FA" }} />
   117→          <p className="text-[13px] font-semibold text-white leading-snug line-clamp-2">{title}</p>
   118→        </div>
   119→        <div className="flex items-center gap-2 text-[10px] font-bold text-white/45 pl-3">
   120→          {format && <span className="uppercase">{format}</span>}
   121→          {item.episodes ? (
   122→            <span className="flex items-center gap-0.5 bg-white/[0.07] rounded px-1 py-0.5">
   123→              <svg className="w-2.5 h-2.5" viewBox="0 0 24 24" fill="currentColor"><path d="M19 4H5a2 2 0 0 0-2 2v12a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2V6a2 2 0 0 0-2-2zm-8 9H6v-2h5v2zm7-4H6V7h12v2z"/></svg>
   124→              {item.episodes}
   125→            </span>
   126→          ) : null}
   127→          {item.averageScore ? (
   128→            <span className="flex items-center gap-0.5">
   129→              <svg className="w-2.5 h-2.5 text-white/45" fill="currentColor" viewBox="0 0 24 24"><path d="M12 17.27L18.18 21l-1.64-7.03L22 9.24l-7.19-.61L12 2 9.19 8.63 2 9.24l5.46 4.73L5.82 21z"/></svg>
   130→              {item.averageScore}
   131→            </span>
   132→          ) : null}
   133→          {subtitle && <span className="uppercase text-white/30">{subtitle}</span>}
   134→        </div>
   135→      </div>
   136→    </button>
   137→  );
   138→}

   195→      // Hardsub tab: true hardsub servers + 4animo (user wants 4animo here too)
   196→      return s.type === "sub" && (s.hardsub === true || s.source === "animo4");

   744→// WatchPageDownloadModal — Black theme, fetches REAL download links
   745→// from AnimeX API (Google Drive, Mega, etc.) — NO proxy URLs
   746→// Matches the HLS player download modal style exactly
   747→// ============================================================
   748→
   749→function WatchPageDownloadModal({
   750→  animeTitle, animeId, episodeNum, downloadUrl, streamData, serverList, onClose
   751→}: {
   752→  animeTitle: string; animeId: number; episodeNum: number;
   753→  downloadUrl: string | null; streamData: any; serverList: any[]; onClose: () => void;
   754→}) {
   755→  const [loading, setLoading] = useState(true);
   756→  const [links, setLinks] = useState<Array<{ text: string; decodedUrl: string }>>([]);
   757→  const [error, setError] = useState('');
   758→
   759→  useEffect(() => {
   760→    const fetchDownloads = async () => {
   761→      try {
   762→        setLoading(true);
   763→        setError('');
   764→        let title = animeTitle || '';
   765→        // If no title, try AniList
   766→        if (!title && animeId) {

   772→                query: `query($id:Int){Media(id:$id,type:ANIME){title{english romaji}}}`,

   777→              const titleData = await titleRes.json();
   778→              title = titleData?.data?.Media?.title?.english || titleData?.data?.Media?.title?.romaji || '';
   779→            }
   780→          } catch { /* ignore */ }

   787→        const res = await fetch(`/api/anime/download?title=${encodeURIComponent(title)}&auto=1`);
   788→        if (!res.ok) {
   789→          setError('Failed to fetch download links');
   790→          setLoading(false);
   791→          return;
   792→        }
   793→        const data = await res.json();
   794→        if (data.links && data.links.length > 0) {
   795→          setLinks(data.links);
   796→        } else {
   797→          setError('No download links found');
   798→        }
   799→      } catch (e: any) {
   800→        setError(e?.message || 'Failed to load downloads');
   801→      } finally {
   802→        setLoading(false);
   803→      }
   804→    };
   805→    fetchDownloads();
   806→  }, [animeId, animeTitle]);

   849→          {!loading && !error && quickLinks.length > 0 && (

   887→            </div>

   890→      </div>
   891→    </div>
   892→  );
   893→}
