   156→// ─────────────────────────────────────────────────────────────────────────────
   157→// In-memory cache
   158→// ─────────────────────────────────────────────────────────────────────────────
   159→
   160→const cache = new Map<number, { data: HindiAnimeEntry; timestamp: number }>();
   161→const CACHE_TTL_MS = 15 * 60 * 1000; // 15 minutes
   162→
   163→function getCached(id: number): HindiAnimeEntry | null {
   164→  const entry = cache.get(id);
   165→  if (!entry) return null;
   166→  if (Date.now() - entry.timestamp > CACHE_TTL_MS) {
   167→    cache.delete(id);
   168→    return null;
   169→  }
   170→  return entry.data;
   171→}
   172→
   173→function setCache(id: number, data: HindiAnimeEntry): void {
   174→  cache.set(id, { data, timestamp: Date.now() });
   175→}

   179→// ─────────────────────────────────────────────────────────────────────────────
   180→
   181→/**
   182→ * Fetch a single anime entry by its AniList ID.
   183→ * Returns null if the anime is not found or the request fails.
   184→ */
   185→export async function fetchHindiAnime(
   186→  anilistId: number
   187→): Promise<HindiAnimeEntry | null> {
   188→  // Check cache first
   189→  const cached = getCached(anilistId);
   190→  if (cached) return cached;
   191→
   192→  try {
   193→    const res = await fetch(`${API_BASE}/${anilistId}`, {
   194→      next: { revalidate: 900 }, // 15 min cache at Next.js level
   195→    });
   196→
   197→    if (!res.ok) {
   198→      if (res.status === 404) return null;
   199→      console.error(
   200→        `[HindiAnimeDB] Failed to fetch ID ${anilistId}: HTTP ${res.status}`
   201→      );
   202→      return null;
   203→    }
   204→
   205→    const data: HindiAnimeEntry = await res.json();
   206→
   207→    // The API might return an error object instead of anime data
   208→    if (!data || !data.title) return null;
   209→
   210→    setCache(anilistId, data);
   211→    return data;
   212→  } catch (err) {
   213→    console.error(
   214→      `[HindiAnimeDB] Error fetching ID ${anilistId}:`,
   215→      err instanceof Error ? err.message : err
   216→    );
   217→    return null;
   218→  }
   219→}

   225→/**
   226→ * Fetch multiple anime entries by their AniList IDs.
   227→ * Uses concurrent requests with a concurrency limit to avoid overwhelming
   228→ * the API. Failed/missing entries are silently omitted from the result.
   229→ */
   230→export async function fetchHindiAnimeBatch(
   231→  ids: number[],
   232→  concurrency: number = 5
   233→): Promise<HindiAnimeEntry[]> {
   234→  const results: HindiAnimeEntry[] = [];
   235→  const queue = [...ids];
   236→
   237→  async function worker(): Promise<void> {
   238→    while (queue.length > 0) {
   239→      const id = queue.shift();
   240→      if (id === undefined) break;
   241→      const entry = await fetchHindiAnime(id);
   242→      if (entry) results.push(entry);
   243→    }
   244→  }
   245→
   246→  const workers = Array.from(
   247→    { length: Math.min(concurrency, queue.length) },
   248→    () => worker()
   249→  );
   250→  await Promise.all(workers);
   251→
   252→  return results;
   253→}

   259→/**
   260→ * Search the pre-populated index for anime matching the query.
   261→ * Matching is case-insensitive and checks title and synonym fields.
   262→ * Returns matching HindiAnimeListItem[] sorted by rating (descending).
   263→ */
   264→export function searchHindiAnime(query: string): HindiAnimeListItem[] {
   265→  if (!query || query.trim().length === 0) return [];
   266→
   267→  const normalized = query.toLowerCase().trim();
   268→  const terms = normalized.split(/\s+/);
   269→
   270→  const scored = PRE_POPULATED_INDEX.map((item) => {
   271→    const titleLower = item.title.toLowerCase();
   272→    let score = 0;
   273→
   274→    // Exact match
   275→    if (titleLower === normalized) score += 100;
   276→    // Starts with
   277→    if (titleLower.startsWith(normalized)) score += 50;
   278→    // Contains full query
   279→    if (titleLower.includes(normalized)) score += 30;
   280→    // Each term matches
   281→    for (const term of terms) {
   282→      if (titleLower.includes(term)) score += 10;
   283→    }
   284→    // Boost by rating
   285→    score += parseFloat(item.rating) || 0;
   286→
   287→    return { item, score };
   288→  }).filter(({ score }) => score > 0);
   289→
   290→  scored.sort((a, b) => b.score - a.score);
   291→  return scored.map(({ item }) => item);
   292→}

   340→export async function getPopularHindiAnime(

   358→      type: entry.type,
   359→      rating: entry.rating,
   360→      episodesCount: entry.episodesCount,
   361→    };

   365→  const fetchedIds = new Set(results.map((r) => r.id));
   366→  for (const item of sortedIndex.slice(0, limit)) {

   368→      results.push(item);

   378→// ─────────────────────────────────────────────────────────────────────────────
   379→// Index discovery — fetch all known anime from the DB
   380→// ─────────────────────────────────────────────────────────────────────────────
   381→
   382→/**
   383→ * Discover anime entries from the Hindi Anime DB by probing a range of IDs.
   384→ * The DB has 838 entries but we don't know all IDs. This function probes
   385→ * a supplied range (or uses sensible defaults) and returns found entries.
   386→ *
   387→ * WARNING: This is expensive — it makes one request per ID in the range.
   388→ * Use sparingly and consider caching the result.
   389→ *
   390→ * @param startId - First AniList ID to probe (default: 1)
   391→ * @param endId - Last AniList ID to probe (default: 160000)
   392→ * @param concurrency - Number of concurrent requests (default: 8)
   393→ * @param onProgress - Optional callback invoked with (found, tried) counts
   394→ */
   395→export async function discoverHindiAnime(
   396→  startId: number = 1,
   397→  endId: number = 160000,
   398→  concurrency: number = 8,
   399→  onProgress?: (found: number, tried: number) => void
   400→): Promise<HindiAnimeListItem[]> {
   401→  const results: HindiAnimeListItem[] = [];
   402→  const ids = Array.from(
   403→    { length: endId - startId + 1 },
   404→    (_, i) => startId + i
   405→  );
   406→  let tried = 0;
   407→
   408→  // For large ranges, use a chunked approach to avoid creating massive arrays
   409→  const CHUNK_SIZE = 500;
   410→  const chunks: number[][] = [];
   411→  for (let i = 0; i < ids.length; i += CHUNK_SIZE) {
   412→    chunks.push(ids.slice(i, i + CHUNK_SIZE));
   413→  }
   414→
   415→  async function probeChunk(chunk: number[]): Promise<void> {
   416→    const probeResults = await Promise.allSettled(
   417→      chunk.map(async (id) => {
   418→        const entry = await fetchHindiAnime(id);
   419→        tried++;
   420→        if (onProgress) onProgress(results.length, tried);
   421→        return entry;
   422→      })
   423→    );
   424→
   425→    for (const result of probeResults) {
   426→      if (result.status === "fulfilled" && result.value) {
   427→        const entry = result.value;
   428→        // Derive the AniList ID from the hostUrl or streamingId
   429→        const anilistId = parseInt(entry.streamingId, 10) || 0;
   430→        if (anilistId > 0) {
   431→          results.push({
   432→            id: anilistId,
   433→            title: entry.title,
   434→            thumbnail: entry.thumbnail,
   435→            type: entry.type,
   436→            rating: entry.rating,
   437→            episodesCount: entry.episodesCount,
   438→          });
   439→        }
   440→      }
   441→    }
   442→  }
   443→
   444→  // Process chunks with limited concurrency
   445→  const workerQueue = [...chunks];
   446→  async function chunkWorker(): Promise<void> {
   447→    while (workerQueue.length > 0) {
   448→      const chunk = workerQueue.shift();
   449→      if (!chunk) break;
   450→      await probeChunk(chunk);
   451→    }
   452→  }
   453→
   454→  await Promise.all(
   455→    Array.from({ length: Math.min(concurrency, workerQueue.length) }, () =>
   456→      chunkWorker()
   457→    )
   458→  );
   459→
   460→  return results;
   461→}

   463→/**
   464→ * Get a list of all known anime from the pre-populated index,
   465→ * optionally augmented with live data from the API.
   466→ *
   467→ * This is the recommended way to browse the catalog — it avoids
   468→ * the expensive probe approach of discoverHindiAnime().
   469→ */
   470→export async function getAllKnownHindiAnime(): Promise<HindiAnimeListItem[]> {
   471→  // Return pre-populated data immediately; caller can enrich with fetchHindiAnimeBatch
   472→  return [...PRE_POPULATED_INDEX].sort(
   473→    (a, b) => parseFloat(b.rating) - parseFloat(a.rating)
   474→  );
   475→}
   476→
   477→// Set of all known AniList IDs that have Hindi dubs (from pre-populated index)
   478→const _knownHindiIds = new Set(PRE_POPULATED_INDEX.map(a => a.id));
   479→
   480→/**
   481→ * Check if an anime has a Hindi dub available.
   482→ *
   483→ * Checks two sources:
   484→ *   1. PRE_POPULATED_INDEX (instant — 79 known IDs)
   485→ *   2. Full API (hindianimedb.simoonabdulla.workers.dev) for IDs not in the index
   486→ *
   487→ * Results are cached in-memory for 15 minutes.
   488→ */
   489→const _hindiCheckCache = new Map<number, boolean>();
   490→export async function isHindiAnime(anilistId: number): Promise<boolean> {
   491→  // Check pre-populated index first (instant)
   492→  if (_knownHindiIds.has(anilistId)) return true;
   493→
   494→  // Check in-memory cache
   495→  if (_hindiCheckCache.has(anilistId)) return _hindiCheckCache.get(anilistId)!;
   496→
   497→  // Check the full API
   498→  try {
   499→    const entry = await fetchHindiAnime(anilistId);
   500→    const hasHindi = !!(entry && entry.episodes && entry.episodes.length > 0);
   501→    _hindiCheckCache.set(anilistId, hasHindi);
   502→    return hasHindi;
   503→  } catch {
   504→    _hindiCheckCache.set(anilistId, false);
   505→    return false;
   506→  }
   507→}

   509→/**
   510→ * Enrich the pre-populated index with live data from the API.
   511→ * Fetches full entries for all IDs in the index and returns updated items.
   512→ * Useful for getting the latest episode counts, ratings, and thumbnails.
   513→ */
   514→export async function enrichHindiAnimeIndex(): Promise<HindiAnimeListItem[]> {
   515→  const ids = PRE_POPULATED_INDEX.map((item) => item.id);
   516→  const entries = await fetchHindiAnimeBatch(ids, 5);
   517→
   518→  const enriched = entries.map((entry, i) => ({
   519→    id: ids[i],
   520→    title: entry.title,
   521→    thumbnail: entry.thumbnail,
   522→    type: entry.type,
   523→    rating: entry.rating,
   524→    episodesCount: entry.episodesCount,
   525→  }));
   526→
   527→  // Merge: prefer enriched data, fall back to pre-populated
   528→  const enrichedMap = new Map(enriched.map((e) => [e.id, e]));
   529→  const merged = PRE_POPULATED_INDEX.map(
   530→    (item) => enrichedMap.get(item.id) ?? item
   531→  );
   532→
   533→  return merged.sort(
   534→    (a, b) => parseFloat(b.rating) - parseFloat(a.rating)
   535→  );
   536→}

   542→/**
   543→ * Get all embeddable stream URLs for a specific episode of an anime.
   544→ * Filters the episode's files and returns URLs that can be used in iframes.
   545→ */
   546→export function getEpisodeStreamUrls(
   547→  entry: HindiAnimeEntry,
   548→  episodeNumber: number | string,
   549→  type?: "sub" | "dub"
   550→): HindiAnimeFile[] {
   551→  const ep = entry.episodes.find(
   552→    (e) => e.episode === String(episodeNumber)
   553→  );
   554→  if (!ep) return [];
   555→  if (type) return ep.files.filter((f) => f.type === type);
   556→  return ep.files;
   557→}

   559→/**
   560→ * Get the preferred stream URL for an episode.
   561→ * Prefers Hindi dub over sub, and the first available server.
   562→ */
   563→export function getPreferredStreamUrl(
   564→  entry: HindiAnimeEntry,
   565→  episodeNumber: number | string
   566→): string | null {
   567→  const files = getEpisodeStreamUrls(entry, episodeNumber);
   568→  if (files.length === 0) return null;
   569→
   570→  // Prefer dub
   571→  const dub = files.find((f) => f.type === "dub");
   572→  if (dub) return dub.url;
   573→
   574→  // Fall back to sub
   575→  return files[0]?.url ?? null;
   576→}
