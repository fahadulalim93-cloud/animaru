    18→declare global {
    19→  interface Window { dashjs: any; }

    21→
    22→function loadDashJs(): Promise<void> {
    23→  return new Promise((resolve, reject) => {
    24→    if (window.dashjs) return resolve();
    25→    const existing = document.querySelector('script[src*="dash.mediaplayer"]') as HTMLScriptElement | null;
    26→    if (existing) {
    27→      existing.addEventListener("load", () => resolve());
    28→      existing.addEventListener("error", () => reject(new Error("dash.js load error")));
    29→      return;
    30→    }
    31→    const script = document.createElement("script");
    32→    script.src = "https://cdn.jsdelivr.net/npm/dashjs@4.7.4/dist/dash.mediaplayer.min.js";
    33→    script.async = true;
    34→    script.onload = () => resolve();
    35→    script.onerror = () => reject(new Error("Failed to load dash.js"));
    36→    document.head.appendChild(script);
    37→  });
    38→}

    40→function DashPlayer({
    41→  url,
    42→  subtitleTracks,
    43→  onEnded,
    44→  autoplay = true,
    45→}: {
    46→  url: string;
    47→  subtitleTracks?: Array<{ url: string; label: string; lang?: string }>;
    48→  onEnded?: () => void;
    49→  autoplay?: boolean;
    50→}) {
    51→  const videoRef = useRef<HTMLVideoElement>(null);
    52→  const playerRef = useRef<any>(null);
    53→  const [error, setError] = useState<string | null>(null);
    54→  const [loading, setLoading] = useState(true);
    55→
    56→  useEffect(() => {
    57→    let cancelled = false;
    58→
    59→    async function init() {
    60→      try {
    61→        await loadDashJs();
    62→        if (cancelled || !videoRef.current) return;
    63→
    64→        const video = videoRef.current;
    65→        const player = window.dashjs.MediaPlayer().create();
    66→        player.initialize(video, url, autoplay);
    67→        player.updateSettings({
    68→          streaming: {
    69→            buffer: {
    70→              fastSwitchEnabled: true,
    71→              bufferTimeAtTopQuality: 30,
    72→              bufferTimeAtTopQualityLongForm: 60,
    73→            },
    74→          },
    75→        });
    76→        playerRef.current = player;
    77→        setLoading(false);
    78→      } catch (e: any) {
    79→        if (!cancelled) {
    80→          setError(e?.message || "Failed to load DASH player");
    81→          setLoading(false);
    82→        }
    83→      }
    84→    }
    85→
    86→    init();
    87→
    88→    return () => {
    89→      cancelled = true;
    90→      if (playerRef.current) {
    91→        try { playerRef.current.reset(); } catch {}
    92→        playerRef.current = null;
    93→      }
    94→    };
    95→  }, [url, autoplay]);
    96→
    97→  return (
    98→    <div className="absolute inset-0 w-full h-full bg-black">
    99→      {loading && (
   100→        <div className="absolute inset-0 flex items-center justify-center">
   101→          <div className="ltv-spinner ltv-spinner-lg" />
   102→        </div>
   103→      )}
   104→      {error && (
   105→        <div className="absolute inset-0 flex items-center justify-center">
   106→          <div className="text-center space-y-2">
   107→            <p className="text-rose-400 text-sm">{error}</p>
   108→            <button onClick={() => window.location.reload()} className="text-xs text-white/60 hover:text-white">
   109→              Reload
   110→            </button>
   111→          </div>
   112→        </div>
   113→      )}
   114→      <video
   115→        ref={videoRef}
   116→        className="w-full h-full"
   117→        controls
   118→        onEnded={onEnded}
   119→        crossOrigin="anonymous"
   120→      />
   121→      {subtitleTracks && subtitleTracks.length > 0 && (
   122→        <div className="absolute bottom-14 right-4 z-10 flex gap-1">
   123→          {subtitleTracks.slice(0, 5).map((s) => (
   124→            <button
   125→              key={s.lang || s.label}
   126→              onClick={() => {
   127→                const video = videoRef.current;
   128→                if (!video) return;
   129→                // Remove existing <track> elements from the DOM (not TextTrackList entries)
   130→                const existingTracks = video.querySelectorAll("track");
   131→                existingTracks.forEach(t => t.remove());
   132→                const track = document.createElement("track");
   133→                track.kind = "subtitles";
   134→                track.label = s.label;
   135→                track.srclang = s.lang || "en";
   136→                track.src = s.url;
   137→                track.default = true;
   138→                video.appendChild(track);
   139→              }}
   140→              className="px-2 py-1 text-[10px] font-bold bg-black/60 text-white/70 hover:bg-white/20 hover:text-white rounded transition-colors"
   141→            >
   142→              {s.lang || s.label}
   143→            </button>
   144→          ))}
   145→        </div>
   146→      )}
   147→    </div>
   148→  );
   149→}

   225→// ── Embed Player with auto-fallback on 410/dead links ──
   226→// If the embed source returns a dead page (410 Gone, etc.), shows a "switch server" overlay
   227→function EmbedPlayerWithFallback({
   228→  src,
   229→  animeTitle,
   230→  episodeNum,
   231→  provider,
   232→  providersForCurrentEp,
   233→  failedProviders,
   234→  onProviderFailed,
   235→  onProviderSelect,
   236→  getProviderDisplayName,
   237→}: {
   238→  src: string;
   239→  animeTitle: string;
   240→  episodeNum: number;
   241→  provider: string;
   242→  providersForCurrentEp: string[];
   243→  failedProviders: Set<string>;
   244→  onProviderFailed: (p: string) => void;
   245→  onProviderSelect: (p: string) => void;
   246→  getProviderDisplayName: (p: string) => string;
   247→}) {
   248→  const iframeRef = useRef<HTMLIFrameElement>(null);
   249→  const [embedFailed, setEmbedFailed] = useState(false);
   250→  const [showServerOverlay, setShowServerOverlay] = useState(false);
   251→
   252→  // Detect iframe load failure — embed sources often return 410 Gone
   253→  useEffect(() => {
   254→    const timer = setTimeout(() => {
   255→      // After 8 seconds, if iframe loaded but content might be dead,
   256→      // show the switch server button as a floating hint
   257→      setShowServerOverlay(true);
   258→    }, 8000);
   259→    return () => clearTimeout(timer);
   260→  }, [src]);
   261→
   262→  const handleIframeLoad = useCallback(() => {
   263→    // We can't read iframe content due to cross-origin, but we can
   264→    // check if the URL might be a dead embed by doing a HEAD request
   265→    try {
   266→      fetch(src, { method: "HEAD", mode: "no-cors" }).catch(() => {});
   267→    } catch {}
   268→  }, [src]);
   269→
   270→  const handleEmbedError = useCallback(() => {
   271→    setEmbedFailed(true);
   272→    onProviderFailed(provider);
   273→  }, [provider, onProviderFailed]);
   274→
   275→  const otherProviders = providersForCurrentEp.filter(
   276→    p => p !== provider && !failedProviders.has(p)
   277→  );
   278→
   279→  if (embedFailed) {
   280→    return (
   281→      <div className="absolute inset-0 flex items-center justify-center bg-black z-20">
   282→        <div className="text-center space-y-4 max-w-sm px-6">
   283→          <div className="w-12 h-12 rounded-xl bg-[#ffffff]/10 border border-[#ffffff]/20 flex items-center justify-center mx-auto">
   284→            <svg className="w-6 h-6 text-[#ffffff]/70" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
   285→              <path d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z" />
   286→            </svg>
   287→          </div>
   288→          <p className="text-white/75 text-sm">This embed source is unavailable (410 Gone)</p>
   289→          {otherProviders.length > 0 ? (
   290→            <div className="flex flex-wrap gap-2 justify-center">
   291→              {otherProviders.map(p => (
   292→                <button
   293→                  key={p}
   294→                  onClick={() => onProviderSelect(p)}
   295→                  className="px-3 py-1.5 rounded-lg bg-[#ffffff] text-black text-xs font-bold hover:bg-white/90 transition-colors"
   296→                >
   297→                  Try {getProviderDisplayName(p)}
   298→                </button>
   299→              ))}
   300→            </div>
   301→          ) : (
   302→            <button
   303→              onClick={() => window.location.reload()}
   304→              className="px-5 py-2 rounded-lg bg-[#ffffff] text-black text-sm font-bold hover:bg-white/90 transition-colors"
   305→            >
   306→              Refresh Page
   307→            </button>
   308→          )}
   309→        </div>
   310→      </div>
   311→    );
   312→  }
   313→
   314→  return (
   315→    <div className="absolute inset-0">
   316→      <iframe
   317→        ref={iframeRef}
   318→        src={src}
   319→        className="w-full h-full border-0"
   320→        allowFullScreen
   321→        allow="autoplay; fullscreen; picture-in-picture; encrypted-media"
   322→        referrerPolicy="no-referrer-when-downgrade"
   323→        title={`${animeTitle} - Episode ${episodeNum}`}
   324→        onLoad={handleIframeLoad}
   325→        onError={handleEmbedError}
   326→      />
   327→      {/* Floating "embed dead?" overlay — shows after 8s so user can switch servers */}
   328→      {showServerOverlay && otherProviders.length > 0 && (
   329→        <div className="absolute top-3 right-3 z-30">
   330→          <div className="flex items-center gap-2 bg-black/80 backdrop-blur-md rounded-lg border border-white/[0.08] px-3 py-2 shadow-xl">
   331→            <span className="text-[10px] text-white/55">Embed not working?</span>
   332→            {otherProviders.slice(0, 3).map(p => (
   333→              <button
   334→                key={p}
   335→                onClick={() => onProviderSelect(p)}
   336→                className="px-2.5 py-1 rounded-md bg-[#ffffff] text-black text-[10px] font-bold hover:bg-white/90 transition-colors"
   337→              >
   338→                {getProviderDisplayName(p)}
   339→              </button>
   340→            ))}
   341→            <button
   342→              onClick={() => setShowServerOverlay(false)}
   343→              className="p-1 text-white/40 hover:text-white transition-colors"
   344→            >
   345→              <svg className="w-3 h-3" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2.5}>
   346→                <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
   347→              </svg>
   348→            </button>
   349→          </div>
   350→        </div>
   351→      )}
   352→    </div>
   353→  );
   354→}
