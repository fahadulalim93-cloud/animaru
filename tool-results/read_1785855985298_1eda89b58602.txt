    13→function ordinal(n: number): string {
    14→  const s = ["th", "st", "nd", "rd"];
    15→  const v = n % 100;
    16→  return n + (s[(v - 20) % 10] || s[v] || s[0]);
    17→}

    55→function getTitle(a: any): string {
    56→  return a?.title?.english || a?.title?.romaji || a?.title?.native || "Unknown";
    57→}
    58→
    59→function getBanner(a: any): string {
    60→  // Prefer bannerImage (high quality), then extraLarge cover
    61→  const banner = a?.bannerImage;
    62→  if (banner) return banner;
    63→  // Fallback to cover image (extra large)
    64→  return a?.coverImage?.extraLarge || a?.coverImage?.large || "";
    65→}
    66→
    67→function getCover(a: any): string {
    68→  return a?.coverImage?.extraLarge || a?.coverImage?.large || a?.coverImage?.medium || "";
    69→}
    70→
    71→function getScore(a: any): number {
    72→  const s = a?.averageScore;
    73→  if (!s) return 0;
    74→  return s > 100 ? Math.round(s / 10) : s;
    75→}

    77→function formatCount(n?: number): string {
    78→  const v = n || 0;
    79→  if (v >= 1_000_000) return `${(v / 1_000_000).toFixed(1)}M`;
    80→  if (v >= 1_000) return `${(v / 1_000).toFixed(1)}K`;
    81→  return String(v);
    82→}

   103→function getStatusPill(status?: string): { label: string; className: string } | null {
   104→  if (!status) return null;
   105→  const s = status.toUpperCase();
   106→  if (s === "RELEASING" || s === "AIRING") return { label: "On-Going", className: "bg-emerald-500/15 text-emerald-400" };
   107→  if (s === "FINISHED") return { label: "Finished", className: "bg-white/10 text-white/60" };
   108→  if (s === "NOT_YET_RELEASED" || s === "NOT_YET_AIRED") return { label: "TBA", className: "bg-white/10 text-white/60" };
   109→  if (s === "CANCELLED") return { label: "Cancelled", className: "bg-white/10 text-white/60" };
   110→  if (s === "HIATUS") return { label: "Hiatus", className: "bg-white/10 text-white/60" };
   111→  return { label: status, className: "bg-white/10 text-white/60" };
   112→}

   118→function HeroCarousel({ items, navigate }: { items: FeaturedAnime[]; navigate: (r: any) => void }) {
   119→  const [current, setCurrent] = useState(0);
   120→  const [paused, setPaused] = useState(false);
   121→  const [logos, setLogos] = useState<Record<number, string>>({});
   122→  const [backdrops, setBackdrops] = useState<Record<number, string>>({});
   123→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   124→
   125→  // Fetch TVDB clearlogos + backgrounds for the hero banners.
   126→  // TVDB has the best anime clearlogos (transparent PNG logos like the
   127→  // One Piece logo). Falls back to AniList banner if TVDB has no background.
   128→  // Each TVDB API call has a 4s timeout to prevent Vercel function timeouts.
   129→  useEffect(() => {
   130→    let cancelled = false;
   131→    (async () => {
   132→      for (const anime of items) {
   133→        if (cancelled) return;
   134→        if (logos[anime.id] || backdrops[anime.id]) continue;
   135→        try {
   136→          const title = getTitle(anime);
   137→          const res = await fetch(`/api/anime/tmdb-images?anilistId=${anime.id}&title=${encodeURIComponent(title)}`);
   138→          if (res.ok) {
   139→            const data = await res.json();
   140→            if (cancelled) return;
   141→            if (data.logoUrl) setLogos(prev => ({ ...prev, [anime.id]: data.logoUrl }));
   142→            if (data.backdropUrl) setBackdrops(prev => ({ ...prev, [anime.id]: data.backdropUrl }));
   143→          }
   144→        } catch { /* keep going to the next banner */ }
   145→      }
   146→    })();
   147→    return () => { cancelled = true; };
   148→    // eslint-disable-next-line react-hooks/exhaustive-deps
   149→  }, [items]);
   150→
   151→  useEffect(() => {
   152→    if (paused || items.length === 0) return;
   153→    timerRef.current = setTimeout(() => {
   154→      setCurrent(prev => (prev + 1) % items.length);
   155→    }, 8000);
   156→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   157→  }, [current, paused, items.length]);
   158→
   159→  if (items.length === 0) {
   160→    return (
   161→      <div className="relative w-full h-[100dvh] bg-[#000000] flex items-center justify-center">
   162→        <div className="w-10 h-10 border-2 border-white/10 border-t-white rounded-full animate-spin" />
   163→      </div>
   164→    );
   165→  }
   166→
   167→  const anime = items[current];
   168→  // Use TVDB backdrop if available, otherwise AniList banner
   169→  const banner = backdrops[anime.id] || anime.bannerImage || getBanner(anime) || "";
   170→  const logoUrl = logos[anime.id];
   171→  const title = getTitle(anime);
   172→  const score = getScore(anime);
   173→  const seasonStr = anime.season && anime.seasonYear ? `${anime.season} ${anime.seasonYear}` : anime.seasonYear ? String(anime.seasonYear) : "";
   174→  const description = anime.description ? anime.description.replace(/<[^>]*>/g, "") : "";
   175→
   176→  return (
   177→    <div
   178→      className="relative w-full h-[85dvh] md:h-[100dvh] min-h-[480px] md:min-h-[560px] overflow-hidden bg-[#000000]"
   179→      onMouseEnter={() => setPaused(true)}
   180→      onMouseLeave={() => setPaused(false)}
   181→    >
   182→      {/* ── Banner image — full-bleed, object-cover for cinematic crop ── */}
   183→      {banner && (
   184→        <img
   185→          src={banner}
   186→          alt=""
   187→          className="object-cover object-[center_30%] md:object-center"
   188→          style={{ position: "absolute", inset: 0, width: "100%", height: "100%", animation: "ltv-hero-crossfade 1.2s ease-in-out" }}
   189→          key={`bg-${current}`}
   190→          onError={(e) => {
   191→            const fallback = anime.bannerImage || getBanner(anime) || "";
   192→            const img = e.currentTarget;
   193→            if (fallback && img.src !== fallback) img.src = fallback;
   194→            else img.style.display = "none";

   196→        />

   198→
   199→      {/* ── Overlay system — matches the original cinematic look the user liked:

   303→// Hover title-tint colors, cycled per card index (4-color loop) instead of

   316→
   317→  const scroll = (dir: "left" | "right") => {
   318→    if (scrollRef.current) {
   319→      scrollRef.current.scrollBy({ left: dir === "right" ? 700 : -700, behavior: "smooth" });
   320→    }
   321→  };
   322→
   323→  return (
   324→    <section className="px-4 md:px-8 lg:px-8 py-4">
   325→      <div className="flex items-center justify-between mb-4">
   326→        <h2

   329→        >
   330→          {title}
   331→        </h2>
   332→        <div className="flex gap-2">
   333→          <button onClick={() => scroll("left")} className="w-8 h-8 rounded-full bg-white/5 hover:bg-white/10 flex items-center justify-center text-white/60 hover:text-white transition-colors">
   334→            <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}><path d="M15 19l-7-7 7-7" /></svg>

   337→            <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}><path d="M9 5l7 7-7 7" /></svg>
   338→          </button>
   339→        </div>
   340→      </div>
   341→      <div ref={scrollRef} className="flex gap-4 overflow-x-auto pb-2" style={{ scrollbarWidth: "none" }}>
   342→        {cards.map((anime, idx) => {
   343→          const cover = getCover(anime);

   422→function getMangaStatusPill(status?: string): { label: string; className: string } | null {
   423→  if (!status) return null;
   424→  const s = status.toLowerCase();
   425→  if (s.includes("ongoing") || s.includes("releasing") || s.includes("airing")) {
   426→    return { label: "On-Going", className: "bg-emerald-500/15 text-emerald-400" };
   427→  }
   428→  if (s.includes("complet") || s.includes("finish")) {
   429→    return { label: "Finished", className: "bg-white/10 text-white/60" };
   430→  }

   733→function GenreSpotlight({ navigate }: { navigate: (r: any) => void }) {
   734→  // genre + items land together so the heading never shows a genre whose rail
   735→  // hasn't loaded. Both are set from the async callback rather than
   736→  // synchronously in the effect body, which avoids a cascading re-render.
   737→  const [data, setData] = useState<{ genre: string; items: any[] }>({ genre: "", items: [] });
   738→  const [loading, setLoading] = useState(true);
   739→  // TMDB clear-logo (transparent PNG) + clean backdrop per anime, same source
   740→  // the hero carousel uses.
   741→  const [art, setArt] = useState<Record<number, { logo?: string; backdrop?: string }>>({});
   742→  const scrollRef = useRef<HTMLDivElement>(null);
   743→  const { genre, items } = data;
   744→
   745→  useEffect(() => {
   746→    let cancelled = false;
   747→    // Chosen inside the effect (not at render) so server and client can't
   748→    // disagree on the date and trip a hydration mismatch.
   749→    const dayIndex = Math.floor(Date.now() / 86_400_000);
   750→    const picked = SPOTLIGHT_GENRES[dayIndex % SPOTLIGHT_GENRES.length];
   751→
   752→    fetch(`/api/anime/browse?genre=${encodeURIComponent(picked)}&sort=most-popular&perPage=14`)
   753→      .then(r => (r.ok ? r.json() : null))
   754→      .then(res => {
   755→        if (cancelled) return;

   757→        // 16:9 looks broken, and a blank tile looks worse.

   759→        setData({ genre: picked, items: withBanner.slice(0, 12) });
   760→      })
   761→      .catch(() => { /* best-effort — section hides itself if nothing loads */ })
   762→      .finally(() => { if (!cancelled) setLoading(false); });

   784→          }
   785→        } catch { /* skip this one, keep going */ }
