     1→'use client';
     2→
     3→import { useEffect, useRef, useState, useCallback } from 'react';
     4→import Hls from 'hls.js';
     5→import { proxify, proxifyM3u8 } from '@/lib/proxy';
     6→import { validateSkipTime } from '@/lib/episode-metadata';
     7→import {
     8→  useSubtitleSettings,
     9→  CustomSubtitleOverlay,
    10→  SubtitleSettingsPanel,
    11→} from './subtitle-settings';
    12→
    13→interface HLSPlayerProps {
    14→  url: string;
    15→  animeId?: string;
    16→  episodeNum?: number;
    17→  sourceType?: 'hls' | 'mp4';
    18→  intro?: { start: number; end: number } | null;
    19→  outro?: { start: number; end: number } | null;
    20→  allStreams?: Array<{ url: string; quality: string; label: string }>;
    21→  subtitleTracks?: Array<{ url: string; lang: string; label: string }>;
    22→  onEnded?: () => void;
    23→  onProviderFailed?: () => void;
    24→  onCanPlay?: () => void;
    25→  autoplay?: boolean;
    26→  autoSkip?: boolean;
    27→}
    28→
    29→// ============================================================
    30→// Glass Player — Animetsu-inspired glassmorphism design
    31→// Features:
    32→//   - Floating glass control bar (backdrop-blur, translucent)
    33→//   - Animated progress bar with glow + gradient
    34→//   - Smooth scale/translate animations on all interactions
    35→//   - Glass menus for quality/subtitles/speed
    36→//   - Animated center play button with pulse ring
    37→//   - Double-tap to seek with ripple animation
    38→//   - Keyboard shortcuts
    39→// ============================================================
    40→
    41→export default function HLSPlayerNew({
    42→  url, animeId, episodeNum, sourceType = 'hls',
    43→  intro: introProp, outro: outroProp, allStreams, subtitleTracks, onEnded, onProviderFailed, onCanPlay, autoplay = true, autoSkip = false,
    44→}: HLSPlayerProps) {
    45→  const videoRef = useRef<HTMLVideoElement>(null);
    46→  const hlsRef = useRef<Hls | null>(null);
    47→  const containerRef = useRef<HTMLDivElement>(null);
    48→
    49→  // Defense-in-depth: validate skip times at the player level too.
    50→  // Even if upstream code fails to filter bad data, this guarantees
    51→  // the outro button never appears at the start of the video (which
    52→  // happens when outro.start = 0 from a provider's "no data" sentinel).
    53→  const intro = validateSkipTime(introProp ?? null, "intro");
    54→  const outro = validateSkipTime(outroProp ?? null, "outro");
    55→
    56→  // Keep the latest callbacks in refs so the stream-loading and video-event
    57→  // effects DON'T list them as deps. Parents (e.g. watch-page-shell) re-render
    58→  // on every `timeupdate` (several times/sec) and often pass inline arrow
    59→  // callbacks — a new identity each render. If those were in the effect deps,
    60→  // the stream effect would destroy + recreate the hls.js instance on every
    61→  // frame, refetching the m3u8 + all segments in an infinite loop. Reading
    62→  // through a ref keeps the effect stable (only url/sourceType/autoplay matter).
    63→  const onProviderFailedRef = useRef(onProviderFailed);
    64→  const onEndedRef = useRef(onEnded);
    65→  const onCanPlayRef = useRef(onCanPlay);
    66→  const onCanPlayFiredRef = useRef(false);
    67→  const autoSkipRef = useRef(autoSkip);
    68→  onProviderFailedRef.current = onProviderFailed;
    69→  onEndedRef.current = onEnded;
    70→  onCanPlayRef.current = onCanPlay;
    71→  autoSkipRef.current = autoSkip;
    72→  const [playing, setPlaying] = useState(false);
    73→  const [currentTime, setCurrentTime] = useState(0);
    74→  const [duration, setDuration] = useState(0);
    75→  const [buffered, setBuffered] = useState(0);
    76→  const [volume, setVolume] = useState(1);
    77→  const [muted, setMuted] = useState(false);
    78→  const [fullscreen, setFullscreen] = useState(false);
    79→  const [loading, setLoading] = useState(true);
    80→  const [error, setError] = useState<string | null>(null);
    81→  const [showControls, setShowControls] = useState(true);
    82→  const isMobileRef = useRef(false);
    83→  if (typeof window !== 'undefined') isMobileRef.current = 'ontouchstart' in window || window.innerWidth < 768;
    84→  const [qualities, setQualities] = useState<any[]>([]);
    85→  const [currentQuality, setCurrentQuality] = useState(-1);
    86→  const [activeMenu, setActiveMenu] = useState<'quality' | 'subtitles' | 'speed' | null>(null);
    87→  const [showSubtitleSettings, setShowSubtitleSettings] = useState(false);
    88→  const [showDownloadModal, setShowDownloadModal] = useState(false);
    89→  const { settings: subtitleSettings, update: updateSubtitleSettings, reset: resetSubtitleSettings } = useSubtitleSettings();
    90→  const [currentSubtitle, setCurrentSubtitle] = useState(-1);
    91→  const currentSubtitleRef = useRef(currentSubtitle);
    92→  currentSubtitleRef.current = currentSubtitle;
    93→  const [hlsSubtitles, setHlsSubtitles] = useState<any[]>([]);
    94→  const [playbackRate, setPlaybackRate] = useState(1);
    95→  const [showSkipIntro, setShowSkipIntro] = useState(false);
    96→  const [showSkipOutro, setShowSkipOutro] = useState(false);
    97→  // Track when user clicked skip — prevents the button from re-appearing
    98→  // immediately after the click (the timeupdate loop would otherwise keep
    99→  // showing it because video.currentTime is still in the skip range).
   100→  const skipIntroClickedRef = useRef(0);
   101→  const skipOutroClickedRef = useRef(0);
   102→  const [hoverTime, setHoverTime] = useState<number | null>(null);
   103→  const [hoverX, setHoverX] = useState(0);
   104→  const [seekRipple, setSeekRipple] = useState<{ x: number; dir: 'left' | 'right' } | null>(null);
   105→  const [volumeHover, setVolumeHover] = useState(false);
   106→  const [screenshotFlash, setScreenshotFlash] = useState(false);
   107→  const [screenshotToast, setScreenshotToast] = useState<string | null>(null);
   108→  const [downloadLoading, setDownloadLoading] = useState(false);
   109→  const [downloadToast, setDownloadToast] = useState<string | null>(null);
   110→  const [streamHealth, setStreamHealth] = useState<'good' | 'fair' | 'poor' | 'unknown'>('unknown');
   111→  const [showStreamInfo, setShowStreamInfo] = useState(false);
   112→  const controlsTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   113→  const retryCountRef = useRef(0);
   114→  const lastTapRef = useRef<{ time: number; x: number } | null>(null);
   115→  const canvasRef = useRef<HTMLCanvasElement>(null);
   116→
   117→  // ─── Load stream ──────────────────────────────────────────────────
   118→  useEffect(() => {
   119→    const video = videoRef.current;
   120→    if (!video || !url) return;
   121→
   122→    setLoading(true);
   123→    setError(null);
   124→    retryCountRef.current = 0;
   125→    setCurrentSubtitle(-1);
   126→    setHlsSubtitles([]);
   127→    onCanPlayFiredRef.current = false; // reset for new stream
   128→
   129→    if (hlsRef.current) {
   130→      hlsRef.current.destroy();
   131→      hlsRef.current = null;
   132→    }
   133→
   134→    if (sourceType === 'mp4' || url.endsWith('.mp4') || url.includes('video.mp4')) {
   135→      // Proxy the MP4 URL through the worker so the correct Referer is sent
   136→      video.src = proxify(url);
   137→      video.load();
   138→      if (autoplay) video.play().catch(() => {});
   139→      return;
   140→    }
   141→
   142→    if (Hls.isSupported()) {
   143→      const hls = new Hls({
   144→        enableWorker: true,
   145→        // ─── PERFORMANCE TUNING — optimized for FASTEST initial load ───
   146→        // Strategy: Start at LOWEST quality (360p) for instant first-frame,
   147→        // then ABR auto-upgrades to 1080p within 2-3 seconds as bandwidth
   148→        // is measured. This gives the FASTEST perceived load time.
   149→        // Previous config started at 1080p which meant downloading 5-10MB
   150→        // before any frame appeared — now we download ~500KB first.
   151→        lowLatencyMode: true,            // Enables reduced buffer targets + faster ABR
   152→        backBufferLength: 20,            // 20s back buffer — saves memory, plenty for seeking
   153→        maxBufferLength: 10,             // 10s ahead buffer — minimal wait before playback starts
   154→        maxMaxBufferLength: 120,         // 120s max buffer — grows for smooth long-form playback
   155→        maxBufferSize: 50 * 1000 * 1000, // 50MB max buffer size
   156→        maxBufferHole: 0.5,
   157→        // Start at the LOWEST quality level for FASTEST time-to-first-frame.
   158→        // ABR will auto-upgrade to 1080p within 2-3s on fast connections.
   159→        // -1 = ABR auto-select, 0 = lowest level (fastest first frame).
   160→        startLevel: 0,
   161→        // Assume 5Mbps bandwidth initially — conservative start ensures
   162→        // smooth playback while ABR measures real bandwidth.
   163→        // After 2-3s of measurement, ABR ramps up to full 1080p.
   164→        abrEwmaDefaultEstimate: 5000000,   // 5Mbps — conservative, ramps up fast
   165→        abrBandWidthFactor: 0.8,           // More aggressive quality upgrades
   166→        abrBandWidthUpFactor: 0.7,
   167→        maxStarvationDelay: 1,             // Switch quality instantly if buffer runs dry
   168→        abrEwmaDefaultEstimateMax: 50000000, // 50Mbps max — allows 4K streams
   169→        abrMaxWithRealBitrate: true,        // Use real bitrate from manifest for better ABR
   170→        // ─── TIMEOUTS — fail FAST, retry FAST ───
   171→        // Aggressive timeouts so dead servers don't hold up playback.
   172→        manifestLoadingTimeOut: 6000,      // 6s — proxy+CDN should respond in <3s
   173→        manifestLoadingMaxRetry: 2,        // Only retry twice — move on quickly
   174→        levelLoadingTimeOut: 6000,         // 6s
   175→        levelLoadingMaxRetry: 2,
   176→        fragLoadingTimeOut: 10000,         // 10s per segment — generous even on slow CDNs
   177→        fragLoadingMaxRetry: 3,            // 3 retries — enough to recover from transient errors
   178→        fragLoadingRetryDelay: 250,        // 250ms — retry almost immediately
   179→        // ─── PREFETCH + PROGRESSIVE for instant start ───
   180→        startFragPrefetch: true,           // Prefetch first fragment while parsing manifest
   181→        progressive: true,                 // Stream data progressively (faster TTFF)
   182→        testBandwidth: true,               // Bandwidth test on first fragment for faster ABR
   183→        // ─── SUBTITLES ───
   184→        enableWebVTT: true,
   185→        xhrSetup: (xhr) => { xhr.withCredentials = false; },
   186→      });
   187→      hlsRef.current = hls;
   188→      hls.loadSource(proxifyM3u8(url));
   189→      hls.attachMedia(video);
   190→
   191→      hls.on(Hls.Events.MANIFEST_PARSED, (_e, data) => {
   192→        setQualities(data.levels);
   193→        setLoading(false);
   194→        // Reset retry counter on successful manifest parse
   195→        retryCountRef.current = 0;
   196→        if (autoplay) {
   197→          video.play().catch(() => {
   198→            video.muted = true;
   199→            video.play().catch(() => {});
   200→          });
   201→        }
   202→      });
   203→
   204→      hls.on(Hls.Events.SUBTITLE_TRACKS_UPDATED, (_e, data) => {
   205→        setHlsSubtitles(data.subtitleTracks || []);
   206→        if (data.subtitleTracks.length > 0 && hls.subtitleTrack === -1) {
   207→          hls.subtitleTrack = 0;
   208→          setCurrentSubtitle(0);
   209→          // After HLS.js sets the subtitle track, it creates a textTrack with
   210→          // mode='showing' (native rendering). We need to switch it to 'hidden'
   211→          // so only our CustomSubtitleOverlay renders (prevents double subtitles).
   212→          setTimeout(() => {
   213→            const video = videoRef.current;
   214→            if (video) {
   215→              for (let i = 0; i < video.textTracks.length; i++) {
   216→                if (i === 0) video.textTracks[i].mode = 'hidden';
   217→              }
   218→            }
   219→          }, 200);
   220→        }
   221→      });
   222→
   223→      hls.on(Hls.Events.LEVEL_SWITCHED, (_e, data) => {
   224→        setCurrentQuality(data.level);
   225→        // After a level switch, HLS.js may re-enable native subtitle rendering.
   226→        // Force the active track back to 'hidden' mode so only our overlay shows.
   227→        if (currentSubtitleRef.current >= 0) {
   228→          setTimeout(() => {
   229→            const video = videoRef.current;
   230→            if (video && video.textTracks.length > currentSubtitleRef.current) {
   231→              video.textTracks[currentSubtitleRef.current].mode = 'hidden';
   232→            }
   233→          }, 100);
   234→        }
   235→      });
   236→
   237→      hls.on(Hls.Events.ERROR, (_e, data) => {
   238→        // Non-fatal errors are expected during streaming (a segment/playlist
   239→        // fetch hiccup) and hls.js already recovers from them on its own —
   240→        // console.warn so they don't trip Next's dev error overlay for
   241→        // something that isn't actually broken.
   242→        const log = data.fatal ? console.error : console.warn;
   243→        log(`[HLS] Error: type=${data.type} details=${data.details} fatal=${data.fatal}`);
   244→        if (data.fatal) {
   245→          if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
   246→            if (retryCountRef.current < 3) {
   247→              retryCountRef.current++;
   248→              setTimeout(() => hls.startLoad(), 2000);
   249→            } else {
   250→              setError('Stream failed. Try another server.');
   251→              setLoading(false);
   252→              onProviderFailedRef.current?.();
   253→            }
   254→          } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
   255→            hls.recoverMediaError();
   256→          } else {
   257→            setError('Playback error. Try another server.');
   258→            setLoading(false);
   259→            onProviderFailedRef.current?.();
   260→          }
   261→        }
   262→      });
   263→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   264→      // Safari/iOS — native HLS
   265→      video.src = proxifyM3u8(url);
   266→      const onNativeError = () => { setError('Stream failed to load.'); setLoading(false); };
   267→      video.addEventListener('error', onNativeError);
   268→      if (autoplay) video.play().catch(() => {});
   269→      return () => {
   270→        video.removeEventListener('error', onNativeError);
   271→      };
   272→    } else {
   273→      setError('HLS not supported');
   274→      setLoading(false);
   275→    }
   276→
   277→    return () => {
   278→      if (hlsRef.current) {
   279→        hlsRef.current.destroy();
   280→        hlsRef.current = null;
   281→      }
   282→    };
   283→  }, [url, sourceType, autoplay]);
   284→
   285→  // ─── Video events ─────────────────────────────────────────────────
   286→  useEffect(() => {
   287→    const video = videoRef.current;
   288→    if (!video) return;
   289→
   290→    const onPlay = () => setPlaying(true);
   291→    const onPause = () => setPlaying(false);
   292→    const onTime = () => {
   293→      setCurrentTime(video.currentTime);
   294→      if (video.buffered.length > 0) {
   295→        setBuffered(video.buffered.end(video.buffered.length - 1));
   296→      }
   297→      // Auto Skip — when enabled, jump straight past the opening/ending.
   298→      if (autoSkipRef.current) {
   299→        if (intro && video.currentTime >= intro.start && video.currentTime < intro.end - 0.3) {
   300→          video.currentTime = intro.end;
   301→          return;
   302→        }
   303→        if (outro && video.currentTime >= outro.start && video.currentTime < outro.end - 0.3) {
   304→          video.currentTime = Math.min(outro.end, video.duration || outro.end);
   305→          return;
   306→        }
   307→      }
   308→      // Show skip intro button from 3s BEFORE intro starts until 5s AFTER it ends.
   309→      // Also show immediately if video just loaded and we're already in the intro range
   310→      // (happens when mimi loads super fast and starts playing at 0s, which is before intro.start)
   311→      //
   312→      // BUT: if the user just clicked Skip Intro (within last 8s), DON'T re-show the
   313→      // button — otherwise it instantly pops back up because video.currentTime is still
   314→      // in the intro range right after the seek.
   315→      const sinceIntroClick = Date.now() - skipIntroClickedRef.current;
   316→      const sinceOutroClick = Date.now() - skipOutroClickedRef.current;
   317→
   318→      if (intro && video.currentTime >= (intro.start - 3) && video.currentTime < (intro.end + 5) && sinceIntroClick > 8000) {
   319→        setShowSkipIntro(true);
   320→      } else {
   321→        setShowSkipIntro(false);
   322→      }
   323→      if (outro && video.currentTime >= (outro.start - 3) && video.currentTime < (outro.end + 10) && sinceOutroClick > 8000) {
   324→        setShowSkipOutro(true);
   325→      } else {
   326→        setShowSkipOutro(false);
   327→      }
   328→    };
   329→
   330→    // Also check on MANIFEST_PARSED / loadedmetadata — if intro data is available
   331→    // and we're at position 0, show the skip button immediately (before timeupdate fires)
   332→    const onLoadedMetadata = () => {
   333→      setDuration(video.duration || 0);
   334→      // If intro starts soon (within first 60s), show the button immediately
   335→      if (intro && intro.start < 60 && video.currentTime < intro.end) {
   336→        setShowSkipIntro(true);
   337→      }
   338→      if (outro && video.duration && outro.start < video.duration) {
   339→        // Don't show outro yet, just confirm we have the data
   340→      }
   341→    };
   342→    const onDur = () => setDuration(video.duration || 0);
   343→
   344→    let waitingTimer: ReturnType<typeof setTimeout> | null = null;
   345→    const onWaiting = () => {
   346→      if (waitingTimer) clearTimeout(waitingTimer);
   347→      // Show loading after 1.5s of waiting (was 5s — too long, user gets confused)
   348→      waitingTimer = setTimeout(() => {
   349→        if (video.readyState < 3) setLoading(true);
   350→      }, 1500);
   351→    };
   352→    const onPlaying = () => {
   353→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   354→      setLoading(false);
   355→    };
   356→    const onCanPlay = () => {
   357→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   358→      setLoading(false);
   359→      // Notify parent that video is ready to play (for loading screen dismissal)
   360→      if (!onCanPlayFiredRef.current) {
   361→        onCanPlayFiredRef.current = true;
   362→        onCanPlayRef.current?.();
   363→      }
   364→    };
   365→    const onEnd = () => onEndedRef.current?.();
   366→    const onErr = () => { setError('Playback error.'); setLoading(false); };
   367→
   368→    video.addEventListener('play', onPlay);
   369→    video.addEventListener('pause', onPause);
   370→    video.addEventListener('timeupdate', onTime);
   371→    video.addEventListener('durationchange', onDur);
   372→    video.addEventListener('loadedmetadata', onLoadedMetadata);
   373→    video.addEventListener('waiting', onWaiting);
   374→    video.addEventListener('playing', onPlaying);
   375→    video.addEventListener('canplay', onCanPlay);
   376→    video.addEventListener('ended', onEnd);
   377→    video.addEventListener('error', onErr);
   378→
   379→    return () => {
   380→      video.removeEventListener('play', onPlay);
   381→      video.removeEventListener('pause', onPause);
   382→      video.removeEventListener('timeupdate', onTime);
   383→      video.removeEventListener('durationchange', onDur);
   384→      video.removeEventListener('loadedmetadata', onLoadedMetadata);
   385→      video.removeEventListener('waiting', onWaiting);
   386→      video.removeEventListener('playing', onPlaying);
   387→      video.removeEventListener('canplay', onCanPlay);
   388→      video.removeEventListener('ended', onEnd);
   389→      video.removeEventListener('error', onErr);
   390→    };
   391→  }, [intro, outro]);
   392→
   393→  // ─── Controls auto-hide ───────────────────────────────────────────
   394→  const CONTROLS_TIMEOUT = isMobileRef.current ? 4000 : 3500;
   395→  const showControlsTemp = useCallback(() => {
   396→    setShowControls(true);
   397→    if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   398→    controlsTimeoutRef.current = setTimeout(() => {
   399→      if (playing) { setShowControls(false); setActiveMenu(null); }
   400→    }, CONTROLS_TIMEOUT);
   401→  }, [playing, CONTROLS_TIMEOUT]);
   402→
   403→  // ─── Auto-hide controls when video starts playing ─────────────────
   404→  // On mobile, there's no mousemove to trigger showControlsTemp(),
   405→  // so the controls stay visible forever. This effect starts the
   406→  // auto-hide timer whenever playback begins.
   407→  useEffect(() => {
   408→    if (playing) {
   409→      // Start the auto-hide timer — controls will disappear after CONTROLS_TIMEOUT
   410→      if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   411→      controlsTimeoutRef.current = setTimeout(() => {
   412→        setShowControls(false);
   413→        setActiveMenu(null);
   414→      }, CONTROLS_TIMEOUT);
   415→    } else {
   416→      // Paused — show controls and cancel any pending hide
   417→      if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   418→      setShowControls(true);
   419→    }
   420→  }, [playing, CONTROLS_TIMEOUT]);
   421→
   422→  // ─── Fullscreen (cross-browser: standard + webkit for iOS Safari) ───
   423→  const getFullscreenElement = () =>
   424→    document.fullscreenElement ||
   425→    (document as any).webkitFullscreenElement ||
   426→    null;
   427→
   428→  const requestFullscreen = async (el: HTMLElement) => {
   429→    if (el.requestFullscreen) return el.requestFullscreen();
   430→    if ((el as any).webkitRequestFullscreen) return (el as any).webkitRequestFullscreen();
   431→    // iOS Safari < 16.4: only <video> can go fullscreen
   432→    if (videoRef.current && (videoRef.current as any).webkitEnterFullscreen) {
   433→      return (videoRef.current as any).webkitEnterFullscreen();
   434→    }
   435→    throw new Error('Fullscreen not supported');
   436→  };
   437→
   438→  const exitFullscreen = async () => {
   439→    if (document.exitFullscreen) return document.exitFullscreen();
   440→    if ((document as any).webkitExitFullscreen) return (document as any).webkitExitFullscreen();
   441→    // iOS video fullscreen
   442→    if (videoRef.current && (videoRef.current as any).webkitExitFullscreen) {
   443→      return (videoRef.current as any).webkitExitFullscreen();
   444→    }
   445→  };
   446→
   447→  useEffect(() => {
   448→    const onFs = () => setFullscreen(!!getFullscreenElement());
   449→    document.addEventListener('fullscreenchange', onFs);
   450→    document.addEventListener('webkitfullscreenchange', onFs);
   451→    return () => {
   452→      document.removeEventListener('fullscreenchange', onFs);
   453→      document.removeEventListener('webkitfullscreenchange', onFs);
   454→    };
   455→  }, []);
   456→
   457→  const toggleFullscreen = async () => {
   458→    try {
   459→      if (getFullscreenElement()) {
   460→        await exitFullscreen();
   461→        // Release orientation lock when exiting fullscreen
   462→        try { await (screen.orientation as any)?.unlock(); } catch {}
   463→      } else if (containerRef.current) {
   464→        await requestFullscreen(containerRef.current);
   465→        // On mobile, lock to landscape so the video fills the screen
   466→        try {
   467→          if (screen.orientation && typeof (screen.orientation as any).lock === 'function') {
   468→            await (screen.orientation as any).lock('landscape');
   469→          }
   470→        } catch {
   471→          // Orientation lock not supported or denied — fullscreen still works
   472→        }
   473→      }
   474→    } catch {
   475→      // Fullscreen request failed silently
   476→    }
   477→  };
   478→
   479→  const seek = (time: number) => { if (videoRef.current) videoRef.current.currentTime = time; };
   480→  const togglePlay = () => {
   481→    const v = videoRef.current; if (!v) return;
   482→    if (v.paused) v.play(); else v.pause();
   483→  };
   484→  const toggleMute = () => {
   485→    const v = videoRef.current; if (!v) return;

  1323→// ─── Download Modal ─────────────────────────────────────────────
  1324→function DownloadModal({ animeId, episodeNum, onClose }: { animeId: string; episodeNum: number; onClose: () => void }) {
  1325→  const [loading, setLoading] = useState(true);
  1326→  const [links, setLinks] = useState<Array<{ text: string; decodedUrl: string }>>([]);
  1327→  const [error, setError] = useState('');
  1328→
  1329→  useEffect(() => {
  1330→    const fetchDownloads = async () => {
  1331→      try {
  1332→        setLoading(true);
  1333→        setError('');
  1334→        let title = '';
  1335→        try {
  1336→          const titleRes = await fetch('https://graphql.anilist.co', {
  1337→            method: 'POST',
  1338→            headers: { 'Content-Type': 'application/json' },
  1339→            body: JSON.stringify({
  1340→              query: `query($id:Int){Media(id:$id,type:ANIME){title{english romaji}}}`,
  1341→              variables: { id: parseInt(animeId) },
  1342→            }),
  1343→          });
  1344→          if (titleRes.ok) {
  1345→            const titleData = await titleRes.json();
  1346→            title = titleData?.data?.Media?.title?.english || titleData?.data?.Media?.title?.romaji || '';
  1347→          }
  1348→        } catch { /* ignore */ }
  1349→
  1350→        if (!title) {
  1351→          setError('Could not determine anime title');
  1352→          setLoading(false);
  1353→          return;
  1354→        }
  1355→
  1356→        const res = await fetch(`/api/anime/download?title=${encodeURIComponent(title)}&auto=1`);
  1357→        if (!res.ok) {
  1358→          setError('Failed to fetch download links');
  1359→          setLoading(false);
  1360→          return;
  1361→        }
  1362→        const data = await res.json();
  1363→        if (data.links && data.links.length > 0) {
  1364→          setLinks(data.links);
  1365→        } else {
  1366→          setError('No download links found for this anime');
  1367→        }
  1368→      } catch (e: any) {
  1369→        setError(e?.message || 'Failed to load downloads');
  1370→      } finally {
  1371→        setLoading(false);
  1372→      }
  1373→    };
  1374→    fetchDownloads();
  1375→  }, [animeId]);
  1376→
  1377→  // Show only top 2 links (quick access) + "explore more" link
  1378→  const quickLinks = links.slice(0, 2);
  1379→
  1380→  return (
  1381→    <div
  1382→      className="absolute inset-0 z-50 flex items-center justify-center"
  1383→      style={{ background: 'rgba(0,0,0,0.85)', backdropFilter: 'blur(4px)' }}
  1384→      onClick={onClose}
  1385→    >
  1386→      <div
  1387→        className="bg-black border border-white/10 rounded-2xl shadow-2xl w-full max-w-sm mx-4 overflow-hidden"
  1388→        onClick={(e) => e.stopPropagation()}
  1389→      >
  1390→        {/* Header */}
  1391→        <div className="flex items-center justify-between px-5 py-3.5 border-b border-white/10">
  1392→          <div className="flex items-center gap-2">
  1393→            <svg className="w-4 h-4 text-white" viewBox="0 0 24 24" fill="currentColor"><path d="M19 9h-4V5a3 3 0 0 0-6 0v4H5a2 2 0 0 0-2 2v8a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-8a2 2 0 0 0-2-2zm-6 6v3h-2v-3H8l4-4 4 4h-3z" /></svg>
  1394→            <h3 className="text-sm font-bold text-white">Download</h3>
  1395→          </div>
  1396→          <button onClick={onClose} className="text-white/40 hover:text-white transition-colors">
  1397→            <svg className="w-4 h-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2}><path d="M18 6L6 18M6 6l12 12" strokeLinecap="round" strokeLinejoin="round" /></svg>
  1398→          </button>
  1399→        </div>
  1400→
  1401→        {/* Body */}
  1402→        <div className="px-4 py-4">
  1403→          {loading && (
  1404→            <div className="flex items-center justify-center py-8">
  1405→              <div className="w-5 h-5 border-2 border-white/20 border-t-white rounded-full animate-spin" />
  1406→              <span className="ml-2.5 text-xs text-white/50">Finding links...</span>
  1407→            </div>
  1408→          )}
  1409→
  1410→          {error && !loading && (
  1411→            <div className="text-center py-6">
  1412→              <p className="text-xs text-white/40 mb-3">{error}</p>
  1413→              {/* Even on error, show the explore link */}
  1414→              <a
  1415→                href="https://animex.one/community/download"
  1416→                target="_blank"
  1417→                rel="noopener noreferrer"
  1418→                className="inline-flex items-center gap-1.5 text-[11px] font-semibold text-[#1E88FF] hover:underline"
  1419→              >
  1420→                Want to explore downloads?
  1421→                <svg className="w-3 h-3" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2}><path d="M7 17L17 7M7 7h10v10" strokeLinecap="round" strokeLinejoin="round" /></svg>
  1422→              </a>
  1423→            </div>
  1424→          )}
  1425→

  1437→                  >
  1438→                    <div className="flex items-center gap-2.5">
  1439→                      <div className="w-7 h-7 rounded-lg bg-white/5 flex items-center justify-center shrink-0 group-hover:bg-white/10 transition-colors">
  1440→                        <svg className="w-3 h-3 text-white/50" viewBox="0 0 24 24" fill="currentColor"><path d="M19 9h-4V5a3 3 0 0 0-6 0v4H5a2 2 0 0 0-2 2v8a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-8a2 2 0 0 0-2-2zm-6 6v3h-2v-3H8l4-4 4 4h-3z" /></svg>

  1464→                </a>
