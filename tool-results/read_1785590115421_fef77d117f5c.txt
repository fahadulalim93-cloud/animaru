    29→// ============================================================
    30→// Glass Player — Animetsu-inspired glassmorphism design
    31→// Features:
    32→//   - Floating glass control bar (backdrop-blur, translucent)
    33→//   - Animated progress bar with glow + gradient
    34→//   - Smooth scale/translate animations on all interactions
    35→//   - Glass menus for quality/subtitles/speed
    36→//   - Animated center play button with pulse ring
    37→//   - Double-tap to seek with ripple animation
    38→//   - Keyboard shortcuts

    40→
    41→export default function HLSPlayerNew({
    42→  url, animeId, episodeNum, sourceType = 'hls',
    43→  intro: introProp, outro: outroProp, allStreams, subtitleTracks, onEnded, onProviderFailed, onCanPlay, autoplay = true, autoSkip = false,
    44→}: HLSPlayerProps) {
    45→  const videoRef = useRef<HTMLVideoElement>(null);
    46→  const hlsRef = useRef<Hls | null>(null);
    47→  const containerRef = useRef<HTMLDivElement>(null);
    48→
    49→  // Defense-in-depth: validate skip times at the player level too.
    50→  // Even if upstream code fails to filter bad data, this guarantees
    51→  // the outro button never appears at the start of the video (which
    52→  // happens when outro.start = 0 from a provider's "no data" sentinel).
    53→  const intro = validateSkipTime(introProp ?? null, "intro");
    54→  const outro = validateSkipTime(outroProp ?? null, "outro");
    55→
    56→  // Keep the latest callbacks in refs so the stream-loading and video-event
    57→  // effects DON'T list them as deps. Parents (e.g. watch-page-shell) re-render
    58→  // on every `timeupdate` (several times/sec) and often pass inline arrow
    59→  // callbacks — a new identity each render. If those were in the effect deps,
    60→  // the stream effect would destroy + recreate the hls.js instance on every
    61→  // frame, refetching the m3u8 + all segments in an infinite loop. Reading
    62→  // through a ref keeps the effect stable (only url/sourceType/autoplay matter).
    63→  const onProviderFailedRef = useRef(onProviderFailed);
    64→  const onEndedRef = useRef(onEnded);
    65→  const onCanPlayRef = useRef(onCanPlay);
    66→  const onCanPlayFiredRef = useRef(false);
    67→  const autoSkipRef = useRef(autoSkip);
    68→  onProviderFailedRef.current = onProviderFailed;
    69→  onEndedRef.current = onEnded;
    70→  onCanPlayRef.current = onCanPlay;
    71→  autoSkipRef.current = autoSkip;
    72→  const [playing, setPlaying] = useState(false);
    73→  const [currentTime, setCurrentTime] = useState(0);
    74→  const [duration, setDuration] = useState(0);
    75→  const [buffered, setBuffered] = useState(0);
    76→  const [volume, setVolume] = useState(1);
    77→  const [muted, setMuted] = useState(false);
    78→  const [fullscreen, setFullscreen] = useState(false);
    79→  const [loading, setLoading] = useState(true);
    80→  const [error, setError] = useState<string | null>(null);
    81→  const [showControls, setShowControls] = useState(true);
    82→  const [qualities, setQualities] = useState<any[]>([]);
    83→  const [currentQuality, setCurrentQuality] = useState(-1);
    84→  const [activeMenu, setActiveMenu] = useState<'quality' | 'subtitles' | 'speed' | null>(null);
    85→  const [showSubtitleSettings, setShowSubtitleSettings] = useState(false);
    86→  const [showDownloadModal, setShowDownloadModal] = useState(false);
    87→  const { settings: subtitleSettings, update: updateSubtitleSettings, reset: resetSubtitleSettings } = useSubtitleSettings();
    88→  const [currentSubtitle, setCurrentSubtitle] = useState(-1);
    89→  const [hlsSubtitles, setHlsSubtitles] = useState<any[]>([]);
    90→  const [playbackRate, setPlaybackRate] = useState(1);
    91→  const [showSkipIntro, setShowSkipIntro] = useState(false);
    92→  const [showSkipOutro, setShowSkipOutro] = useState(false);
    93→  // Track when user clicked skip — prevents the button from re-appearing
    94→  // immediately after the click (the timeupdate loop would otherwise keep
    95→  // showing it because video.currentTime is still in the skip range).
    96→  const skipIntroClickedRef = useRef(0);
    97→  const skipOutroClickedRef = useRef(0);
    98→  const [hoverTime, setHoverTime] = useState<number | null>(null);
    99→  const [hoverX, setHoverX] = useState(0);
   100→  const [seekRipple, setSeekRipple] = useState<{ x: number; dir: 'left' | 'right' } | null>(null);
   101→  const [volumeHover, setVolumeHover] = useState(false);
   102→  const [screenshotFlash, setScreenshotFlash] = useState(false);
   103→  const [screenshotToast, setScreenshotToast] = useState<string | null>(null);
   104→  const [downloadLoading, setDownloadLoading] = useState(false);
   105→  const [downloadToast, setDownloadToast] = useState<string | null>(null);
   106→  const [streamHealth, setStreamHealth] = useState<'good' | 'fair' | 'poor' | 'unknown'>('unknown');
   107→  const [showStreamInfo, setShowStreamInfo] = useState(false);
   108→  const controlsTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   109→  const retryCountRef = useRef(0);
   110→  const lastTapRef = useRef<{ time: number; x: number } | null>(null);
   111→  const canvasRef = useRef<HTMLCanvasElement>(null);
   112→
   113→  // ─── Load stream ──────────────────────────────────────────────────
   114→  useEffect(() => {
   115→    const video = videoRef.current;
   116→    if (!video || !url) return;
   117→
   118→    setLoading(true);
   119→    setError(null);
   120→    retryCountRef.current = 0;
   121→    setCurrentSubtitle(-1);
   122→    setHlsSubtitles([]);
   123→    onCanPlayFiredRef.current = false; // reset for new stream
   124→
   125→    if (hlsRef.current) {
   126→      hlsRef.current.destroy();
   127→      hlsRef.current = null;
   128→    }
   129→
   130→    if (sourceType === 'mp4' || url.endsWith('.mp4') || url.includes('video.mp4')) {
   131→      video.src = url;
   132→      video.load();
   133→      if (autoplay) video.play().catch(() => {});
   134→      return;
   135→    }
   136→
   137→    if (Hls.isSupported()) {
   138→      const hls = new Hls({
   139→        enableWorker: true,
   140→        lowLatencyMode: false,
   141→        backBufferLength: 90,
   142→        maxBufferLength: 90,
   143→        maxMaxBufferLength: 180,
   144→        maxBufferSize: 90 * 1000 * 1000,
   145→        maxBufferHole: 0.5,
   146→        // Start at the HIGHEST quality level (1080p), not lowest (360p).
   147→        // -1 = auto-select based on bandwidth, but we set a high default estimate
   148→        // so it picks the best quality immediately on fast connections.
   149→        startLevel: -1,
   150→        // Assume 10Mbps bandwidth by default — this makes hls.js start at 1080p
   151→        // instead of 360p. The ABR controller will adjust down if needed.
   152→        abrEwmaDefaultEstimate: 10000000,
   153→        abrBandWidthFactor: 0.95,
   154→        abrBandWidthUpFactor: 0.7,
   155→        maxStarvationDelay: 4,
   156→        abrEwmaDefaultEstimateMax: 20000000,
   157→        manifestLoadingTimeOut: 20000,
   158→        manifestLoadingMaxRetry: 4,
   159→        levelLoadingTimeOut: 20000,
   160→        levelLoadingMaxRetry: 4,
   161→        fragLoadingTimeOut: 45000,
   162→        fragLoadingMaxRetry: 8,
   163→        fragLoadingRetryDelay: 1000,
   164→        enableWebVTT: true,
   165→        xhrSetup: (xhr) => { xhr.withCredentials = false; },
   166→      });
   167→      hlsRef.current = hls;
   168→      hls.loadSource(url);
   169→      hls.attachMedia(video);
   170→
   171→      hls.on(Hls.Events.MANIFEST_PARSED, (_e, data) => {
   172→        setQualities(data.levels);
   173→        setLoading(false);
   174→        if (autoplay) {
   175→          video.play().catch(() => {
   176→            video.muted = true;
   177→            video.play().catch(() => {});
   178→          });
   179→        }
   180→      });
   181→
   182→      hls.on(Hls.Events.SUBTITLE_TRACKS_UPDATED, (_e, data) => {
   183→        setHlsSubtitles(data.subtitleTracks || []);
   184→        if (data.subtitleTracks.length > 0 && hls.subtitleTrack === -1) {
   185→          hls.subtitleTrack = 0;
   186→          setCurrentSubtitle(0);
   187→          // After HLS.js sets the subtitle track, it creates a textTrack with
   188→          // mode='showing' (native rendering). We need to switch it to 'hidden'
   189→          // so only our CustomSubtitleOverlay renders (prevents double subtitles).
   190→          setTimeout(() => {
   191→            const video = videoRef.current;
   192→            if (video) {
   193→              for (let i = 0; i < video.textTracks.length; i++) {
   194→                if (i === 0) video.textTracks[i].mode = 'hidden';
   195→              }
   196→            }
   197→          }, 200);
   198→        }
   199→      });
   200→
   201→      hls.on(Hls.Events.LEVEL_SWITCHED, (_e, data) => {
   202→        setCurrentQuality(data.level);
   203→        // After a level switch, HLS.js may re-enable native subtitle rendering.
   204→        // Force the active track back to 'hidden' mode so only our overlay shows.
   205→        if (currentSubtitle >= 0) {
   206→          setTimeout(() => {
   207→            const video = videoRef.current;
   208→            if (video && video.textTracks.length > currentSubtitle) {
   209→              video.textTracks[currentSubtitle].mode = 'hidden';
   210→            }
   211→          }, 100);
   212→        }
   213→      });
   214→
   215→      hls.on(Hls.Events.ERROR, (_e, data) => {
   216→        // Non-fatal errors are expected during streaming (a segment/playlist
   217→        // fetch hiccup) and hls.js already recovers from them on its own —
   218→        // console.warn so they don't trip Next's dev error overlay for
   219→        // something that isn't actually broken.
   220→        const log = data.fatal ? console.error : console.warn;
   221→        log(`[HLS] Error: type=${data.type} details=${data.details} fatal=${data.fatal}`);
   222→        if (data.fatal) {
   223→          if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
   224→            if (retryCountRef.current < 3) {
   225→              retryCountRef.current++;
   226→              setTimeout(() => hls.startLoad(), 2000);
   227→            } else {
   228→              setError('Stream failed. Try another server.');
   229→              setLoading(false);
   230→              onProviderFailedRef.current?.();
   231→            }
   232→          } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
   233→            hls.recoverMediaError();
   234→          } else {
   235→            setError('Playback error. Try another server.');
   236→            setLoading(false);
   237→            onProviderFailedRef.current?.();
   238→          }
   239→        }
   240→      });
   241→    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
   242→      video.src = url;
   243→      if (autoplay) video.play().catch(() => {});
   244→    } else {
   245→      setError('HLS not supported');
   246→      setLoading(false);
   247→    }
   248→
   249→    return () => {
   250→      if (hlsRef.current) {
   251→        hlsRef.current.destroy();
   252→        hlsRef.current = null;
   253→      }
   254→    };
   255→  }, [url, sourceType, autoplay]);
   256→
   257→  // ─── Video events ─────────────────────────────────────────────────
   258→  useEffect(() => {
   259→    const video = videoRef.current;
   260→    if (!video) return;
   261→
   262→    const onPlay = () => setPlaying(true);
   263→    const onPause = () => setPlaying(false);
   264→    const onTime = () => {
   265→      setCurrentTime(video.currentTime);
   266→      if (video.buffered.length > 0) {
   267→        setBuffered(video.buffered.end(video.buffered.length - 1));
   268→      }
   269→      // Auto Skip — when enabled, jump straight past the opening/ending.
   270→      if (autoSkipRef.current) {
   271→        if (intro && video.currentTime >= intro.start && video.currentTime < intro.end - 0.3) {
   272→          video.currentTime = intro.end;
   273→          return;
   274→        }
   275→        if (outro && video.currentTime >= outro.start && video.currentTime < outro.end - 0.3) {
   276→          video.currentTime = Math.min(outro.end, video.duration || outro.end);
   277→          return;
   278→        }
   279→      }
   280→      // Show skip intro button from 3s BEFORE intro starts until 5s AFTER it ends.
   281→      // Also show immediately if video just loaded and we're already in the intro range
   282→      // (happens when mimi loads super fast and starts playing at 0s, which is before intro.start)
   283→      //
   284→      // BUT: if the user just clicked Skip Intro (within last 8s), DON'T re-show the
   285→      // button — otherwise it instantly pops back up because video.currentTime is still
   286→      // in the intro range right after the seek.
   287→      const sinceIntroClick = Date.now() - skipIntroClickedRef.current;
   288→      const sinceOutroClick = Date.now() - skipOutroClickedRef.current;
   289→
   290→      if (intro && video.currentTime >= (intro.start - 3) && video.currentTime < (intro.end + 5) && sinceIntroClick > 8000) {
   291→        setShowSkipIntro(true);
   292→      } else {
   293→        setShowSkipIntro(false);
   294→      }
   295→      if (outro && video.currentTime >= (outro.start - 3) && video.currentTime < (outro.end + 10) && sinceOutroClick > 8000) {
   296→        setShowSkipOutro(true);
   297→      } else {
   298→        setShowSkipOutro(false);
   299→      }
   300→    };
   301→
   302→    // Also check on MANIFEST_PARSED / loadedmetadata — if intro data is available
   303→    // and we're at position 0, show the skip button immediately (before timeupdate fires)
   304→    const onLoadedMetadata = () => {
   305→      setDuration(video.duration || 0);
   306→      // If intro starts soon (within first 60s), show the button immediately
   307→      if (intro && intro.start < 60 && video.currentTime < intro.end) {
   308→        setShowSkipIntro(true);
   309→      }
   310→      if (outro && video.duration && outro.start < video.duration) {
   311→        // Don't show outro yet, just confirm we have the data
   312→      }
   313→    };
   314→    const onDur = () => setDuration(video.duration || 0);
   315→
   316→    let waitingTimer: ReturnType<typeof setTimeout> | null = null;
   317→    const onWaiting = () => {
   318→      if (waitingTimer) clearTimeout(waitingTimer);
   319→      // Show loading after 1.5s of waiting (was 5s — too long, user gets confused)
   320→      waitingTimer = setTimeout(() => {
   321→        if (video.readyState < 3) setLoading(true);
   322→      }, 1500);
   323→    };
   324→    const onPlaying = () => {
   325→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   326→      setLoading(false);
   327→    };
   328→    const onCanPlay = () => {
   329→      if (waitingTimer) { clearTimeout(waitingTimer); waitingTimer = null; }
   330→      setLoading(false);
   331→      // Notify parent that video is ready to play (for loading screen dismissal)
   332→      if (!onCanPlayFiredRef.current) {
   333→        onCanPlayFiredRef.current = true;
   334→        onCanPlayRef.current?.();
   335→      }
   336→    };
   337→    const onEnd = () => onEndedRef.current?.();
   338→    const onErr = () => { setError('Playback error.'); setLoading(false); };
   339→
   340→    video.addEventListener('play', onPlay);
   341→    video.addEventListener('pause', onPause);
   342→    video.addEventListener('timeupdate', onTime);
   343→    video.addEventListener('durationchange', onDur);
   344→    video.addEventListener('loadedmetadata', onLoadedMetadata);
   345→    video.addEventListener('waiting', onWaiting);
   346→    video.addEventListener('playing', onPlaying);
   347→    video.addEventListener('canplay', onCanPlay);
   348→    video.addEventListener('ended', onEnd);
   349→    video.addEventListener('error', onErr);
   350→
   351→    return () => {
   352→      video.removeEventListener('play', onPlay);
   353→      video.removeEventListener('pause', onPause);
   354→      video.removeEventListener('timeupdate', onTime);
   355→      video.removeEventListener('durationchange', onDur);
   356→      video.removeEventListener('loadedmetadata', onLoadedMetadata);
   357→      video.removeEventListener('waiting', onWaiting);
   358→      video.removeEventListener('playing', onPlaying);
   359→      video.removeEventListener('canplay', onCanPlay);
   360→      video.removeEventListener('ended', onEnd);
   361→      video.removeEventListener('error', onErr);
   362→    };
   363→  }, [intro, outro]);
   364→
   365→  // ─── Controls auto-hide ───────────────────────────────────────────
   366→  const showControlsTemp = useCallback(() => {
   367→    setShowControls(true);
   368→    if (controlsTimeoutRef.current) clearTimeout(controlsTimeoutRef.current);
   369→    controlsTimeoutRef.current = setTimeout(() => {
   370→      if (playing) { setShowControls(false); setActiveMenu(null); }
   371→    }, 3500);
   372→  }, [playing]);
   373→
   374→  // ─── Fullscreen ───────────────────────────────────────────────────
   375→  useEffect(() => {
   376→    const onFs = () => setFullscreen(!!document.fullscreenElement);
   377→    document.addEventListener('fullscreenchange', onFs);
   378→    return () => document.removeEventListener('fullscreenchange', onFs);
   379→  }, []);
   380→
   381→  const toggleFullscreen = async () => {
   382→    try {
   383→      if (document.fullscreenElement) {
   384→        await document.exitFullscreen();
   385→        // Release orientation lock when exiting fullscreen
   386→        try { await (screen.orientation as any)?.unlock(); } catch {}
   387→      } else {
   388→        await containerRef.current?.requestFullscreen();
   389→        // On mobile, lock to landscape so the video fills the screen
   390→        try {
   391→          if (screen.orientation && typeof (screen.orientation as any).lock === 'function') {
   392→            await (screen.orientation as any).lock('landscape');
   393→          }
   394→        } catch {
   395→          // Orientation lock not supported or denied — fullscreen still works
   396→        }
   397→      }
   398→    } catch {
   399→      // Fullscreen request failed silently
   400→    }
   401→  };
   402→
   403→  const seek = (time: number) => { if (videoRef.current) videoRef.current.currentTime = time; };
   404→  const togglePlay = () => {
   405→    const v = videoRef.current; if (!v) return;
   406→    if (v.paused) v.play(); else v.pause();
   407→  };
   408→  const toggleMute = () => {
   409→    const v = videoRef.current; if (!v) return;
   410→    v.muted = !v.muted; setMuted(v.muted);
   411→  };
   412→  const changeVolume = (vol: number) => {
   413→    const v = videoRef.current; if (!v) return;
   414→    v.volume = vol; v.muted = vol === 0;
   415→    setVolume(vol); setMuted(vol === 0);
   416→  };
   417→  const changeQuality = (level: number) => {
   418→    if (hlsRef.current) { hlsRef.current.currentLevel = level; setCurrentQuality(level); }
   419→    setActiveMenu(null);
   420→  };
   421→
   422→  // ── Auto-enable the first external subtitle track ──
   423→  // The `default` attribute on <track> doesn't always work reliably across
   424→  // browsers (especially when the track loads after the video starts). This
   425→  // effect explicitly enables the first external subtitle track once the
   426→  // video metadata has loaded AND there are no HLS-embedded subtitles.
   427→  // This ensures subtitles show automatically when the user starts watching.
   428→  useEffect(() => {
   429→    if (hlsSubtitles.length > 0) return; // HLS-embedded subs handle their own default
   430→    if (!subtitleTracks || subtitleTracks.length === 0) return;
   431→    const video = videoRef.current;
   432→    if (!video) return;
   433→
   434→    const enableFirstSub = () => {
   435→      // video.textTracks includes both HLS-embedded and external <track> elements.
   436→      // External tracks come AFTER HLS tracks in the list. Since we have no HLS
   437→      // subs (checked above), the first external track is at index 0.
   438→      if (video.textTracks.length > 0 && currentSubtitle === -1) {
   439→        // Don't override if the user explicitly turned subs off
   440→        // Use 'hidden' mode — cues fire but native rendering is suppressed
   441→        // (our CustomSubtitleOverlay handles rendering)
   442→        video.textTracks[0].mode = 'hidden';
   443→        setCurrentSubtitle(0);
   444→      }
   445→    };
   446→
   447→    // Try immediately (in case metadata already loaded)
   448→    if (video.readyState >= 1) {
   449→      enableFirstSub();
   450→    } else {
   451→      video.addEventListener('loadedmetadata', enableFirstSub, { once: true });
   452→      return () => video.removeEventListener('loadedmetadata', enableFirstSub);
   453→    }
   454→  }, [hlsSubtitles.length, subtitleTracks, currentSubtitle]);
   455→
   456→  // ── Force textTracks to 'hidden' mode (prevents double subtitles) ──
   457→  // HLS.js periodically sets textTrack.mode back to 'showing' (native rendering),
   458→  // which causes our CustomSubtitleOverlay + native ::cue to both render.
   459→  // This poll checks every 500ms and forces any 'showing' track to 'hidden'.
   460→  useEffect(() => {
   461→    const interval = setInterval(() => {
   462→      const video = videoRef.current;
   463→      if (!video) return;
   464→      for (let i = 0; i < video.textTracks.length; i++) {
   465→        const track = video.textTracks[i];
   466→        if (track.mode === 'showing') {
   467→          track.mode = 'hidden';
   468→        }
   469→      }
   470→    }, 500);
   471→    return () => clearInterval(interval);
   472→  }, []);
   473→
   474→  const changeSubtitle = (track: number) => {
   475→    if (track === -1) {
   476→      if (hlsRef.current) hlsRef.current.subtitleTrack = -1;
   477→      const video = videoRef.current;
   478→      if (video) for (let i = 0; i < video.textTracks.length; i++) video.textTracks[i].mode = 'disabled';
   479→      setCurrentSubtitle(-1);
   480→    } else if (hlsRef.current && track < hlsSubtitles.length) {
   481→      // HLS-embedded subtitle track
   482→      hlsRef.current.subtitleTrack = track;
   483→      const video = videoRef.current;
   484→      // Set the selected track to 'hidden' (cues fire, native rendering suppressed)
   485→      // and disable all others. This prevents double subtitles.
   486→      if (video) {
   487→        setTimeout(() => {
   488→          for (let i = 0; i < video.textTracks.length; i++) {
   489→            video.textTracks[i].mode = (i === track) ? 'hidden' : 'disabled';
   490→          }
   491→        }, 100);
   492→      }
   493→      setCurrentSubtitle(track);
   494→    } else {
   495→      // External subtitle (<track> element)
   496→      if (hlsRef.current) hlsRef.current.subtitleTrack = -1;
   497→      const video = videoRef.current;

  1193→// ─── Download Modal ─────────────────────────────────────────────
  1194→function DownloadModal({ animeId, episodeNum, onClose }: { animeId: string; episodeNum: number; onClose: () => void }) {
  1195→  const [loading, setLoading] = useState(true);
  1196→  const [links, setLinks] = useState<Array<{ text: string; decodedUrl: string }>>([]);
  1197→  const [error, setError] = useState('');
  1198→
  1199→  useEffect(() => {
  1200→    const fetchDownloads = async () => {
  1201→      try {
  1202→        setLoading(true);
  1203→        setError('');
  1204→        let title = '';
  1205→        try {
  1206→          const titleRes = await fetch('https://graphql.anilist.co', {
  1207→            method: 'POST',
  1208→            headers: { 'Content-Type': 'application/json' },
  1209→            body: JSON.stringify({
  1210→              query: `query($id:Int){Media(id:$id,type:ANIME){title{english romaji}}}`,
  1211→              variables: { id: parseInt(animeId) },
  1212→            }),
  1213→          });
  1214→          if (titleRes.ok) {
  1215→            const titleData = await titleRes.json();
  1216→            title = titleData?.data?.Media?.title?.english || titleData?.data?.Media?.title?.romaji || '';
  1217→          }
  1218→        } catch { /* ignore */ }
  1219→
  1220→        if (!title) {
  1221→          setError('Could not determine anime title');
  1222→          setLoading(false);
  1223→          return;
  1224→        }
  1225→
  1226→        const res = await fetch(`/api/anime/download?title=${encodeURIComponent(title)}&auto=1`);
  1227→        if (!res.ok) {
  1228→          setError('Failed to fetch download links');
  1229→          setLoading(false);
  1230→          return;
  1231→        }
  1232→        const data = await res.json();
  1233→        if (data.links && data.links.length > 0) {
  1234→          setLinks(data.links);
  1235→        } else {
  1236→          setError('No download links found for this anime');
  1237→        }
  1238→      } catch (e: any) {
  1239→        setError(e?.message || 'Failed to load downloads');
  1240→      } finally {
  1241→        setLoading(false);
  1242→      }
  1243→    };
