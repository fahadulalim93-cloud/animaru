    49→function useMembers(refreshKey: number) {
    50→  const [members, setMembers] = useState<Member[] | null>(null);
    51→  const [loading, setLoading] = useState(true);
    52→  const [error, setError] = useState<string | null>(null);
    53→  const [bump, setBump] = useState(0);
    54→
    55→  useEffect(() => {
    56→    let cancel = false;
    57→    Promise.resolve().then(async () => {
    58→      const token = getAdminToken();
    59→      if (!token) throw new Error("No admin token");
    60→      const r = await fetch("/api/admin/moderation", { headers: { "x-admin-token": token }, cache: "no-store" });
    61→      return r.json();
    62→    }).then((d) => {
    63→      if (cancel) return;
    64→      if (d.ok) { setMembers(d.members); setError(null); }
    65→      else setError(d.error || "Failed to load members");
    66→    }).catch((e) => { if (!cancel) setError(e instanceof Error ? e.message : "Failed to load members"); })
    67→      .finally(() => { if (!cancel) setLoading(false); });
    68→    return () => { cancel = true; };
    69→  }, [refreshKey, bump]);
    70→
    71→  const mutate = useCallback(async (username: string, action: "ban" | "unban" | "suspend" | "unsuspend" | "promote" | "demote", opts?: { reason?: string; hours?: number }) => {
    72→    const token = getAdminToken();
    73→    if (!token) return { ok: false, error: "No admin token" };
    74→    const res = await fetch("/api/admin/moderation", {
    75→      method: "POST",
    76→      headers: { "content-type": "application/json", "x-admin-token": token },
    77→      body: JSON.stringify({ username, action, ...opts }),
    78→    }).then((r) => r.json()).catch(() => ({ ok: false, error: "Network error" }));
    79→    if (res.ok) setBump((b) => b + 1);
    80→    return res;
    81→  }, []);
    82→
    83→  return { members, loading, error, mutate };
    84→}

    86→function useStats(range: number, refreshKey: number) {
    87→  const [data, setData] = useState<Stats | null>(null);
    88→  const [loading, setLoading] = useState(true);
    89→  useEffect(() => {
    90→    let cancel = false;
    91→    fetch(`/api/analytics/stats?days=${range}`, { cache: "no-store" })
    92→      .then((r) => r.json())
    93→      .then((d: Stats) => { if (!cancel) { setData(d); setLoading(false); } })
    94→      .catch(() => { if (!cancel) setLoading(false); });
    95→    return () => { cancel = true; };
    96→  }, [range, refreshKey]);
    97→  return { data, loading };
    98→}

   350→function Delta({ v, invert = false }: { v: number | null; invert?: boolean }) {
   351→  if (v === null) return <span className="text-[11px] text-white/30">— new</span>;
   352→  const good = invert ? v <= 0 : v >= 0;
   353→  return (
   354→    <span className={`text-[11px] font-bold flex items-center gap-0.5 ${good ? "text-emerald-400" : "text-red-400"}`}>
   355→      {v >= 0 ? "▲" : "▼"} {Math.abs(v)}%
   356→    </span>
   357→  );

   374→function Sparkline({ data, accent }: { data: number[]; accent: string }) {
   375→  const max = Math.max(1, ...data);
   376→  const pts = data.map((d, i) => `${(i / (data.length - 1)) * 100},${28 - (d / max) * 24}`).join(" ");
   377→  return (
   378→    <svg viewBox="0 0 100 28" preserveAspectRatio="none" className="w-full h-6 mt-2">
   379→      <polyline points={pts} fill="none" stroke={accent} strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
   380→    </svg>
   381→  );
   382→}

   384→function AreaChart({ series, accent }: { series: { day: number; views: number; sessions: number }[]; accent: string }) {
   385→  const W = 700, H = 220, P = 8;
   386→  const max = Math.max(1, ...series.map((s) => Math.max(s.views, s.sessions)));
   387→  const x = (i: number) => P + (i / Math.max(1, series.length - 1)) * (W - 2 * P);
   388→  const y = (v: number) => H - P - (v / max) * (H - 2 * P - 16);
   389→  const line = (key: "views" | "sessions") => series.map((s, i) => `${i === 0 ? "M" : "L"}${x(i)},${y(s[key])}`).join(" ");
   390→  const area = `${line("views")} L${x(series.length - 1)},${H - P} L${x(0)},${H - P} Z`;
   391→  return (
   392→    <div className="w-full overflow-hidden">
   393→      <svg viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" className="w-full h-56">
   394→        <defs>
   395→          <linearGradient id="areaFill" x1="0" y1="0" x2="0" y2="1">
   396→            <stop offset="0%" stopColor={accent} stopOpacity="0.35" />
   397→            <stop offset="100%" stopColor={accent} stopOpacity="0" />
   398→          </linearGradient>
   399→        </defs>
   400→        {[0.25, 0.5, 0.75, 1].map((g) => (
   401→          <line key={g} x1={P} x2={W - P} y1={H - P - g * (H - 2 * P - 16)} y2={H - P - g * (H - 2 * P - 16)} stroke="rgba(255,255,255,0.05)" strokeWidth="1" />
   402→        ))}
   403→        <path d={area} fill="url(#areaFill)" />
   404→        <path d={line("views")} fill="none" stroke={accent} strokeWidth="2" vectorEffect="non-scaling-stroke" />
   405→        <path d={line("sessions")} fill="none" stroke="#22D3EE" strokeWidth="2" strokeDasharray="4 3" vectorEffect="non-scaling-stroke" />
   406→      </svg>
   407→      <div className="flex items-center gap-4 mt-2 text-[11px] text-white/45 px-1">
   408→        <span className="flex items-center gap-1.5"><span className="w-3 h-0.5 rounded" style={{ backgroundColor: accent }} /> Page views</span>
   409→        <span className="flex items-center gap-1.5"><span className="w-3 h-0.5 rounded bg-[#22D3EE]" /> Sessions</span>
   410→        <span className="ml-auto">{new Date(series[0]?.day).toLocaleDateString(undefined, { month: "short", day: "numeric" })} – today</span>
   411→      </div>
   412→    </div>
   413→  );

   416→function Donut({ data, accent }: { data: { label: string; value: number; color: string }[]; accent: string }) {
   417→  const total = data.reduce((s, d) => s + d.value, 0);
   418→  if (total === 0) return <p className="text-sm text-white/40 text-center py-8">No data yet.</p>;
   419→  const R = 42, C = 2 * Math.PI * R;
   420→  const segments = data.reduce<{ d: typeof data[number]; start: number }[]>((acc, d) => {
   421→    const prev = acc.length ? acc[acc.length - 1] : null;
   422→    const start = prev ? prev.start + prev.d.value / total : 0;
   423→    return [...acc, { d, start }];
   424→  }, []);
   425→  return (
   426→    <div className="flex items-center gap-5">
   427→      <svg viewBox="0 0 100 100" className="w-28 h-28 shrink-0 -rotate-90">
   428→        <circle cx="50" cy="50" r={R} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth="12" />
   429→        {segments.map(({ d, start }, i) => (
   430→          <circle key={i} cx="50" cy="50" r={R} fill="none" stroke={d.color} strokeWidth="12" strokeDasharray={`${(d.value / total) * C} ${C}`} strokeDashoffset={-start * C} />
   431→        ))}
   432→      </svg>
   433→      <div className="flex-1 min-w-0 space-y-1.5">
   434→        {data.map((d) => (
   435→          <div key={d.label} className="flex items-center gap-2 text-xs">
   436→            <span className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ backgroundColor: d.color }} />
   437→            <span className="text-white/60 flex-1 truncate">{d.label}</span>
   438→            <span className="font-bold tabular-nums">{Math.round((d.value / total) * 100)}%</span>
   439→          </div>
   440→        ))}
   441→      </div>
   442→    </div>
   443→  );
   444→}
